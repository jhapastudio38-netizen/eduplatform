   257→@Composable
   258→fun BottomNavBar(activeTab: BottomTab, onTabClick: (BottomTab) -> Unit) {
   259→    Surface(
   260→        color = CardWhite,
   261→        shadowElevation = 8.dp,
   262→        tonalElevation = 0.dp,
   263→    ) {
   264→        Row(
   265→            modifier = Modifier
   266→                .fillMaxWidth()
   267→                .navigationBarsPadding()
   268→                .padding(vertical = 8.dp),
   269→            horizontalArrangement = Arrangement.SpaceEvenly,
   270→            verticalAlignment = Alignment.CenterVertically,
   271→        ) {
   272→            BottomTab.values().forEach { tab ->
   273→                val isActive = tab == activeTab
   274→                val iconColor = if (isActive) NavyBlue else Color(0xFF94A3B8)
   275→                val textColor = if (isActive) NavyBlue else Color(0xFF94A3B8)
   276→
   277→                // Animated scale on press
   278→                var pressed by remember { mutableStateOf(false) }
   279→                val scale by animateFloatAsState(
   280→                    targetValue = if (pressed) 0.85f else 1f,
   281→                    animationSpec = spring(dampingRatio = Spring.DampingRatioMediumBouncy),
   282→                    label = "navScale"
   283→                )
   284→                // Animated background for active tab
   285→                val bgAlpha by animateFloatAsState(
   286→                    targetValue = if (isActive) 1f else 0f,
   287→                    animationSpec = tween(300, easing = FastOutSlowInEasing),
   288→                    label = "navBgAlpha"
   289→                )
   290→
   291→                Box(
   292→                    modifier = Modifier
   293→                        .clip(RoundedCornerShape(16.dp))
   294→                        .background(NavyBlue.copy(alpha = 0.08f * bgAlpha))
   295→                        .clickable { onTabClick(tab) }
   296→                        .padding(horizontal = 16.dp, vertical = 6.dp)
   297→                        .scale(scale),
   298→                    contentAlignment = Alignment.Center,
   299→                ) {
   300→                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
   301→                        Icon(
   302→                            tab.icon,
   303→                            tab.label,
   304→                            tint = iconColor,
   305→                            modifier = Modifier.size(24.dp),
   306→                        )
   307→                        Spacer(Modifier.height(3.dp))
   308→                        Text(
   309→                            tab.label,
   310→                            color = textColor,
   311→                            fontSize = 10.sp,
   312→                            fontWeight = if (isActive) FontWeight.SemiBold else FontWeight.Normal,
   313→                        )
   314→                    }
   315→                }
   316→            }
   317→        }
   318→    }
   319→}

   349→// Async image loader — uses Coil
   350→@Composable
   351→fun AsyncImageLoader(url: String, modifier: Modifier = Modifier) {
   352→    val absoluteUrl = if (url.isBlank()) "" else if (url.startsWith("http")) url else "https://my-project-five-sepia.vercel.app$url"
   353→    if (absoluteUrl.isBlank()) {
   354→        Box(modifier = modifier.background(NavyBlue.copy(alpha = 0.05f)), contentAlignment = Alignment.Center) {
   355→            Icon(Icons.Default.Image, null, tint = NavyBlue.copy(alpha = 0.2f), modifier = Modifier.size(32.dp))
   356→        }
   357→        return
   358→    }
   359→    // Use Crop so the image fills the entire card — no empty space, no "cut" look
   360→    coil.compose.AsyncImage(model = absoluteUrl, contentDescription = null, modifier = modifier, contentScale = ContentScale.Crop)
   361→}

   363→// ─── HOME SCREEN — Professional EdTech design ─────────────────────────────────
   364→@Composable
   365→fun HomeScreen(theme: AppTheme, sound: SoundManager, userName: String, onNavigate: (Screen) -> Unit) {
   366→    var stats by remember { mutableStateOf<UserStats?>(null) }
   367→    var homeCards by remember {
   368→        mutableStateOf<List<HomeCard>>(AppState.getCachedNow<List<HomeCard>>(AppState.KEY_HOME_CARDS) ?: emptyList())
   369→    }
   370→    var loading by remember { mutableStateOf(homeCards.isEmpty()) }
   371→    val configuration = androidx.compose.ui.platform.LocalConfiguration.current
   372→    val isLandscape = configuration.screenWidthDp > configuration.screenHeightDp
   373→    // Adapt column count to orientation — 2 cols portrait, 3-4 cols landscape
   374→    val cardColumns = when {
   375→        isLandscape && configuration.screenWidthDp >= 900 -> 4
   376→        isLandscape -> 3
   377→        else -> 2
   378→    }
   379→
   380→    // Real-time refresh: load on mount, and refresh every time the screen
   381→    // becomes visible (e.g. when user navigates back from another screen).
   382→    val lifecycleOwner = androidx.compose.ui.platform.LocalLifecycleOwner.current
   383→    var refreshTrigger by remember { mutableStateOf(0) }
   384→    LaunchedEffect(refreshTrigger) {
   385→        if (homeCards.isNotEmpty()) loading = false
   386→        try {
   387→            homeCards = AppState.cachedFresh(AppState.KEY_HOME_CARDS) {
   388→                AppState.api.getHomeCards().cards
   389→            }
   390→            stats = AppState.api.getStats().stats
   391→        } catch (_: Exception) {}
   392→        finally { loading = false }
   393→    }
   394→    // Refresh when the app comes to foreground (real-time updates)
   395→    DisposableEffect(lifecycleOwner) {
   396→        val observer = androidx.lifecycle.LifecycleEventObserver { _, event ->
   397→            if (event == androidx.lifecycle.Lifecycle.Event.ON_RESUME) {
   398→                refreshTrigger++ // triggers LaunchedEffect above to re-fetch
   399→            }
   400→        }
   401→        lifecycleOwner.lifecycle.addObserver(observer)
   402→        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
   403→    }
   404→
   405→    val effectiveCards = remember(homeCards, loading) {
   406→        if (loading) emptyList()
   407→        else if (homeCards.isNotEmpty()) {
   408→            // Merge server cards with guaranteed QBank + Batch cards
   409→            val serverKeys = homeCards.map { it.key }.toSet()
   410→            val serverRoutes = homeCards.map { it.route }.toSet()
   411→            val extras = mutableListOf<HomeCard>()
   412→            // Only add QBank card if server doesn't have one (by key OR by route)
   413→            if (!serverKeys.contains("qbank_packages") && !serverRoutes.contains("questionbank")) {
   414→                extras.add(HomeCard(key = "qbank_packages", title = "Question Bank", section = "test", sortOrder = 99, route = "questionbank", imageUrl = ""))
   415→            }
   416→            // Only add Batch card if server doesn't have one (by key OR by route)
   417→            if (!serverKeys.contains("batch_packages") && !serverRoutes.contains("batch")) {
   418→                extras.add(HomeCard(key = "batch_packages", title = "Batch", section = "test", sortOrder = 100, route = "batch", imageUrl = ""))
   419→            }
   420→            homeCards + extras
   421→        } else listOf(
   422→            HomeCard(key = "ubt_test", title = "UBT Test", section = "test", sortOrder = 0, route = "tests", imageUrl = ""),
   423→            HomeCard(key = "demo_exam", title = "Demo Exam", section = "test", sortOrder = 1, route = "tests", imageUrl = ""),
   424→            HomeCard(key = "qbank_packages", title = "Question Bank", section = "test", sortOrder = 2, route = "questionbank", imageUrl = ""),
   425→            HomeCard(key = "batch_packages", title = "Batch", section = "test", sortOrder = 3, route = "batch", imageUrl = ""),
   426→            HomeCard(key = "results", title = "Results", section = "test", sortOrder = 5, route = "results", imageUrl = ""),
   427→            HomeCard(key = "all_books", title = "Books", section = "resources", sortOrder = 0, route = "books", imageUrl = ""),
   428→            HomeCard(key = "eye_vision", title = "Eye Vision", section = "resources", sortOrder = 2, route = "eyevision", imageUrl = ""),
   429→            HomeCard(key = "dictionary", title = "Dictionary", section = "resources", sortOrder = 4, route = "dictionary", imageUrl = ""),
   430→            HomeCard(key = "grammar", title = "Grammar", section = "resources", sortOrder = 5, route = "grammar", imageUrl = ""),
   431→            HomeCard(key = "alarms", title = "Alarms", section = "resources", sortOrder = 6, route = "alarms", imageUrl = ""),
   432→            HomeCard(key = "join", title = "Join Live", section = "resources", sortOrder = 3, route = "join", imageUrl = "")
   433→        )
   434→    }
   435→
   436→    if (loading) {
   437→        Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
   438→            CircularProgressIndicator(color = NavyBlue)
   439→        }
   440→        return
   441→    }
   442→
   443→    val sections = effectiveCards.groupBy { it.section }

   781→fun BooksScreen(theme: AppTheme, sound: SoundManager, onBack: () -> Unit, onBookClick: (Book) -> Unit = {}) {

   783→        mutableStateOf<List<Book>>(AppState.getCachedNow<List<Book>>(AppState.KEY_BOOKS) ?: emptyList())
   784→    }
   785→    var loading by remember { mutableStateOf(books.isEmpty()) }
   786→    var retryCount by remember { mutableStateOf(0) }
   787→
   788→    // Use cachedFresh so we ALWAYS hit the network when this screen mounts —
   789→    // the user expects to see new books the admin just published, not stale
   790→    // cache. Cache is still updated so getCachedNow() returns fresh data
   791→    // after a screen rotation.
   792→    LaunchedEffect(retryCount) {
   793→        if (books.isNotEmpty()) loading = false else loading = true
   794→        try {
   795→            books = AppState.cachedFresh(AppState.KEY_BOOKS) { AppState.api.getBooks().books }
   796→        } catch (_: Exception) {}
   797→        finally { loading = false }

   811→                AnimatedListItem(index = i, theme = theme) {

   852→// ─── Tests Screen ─────────────────────────────────────────────────────────────
   853→@Composable
   854→fun TestsScreen(theme: AppTheme, sound: SoundManager, filter: String = "all", title: String = "All Exams", onBack: () -> Unit, onStartExam: (String) -> Unit, onOpenPackages: () -> Unit = {}) {
   855→    // Local filter state — allows in-screen filtering via tabs
   856→    var activeFilter by remember { mutableStateOf(filter) }
   857→    var tests by remember {
   858→        mutableStateOf<List<TestItem>>(AppState.getCachedNow<List<TestItem>>(AppState.keyTests(activeFilter)) ?: emptyList())
   859→    }
   860→    var loading by remember { mutableStateOf(tests.isEmpty()) }
   861→    var error by remember { mutableStateOf("") }
   862→    var retryCount by remember { mutableStateOf(0) }
   863→
   864→    LaunchedEffect(activeFilter, retryCount) {
   865→        if (tests.isNotEmpty()) loading = false else loading = true
   866→        error = ""
   867→        try {
   868→            // Always fetch fresh — admin may have just published a new exam
   869→            AppState.invalidateCache(AppState.keyTests(activeFilter))
   870→            val result = kotlinx.coroutines.withTimeoutOrNull(20_000L) { AppState.getCachedTests(activeFilter) }
   871→            if (result != null) tests = result
   872→            else if (tests.isEmpty()) error = "The request timed out. Check your internet and try again."
   873→        } catch (e: retrofit2.HttpException) {
   874→            if (tests.isEmpty()) error = when (e.code()) { 401 -> "Your session has expired." else -> "Could not load (HTTP ${e.code()})." }
   875→        } catch (e: java.net.UnknownHostException) { if (tests.isEmpty()) error = "No internet connection." }
   876→        catch (e: java.net.SocketTimeoutException) { if (tests.isEmpty()) error = "The request timed out." }
   877→        catch (e: java.io.IOException) { if (tests.isEmpty()) error = "Network error." }
   878→        catch (e: Exception) { if (tests.isEmpty()) error = "Unexpected error." }
   879→        finally { loading = false }
   880→    }
   881→
   882→    if (loading) {
   883→        Column(Modifier.fillMaxSize()) {
   884→            LinearProgressIndicator(modifier = Modifier.fillMaxWidth(), color = NavyBlue, trackColor = NavyBlue.copy(alpha = 0.1f))
   885→            ScreenHeader(theme, sound, title, "Tap a test to start.", onBack)
   886→        }
   887→        return
   888→    }
   889→
   890→    if (error.isNotEmpty()) {
   891→        Column(Modifier.fillMaxSize().padding(24.dp), horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.Center) {
   892→            Icon(Icons.Default.CloudOff, null, tint = TextLight, modifier = Modifier.size(56.dp))
   893→            Spacer(Modifier.height(16.dp))
