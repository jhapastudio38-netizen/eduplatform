   815→/// Reference-style tab: flex:1, centered, blue bottom border (3dp) when active.
   816→@Composable
   817→private fun RefTab(label: String, active: Boolean, accent: Color, modifier: Modifier, onClick: () -> Unit) {
   818→    Column(
   819→        modifier = modifier
   820→            .fillMaxHeight()
   821→            .clickable { onClick() }
   822→            .background(if (active) Color(0xFFF2F2F2) else Color.White),
   823→        horizontalAlignment = Alignment.CenterHorizontally,
   824→        verticalArrangement = Arrangement.Center
   825→    ) {
   826→        Text(
   827→            label,
   828→            color = if (active) Color(0xFF111111) else Color(0xFF444444),
   829→            fontSize = 13.sp,
   830→            fontWeight = if (active) FontWeight.SemiBold else FontWeight.Normal
   831→        )
   832→        Spacer(Modifier.height(6.dp))
   833→        Box(
   834→            modifier = Modifier
