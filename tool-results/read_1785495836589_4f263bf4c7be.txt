    53→/**
    54→ * Exam taking screen — full flow:
    55→ * 1. Loading skeleton (with timeout + retry)
    56→ * 2. Question-by-question with audio (loops N times), image, options
    57→ * 3. Click sound on every interaction
    58→ * 4. Wrong → shows correct answer in red
    59→ * 5. Correct → green confirmation
    60→ * 6. Auto-advances to next question
    61→ * 7. Final score + review screen
    62→ * 8. Stats auto-update via /api/student/tests/[id]/submit
    63→ */
    64→@Composable
    65→fun ExamScreen(theme: AppTheme, testId: String, onExit: () -> Unit) {
    66→    val sound = rememberSoundManager()
    67→    val scope = rememberCoroutineScope()
    68→    val context = LocalContext.current
    69→    var test by remember { mutableStateOf<TestDetail?>(null) }
    70→    var loading by remember { mutableStateOf(true) }
    71→    var error by remember { mutableStateOf("") }
    72→    var currentIdx by remember { mutableStateOf(0) }
    73→    val answers = remember { mutableStateMapOf<String, Any>() }
    74→    // Persistent audio play counts per question ID — survives navigation.
    75→    // Prevents cheat where student navigates away and back to reset plays.
    76→    val audioPlayCounts = remember { mutableStateMapOf<String, Int>() }
    77→    var submitResult by remember { mutableStateOf<SubmitResponse?>(null) }
    78→    var submitting by remember { mutableStateOf(false) }
    79→    // Per-question feedback (after answering, before moving on)
    80→    var questionFeedback by remember { mutableStateOf<QuestionFeedback?>(null) }
    81→    // Timer
    82→    var timeLeft by remember { mutableStateOf(0) }
    83→
    84→    // ── ORIENTATION ── FORCE LANDSCAPE. No onDispose PORTRAIT —
    85→    // MainScreen handles portrait when screen changes to non-exam.
    86→    DisposableEffect(Unit) {
    87→        val activity = context as? Activity
    88→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    89→        onDispose { }
    90→    }
    91→    LaunchedEffect(Unit) {
    92→        kotlinx.coroutines.delay(100)
    93→        val activity = context as? Activity
    94→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    95→    }
    96→    // Retry trigger — increment to force reload
    97→    var retryCount by remember { mutableStateOf(0) }
    98→
    99→    // Load test detail — uses LaunchedEffect's own scope, finally block guarantees loading cleanup
   100→    // ─── Back button handler — show exit warning during exam ──────────────
   101→    var showExitDialog by remember { mutableStateOf(false) }
   102→    androidx.activity.compose.BackHandler(enabled = !loading && error.isEmpty()) {
   103→        showExitDialog = true
   104→    }
   105→    if (showExitDialog) {
   106→        androidx.compose.material3.AlertDialog(
   107→            onDismissRequest = { showExitDialog = false },
   108→            title = { Text("Exit Exam?") },
   109→            text = { Text("Your exam is in progress. Are you sure you want to exit? Your answers will be lost.") },
   110→            confirmButton = { Button(onClick = { showExitDialog = false; onExit() }) { Text("Exit") } },
   111→            dismissButton = { OutlinedButton(onClick = { showExitDialog = false }) { Text("Continue Exam") } }
   112→        )
   113→    }
   114→
   115→    LaunchedEffect(testId, retryCount) {
   116→        loading = true
   117→        error = ""
   118→        try {
   119→            // 30-second timeout — combined exams may need multiple requests
   120→            val result = withTimeoutOrNull(30_000L) {
   121→                when {
   122→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   123→                    testId == "qbank-combined" -> {
   124→                        try {
   125→                            AppState.api.getQBankCombined().test
   126→                        } catch (e: retrofit2.HttpException) {
   127→                            if (e.code() == 404) buildQBankCombinedClientSide()
   128→                            else throw e
   129→                        }
   130→                    }
   131→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   132→                    testId.startsWith("bundle-") -> {
   133→                        val bundleId = testId.removePrefix("bundle-")
   134→                        try {
   135→                            AppState.api.getBundleCombined(bundleId).test
   136→                        } catch (e: retrofit2.HttpException) {
   137→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   138→                            else throw e
   139→                        }
   140→                    }
   141→                    // Normal test — fetch by ID
   142→                    else -> AppState.api.getTestDetail(testId).test
   143→                }
   144→            }
   145→            if (result != null) {
   146→                test = result
   147→                timeLeft = (result.durationMin.coerceAtLeast(1)) * 60
   148→            } else {
   149→                error = "The request timed out. Check your internet connection and try again."
   150→            }
   151→        } catch (e: retrofit2.HttpException) {
   152→            // Try to read the actual error message from the response body
   153→            val rawBody = try { e.response()?.errorBody()?.string() } catch (_: Exception) { null }
   154→            error = when (e.code()) {
   155→                401 -> "Your session has expired. Please log out and sign in again."
   156→                403 -> "This exam is not available right now."
   157→                404 -> "This test could not be found. It may have been removed."
   158→                500 -> "Server error. Please try again in a moment.${if (rawBody != null) " ($rawBody)" else ""}"
   159→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   160→            }
   161→        } catch (e: java.net.UnknownHostException) {
   162→            error = "No internet connection. Please check your network and try again."
   163→        } catch (e: java.net.SocketTimeoutException) {
   164→            error = "The request timed out. Please try again."
   165→        } catch (e: java.io.IOException) {
   166→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   167→        } catch (e: Exception) {
   168→            error = "Unexpected error: ${e.message ?: "Please try again."}"
   169→        } finally {
   170→            // GUARANTEED: loading is always reset, even if the coroutine is cancelled
   171→            loading = false
   172→        }
   173→    }
   174→
   175→    // Countdown timer
   176→    LaunchedEffect(test, submitResult) {
   177→        if (test != null && submitResult == null) {
   178→            while (timeLeft > 0) {
   179→                delay(1000)
   180→                timeLeft--
   181→            }
   182→            // Auto-submit when timer reaches zero
   183→            val currentTest = test
   184→            if (timeLeft == 0 && currentTest != null && submitResult == null && !submitting) {
   185→                submitting = true
   186→                sound.swoosh()
   187→                try {
   188→                    // Use fallback for combined exams (qbank-combined / bundle-{id})
   189→                    submitResult = if (currentTest.id == "qbank-combined" || currentTest.id.startsWith("bundle-")) {
   190→                        submitCombinedExamWithFallback(currentTest, answers.toMap())
   191→                    } else {
   192→                        submitExamWithFallback(currentTest, answers.toMap())
   193→                    }
   194→                    sound.success()
   195→                } catch (e: java.net.UnknownHostException) {
   196→                    error = "No internet connection. Could not submit your answers."
   197→                } catch (e: java.io.IOException) {
   198→                    error = "Network error. Could not submit your answers."
   199→                } catch (e: Exception) {
   200→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   201→                }
   202→                submitting = false
   203→            }
   204→        }
   205→    }
   206→
   207→    if (loading) {
   208→        // Loading skeleton with a subtle "Loading..." label so users know it's working
   209→        Column(Modifier.fillMaxSize()) {
   210→            LinearProgressIndicator(
   211→                modifier = Modifier.fillMaxWidth(),
   212→                color = theme.primary,
   213→                trackColor = theme.primary.copy(alpha = 0.1f),
   214→            )
   215→            SkeletonListScreen(theme, itemCount = 4)
   216→        }
   217→        return
   218→    }
   219→
   220→    if (error.isNotEmpty()) {
   221→        Column(
   222→            Modifier.fillMaxSize().padding(24.dp),
   223→            horizontalAlignment = Alignment.CenterHorizontally,
   224→            verticalArrangement = Arrangement.Center
   225→        ) {
   226→            Icon(
   227→                Icons.Default.CloudOff,
   228→                null,
   229→                tint = theme.errorRed.copy(alpha = 0.7f),
   230→                modifier = Modifier.size(56.dp)
   231→            )
   232→            Spacer(Modifier.height(16.dp))
   233→            Text(
   234→                "Couldn't load the test",
   235→                color = theme.darkText,
   236→                fontSize = 18.sp,
   237→                fontWeight = FontWeight.Bold
   238→            )
   239→            Spacer(Modifier.height(8.dp))
   240→            Text(
   241→                error,
   242→                color = theme.subText,
   243→                fontSize = 13.sp,
   244→                textAlign = TextAlign.Center,
   245→                modifier = Modifier.padding(horizontal = 16.dp)
   246→            )
   247→            Spacer(Modifier.height(8.dp))
   248→            Text(
   249→                "Test ID: $testId",
   250→                color = theme.subText.copy(alpha = 0.5f),
   251→                fontSize = 10.sp,
   252→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   253→            )
   254→            Spacer(Modifier.height(24.dp))
   255→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   256→                OutlinedButton(
   257→                    onClick = onExit,
   258→                    shape = RoundedCornerShape(10.dp)
   259→                ) { Text("Go back") }
   260→                Button(
   261→                    onClick = {
   262→                        sound.click()
   263→                        retryCount++ // triggers LaunchedEffect reload
   264→                    },
   265→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   266→                    shape = RoundedCornerShape(10.dp)
   267→                ) {
   268→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   269→                    Spacer(Modifier.width(6.dp))
   270→                    Text("Retry")
   271→                }
   272→            }
   273→        }
   274→        return
   275→    }
   276→
   277→    val t = test ?: return
   278→    // Guard: if the test has no questions, show a friendly message instead of crashing
   279→    if (t.items.isEmpty()) {
   280→        Column(
   281→            Modifier.fillMaxSize().padding(20.dp),
   282→            horizontalAlignment = Alignment.CenterHorizontally,
   283→            verticalArrangement = Arrangement.Center
   284→        ) {
   285→            Icon(Icons.Default.Quiz, null, tint = theme.subText, modifier = Modifier.size(48.dp))
   286→            Spacer(Modifier.height(12.dp))
   287→            Text("This test has no questions yet.", color = theme.darkText, fontSize = 14.sp, textAlign = TextAlign.Center)
   288→            Spacer(Modifier.height(16.dp))
   289→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }
   290→        }
   291→        return
   292→    }
   293→
   294→    val currentQuestion = t.items.getOrNull(currentIdx)
   295→
   296→    // ─── Result screen ──────────────────────────────────────────────────────
   297→    if (submitResult != null) {
   298→        ExamResultScreen(theme, submitResult!!, onExit, sound, examTitle = t.title, examDescription = t.description)
   299→        return
   300→    }
   301→
   302→    // ═══ EXAM UI — spec-compliant landscape layout ═══
   303→    val item = t.items.getOrNull(currentIdx) ?: return
   304→    val q = item.question
   305→    val options = q.options ?: emptyList()
   306→    val textItems = t.items.filter { it.question.blockType == "text" }
   307→    val readingCount = if (textItems.isNotEmpty()) textItems.size else t.items.size
   308→    val listeningCount = if (textItems.isNotEmpty()) t.items.size - textItems.size else 0
   309→    val answeredCount = answers.size
   310→    val remainingCount = t.items.size - answeredCount
   311→    // Start with the grid view so the student can pick which question to answer
   312→    // first. They tap a number → answer → return to grid → pick next → submit.
   313→    var showGrid by remember { mutableStateOf(true) }
   314→
   315→    Column(modifier = Modifier.fillMaxSize().background(Color(0xFFF8FAFC))) {
   316→        // ── 1. TOP STATUS HEADER ── compact white bar with thin border
   317→        Surface(
   318→            color = Color.White,
   319→            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   320→        ) {
   321→            Row(modifier = Modifier.fillMaxWidth().height(36.dp), verticalAlignment = Alignment.CenterVertically) {
   322→                // Section type
   323→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   324→                    val sectionLabel = if (q.blockType == "audio") "Listening ($listeningCount)" else "Reading ($readingCount)"
   325→                    Text(sectionLabel, color = Color(0xFF003478), fontSize = 10.sp, fontWeight = FontWeight.Bold)
   326→                }
   327→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   328→                // Total
   329→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   330→                    Text("Total: ${t.items.size}", color = Color(0xFF64748B), fontSize = 10.sp)
   331→                }
   332→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   333→                // Remaining
   334→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   335→                    Text("Left: $remainingCount", color = Color(0xFF64748B), fontSize = 10.sp)
   336→                }
   337→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   338→                // Timer
   339→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   340→                    val mm = timeLeft / 60; val ss = timeLeft % 60
   341→                    Text(String.format("%02d:%02d", mm, ss), color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   342→                }
   343→            }
   344→        }
   345→
   346→        // ── 2. INSTRUCTION ROW ── compact question number + title + FREE badge
   347→        Surface(modifier = Modifier.fillMaxWidth().padding(horizontal = 6.dp, vertical = 2.dp)) {
   348→            Row(verticalAlignment = Alignment.CenterVertically) {
   349→                Text("${currentIdx + 1}. ", color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   350→                val displayText = if (!q.title.isNullOrBlank()) q.title else q.stem.take(80)
   351→                Text(displayText, color = Color(0xFF1E293B), fontSize = 11.sp, fontWeight = FontWeight.SemiBold, maxLines = 1, overflow = TextOverflow.Ellipsis, modifier = Modifier.weight(1f, fill = false))
   352→                if (q.isFree) {
   353→                    Spacer(Modifier.width(4.dp))
   354→                    Surface(color = Color(0xFF22C55E), shape = RoundedCornerShape(3.dp)) {
   355→                        Text("FREE", color = Color.White, fontSize = 8.sp, fontWeight = FontWeight.Bold, modifier = Modifier.padding(horizontal = 4.dp, vertical = 1.dp))
   356→                    }
   357→                }
   358→                Spacer(Modifier.weight(1f))
   359→                IconButton(onClick = { showGrid = true }, modifier = Modifier.size(24.dp)) {
   360→                    Icon(Icons.Default.GridView, null, tint = Color(0xFF64748B), modifier = Modifier.size(14.dp))
   361→                }
   362→            }
   363→        }
   364→        Box(modifier = Modifier.fillMaxWidth().height(2.dp).background(Color.Black))
   365→
   366→        // ── 3. MAIN CONTENT ── 60% question (left, scrollable) | 40% answers (right, scrollable)
   367→        // The DreamKorea logo is rendered as a faded watermark in the background
   368→        // (centered, low alpha) so it blends in nicely without taking screen space.
   369→        Box(modifier = Modifier.weight(1f).fillMaxWidth()) {
   370→            // Watermark logo at the center of the background
   371→            Image(
   372→                painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.dreamkorea_logo),
   373→                contentDescription = null,
   374→                modifier = Modifier.align(Alignment.Center).size(280.dp).alpha(0.08f),
   375→                contentScale = ContentScale.Fit
   376→            )
   377→            Row(modifier = Modifier.fillMaxSize()) {
   378→            // LEFT: Question content (60%) — scrollable so long titles/stems/images don't get cut
   379→            Column(
   380→                modifier = Modifier.weight(0.6f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   381→                verticalArrangement = Arrangement.Center,
   382→                horizontalAlignment = Alignment.CenterHorizontally
   383→            ) {
   384→                // Question title (if set by admin) — prominent header
   385→                if (!q.title.isNullOrBlank()) {
   386→                    Surface(
   387→                        color = Color(0xFF003478),
   388→                        shape = RoundedCornerShape(10.dp),
   389→                        modifier = Modifier.fillMaxWidth(0.92f)
   390→                    ) {
   391→                        Text(
   392→                            q.title,
   393→                            color = Color.White,
   394→                            fontSize = 14.sp,
   395→                            fontWeight = FontWeight.Bold,
   396→                            textAlign = TextAlign.Center,
   397→                            modifier = Modifier.padding(vertical = 8.dp, horizontal = 12.dp),
   398→                            maxLines = 2,
   399→                        )
   400→                    }
   401→                    Spacer(Modifier.height(6.dp))
   402→                }
   403→                // Question card — shows stem (the Question field) OR mediaText
   404→                // Title is already shown above in the blue header.
   405→                val questionText = q.stem.ifBlank { q.mediaText ?: "" }
   406→                if (questionText.isNotBlank()) {
   407→                    Surface(
   408→                        color = Color.White,
   409→                        shape = RoundedCornerShape(14.dp),
   410→                        shadowElevation = 2.dp,
   411→                        border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   412→                        modifier = Modifier.fillMaxWidth(0.92f)
   413→                    ) {
   414→                        Text(
   415→                            questionText,
   416→                            color = Color(0xFF1E293B),
   417→                            fontSize = 14.sp,
   418→                            fontWeight = FontWeight.Medium,
   419→                            modifier = Modifier.padding(14.dp)
   420→                        )
   421→                    }
   422→                    Spacer(Modifier.height(6.dp))
   423→                }
   424→                // Media images — ContentScale.Fit (contain, not cover)
   425→                if (q.descType == "image" && !q.descImageUrl.isNullOrBlank()) {
   426→                    val url = q.descImageUrl!!.toAbsoluteUrl()
   427→                    coil.compose.AsyncImage(
   428→                        model = url, contentDescription = null,
   429→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 120.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(url) },
   430→                        contentScale = ContentScale.Fit
   431→                    )
   432→                }
   433→                val mediaImgUrl = (q.mediaImageUrl ?: q.imageUrl)?.toAbsoluteUrl()
   434→                if (!mediaImgUrl.isNullOrBlank()) {
   435→                    coil.compose.AsyncImage(
   436→                        model = mediaImgUrl, contentDescription = null,
   437→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 140.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(mediaImgUrl) },
   438→                        contentScale = ContentScale.Fit
   439→                    )
   440→                }
   441→                val mediaAudUrl = (q.mediaAudioUrl ?: q.audioUrl)?.toAbsoluteUrl()
   442→                if (!mediaAudUrl.isNullOrBlank()) {
   443→                    // key(q.id) ensures the AudioPlayerCard is FULLY RECREATED when
   444→                    // the question changes — state (playCount, disabled) resets completely.
   445→                    // This fixes the bug where audio was locked from the previous question.
   446→                    key(q.id) {
   447→                        AudioPlayerCard(theme = theme, url = mediaAudUrl, loopCount = q.audioLoop, loopDelaySec = q.audioLoopDelay, sound = sound, questionId = q.id, playCounts = audioPlayCounts)
   448→                    }
   449→                }
   450→            }
   451→
   452→            // Vertical divider
   453→            Box(modifier = Modifier.width(2.dp).fillMaxHeight().background(Color.Black))
   454→
   455→            // RIGHT: Answer options (40%) — scrollable so long options don't get cut
   456→            Column(
   457→                modifier = Modifier.weight(0.4f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   458→                verticalArrangement = Arrangement.Center
   459→            ) {
   460→                when (q.answerType) {
   461→                    "text", "choose" -> {
   462→                        (0 until minOf(4, options.size)).forEach { i ->
   463→                            val isSelected = answers[q.id] == options.getOrNull(i)
   464→                            val optText = options.getOrNull(i) ?: ""
   465→                            val blankWord = q.optionBlanks.getOrNull(i)?.takeIf { it.isNotBlank() }
   466→                            Row(
   467→                                modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp).clickable { sound.click(); answers[q.id] = optText },
   468→                                verticalAlignment = Alignment.CenterVertically
   469→                            ) {
   470→                                Surface(
   471→                                    color = if (isSelected) theme.primary else Color.White,
   472→                                    border = androidx.compose.foundation.BorderStroke(2.dp, Color.Black),
   473→                                    shape = androidx.compose.foundation.shape.CircleShape,
   474→                                    modifier = Modifier.size(40.dp)
   475→                                ) {

   846→/// Reference-style 4-column grid of PERFECT SQUARE cells with 2px black borders.
   847→/// Solved = blue fill + white text. Current = blue border + glow.
   848→/// Uses continuous numbering: Reading 1-20, Listening 21-40 (from globalIdx).
   849→/// showAllBlocks: true = pad grid to expected count (20) with blank cells so
   850→/// the user sees which questions are added vs blank.
   851→/// false = only show cells for questions that actually exist.
   852→/// Cells ALWAYS maintain 1:1 aspect ratio (perfect square) regardless of count.
   853→/// Grid is scrollable so all blocks are reachable.
   854→@Composable
   855→private fun QuestionGridRef(
   856→    test: TestDetail,
   857→    items: List<TestItemDetail>,
   858→    answers: SnapshotStateMap<String, Any>,
   859→    currentIdx: Int,
   860→    sound: SoundManager,
   861→    haptic: androidx.compose.ui.hapticfeedback.HapticFeedback,
   862→    filterMode: Boolean?,
   863→    accentBlue: Color,
   864→    showAllBlocks: Boolean,
   865→    onPick: (Int) -> Unit,
   866→) {
   867→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   868→    val cols = 4  // 4 columns per the HTML reference
   869→    val gridScrollState = rememberScrollState()
   870→
   871→    // Determine total cells to render
   872→    val expectedTotal = if (showAllBlocks) {
   873→        val isAudio = items.isNotEmpty() && items[0].question.blockType == "audio"
   874→        val blockCount = if (isAudio) test.audioBlockCount else test.textBlockCount
   875→        maxOf(blockCount, items.size)
   876→    } else {
   877→        items.size
   878→    }
   879→    val rowsCount = (expectedTotal + cols - 1) / cols
   880→
   881→    // Scrollable grid — cells are PERFECT SQUARES (aspectRatio 1f)
   882→    Column(
   883→        modifier = Modifier
   884→            .fillMaxWidth()
   885→            .verticalScroll(gridScrollState),
   886→        verticalArrangement = Arrangement.spacedBy(8.dp)
   887→    ) {
   888→        for (rowIdx in 0 until rowsCount) {
   889→            Row(
   890→                modifier = Modifier.fillMaxWidth(),
   891→                horizontalArrangement = Arrangement.spacedBy(8.dp)
   892→            ) {
   893→                for (colIdx in 0 until cols) {
   894→                    val localIdx = rowIdx * cols + colIdx
   895→                    if (localIdx < items.size) {
   896→                        val globalIdx = globalIndices[localIdx]
   897→                        // DISPLAY NUMBER: Reading shows 1-20, Listening shows 21-40
   898→                        // Uses blockNumber (1-20 within each block) + 20 offset for audio
   899→                        val q = items[localIdx].question
   900→                        val displayNum = if (q.blockType == "audio") {
   901→                            (q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)) + 20
   902→                        } else {
   903→                            q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)
   904→                        }
   905→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   906→                        val isCurrent = globalIdx == currentIdx
   907→                        val isFilteredOut = when (filterMode) {
   908→                            true -> !isAnswered
   909→                            false -> isAnswered
   910→                            null -> false
   911→                        }
   912→                        Box(
   913→                            modifier = Modifier
   914→                                .weight(1f)
   915→                                .aspectRatio(1f)  // PERFECT SQUARE — always 1:1
   916→                                .clip(RoundedCornerShape(6.dp))
   917→                                .border(
   918→                                    width = if (isCurrent) 2.5.dp else 1.5.dp,
   919→                                    color = when {
   920→                                        isCurrent -> accentBlue
   921→                                        isAnswered -> accentBlue
   922→                                        else -> Color(0xFF111111)
   923→                                    },
   924→                                    shape = RoundedCornerShape(6.dp)
   925→                                )
   926→                                .background(if (isAnswered) accentBlue else Color.White)
   927→                                .alpha(if (isFilteredOut) 0.15f else 1f)
   928→                                .clickable(enabled = !isFilteredOut) {
   929→                                    sound.click()
   930→                                    haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
   931→                                    onPick(globalIdx)
   932→                                },
   933→                            contentAlignment = Alignment.Center
   934→                        ) {
   935→                            Text(
   936→                                "$displayNum",
   937→                                color = if (isAnswered) Color.White else Color(0xFF111111),
   938→                                fontSize = 15.sp,
   939→                                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Medium,
   940→                            )
   941→                        }
   942→                    } else if (showAllBlocks) {
   943→                        // Empty placeholder cell — perfect square
   944→                        // Reading placeholders: 1-20, Listening placeholders: 21-40
   945→                        val isAudioGrid = items.isNotEmpty() && items[0].question.blockType == "audio"
   946→                        val placeholderNum = if (isAudioGrid) localIdx + 21 else localIdx + 1
   947→                        Box(
   948→                            modifier = Modifier
   949→                                .weight(1f)
   950→                                .aspectRatio(1f)  // PERFECT SQUARE
   951→                                .clip(RoundedCornerShape(6.dp))
   952→                                .border(1.5.dp, Color(0xFFEEEEEE), RoundedCornerShape(6.dp))
   953→                                .background(Color(0xFFFAFAFA)),
   954→                            contentAlignment = Alignment.Center
   955→                        ) {
   956→                            Text(
   957→                                "$placeholderNum",
   958→                                color = Color(0xFFCCCCCC),
   959→                                fontSize = 13.sp,

  1004→// ─── Audio player with loop support ───────────────────────────────────────────

  1006→//   0 or 1 = plays once
  1007→//   2 = plays twice
  1008→//   N = plays N times
  1009→//  -1 = infinite loop
  1010→@Composable
  1011→fun AudioPlayerCard(
  1012→    theme: AppTheme,
  1013→    url: String,
  1014→    loopCount: Int,
  1015→    loopDelaySec: Int,
  1016→    sound: SoundManager,
  1017→    questionId: String? = null,
  1018→    playCounts: SnapshotStateMap<String, Int>? = null,
  1019→) {
  1020→    val context = LocalContext.current
  1021→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
  1022→    var isPlaying by remember { mutableStateOf(false) }
  1023→
  1024→    // PERSISTENT play count — stored in parent map (survives navigation).
  1025→    // Prevents cheat where student navigates away and back to reset plays.
  1026→    val persistentCount = playCounts?.get(questionId) ?: 0
  1027→    val maxPlays = loopCount.coerceAtLeast(1)
  1028→    val disabled = persistentCount >= maxPlays
  1029→    val scope = rememberCoroutineScope()
  1030→
  1031→    fun incrementPlayCount() {
  1032→        if (questionId != null && playCounts != null) {
  1033→            playCounts[questionId] = (playCounts[questionId] ?: 0) + 1
  1034→        }
  1035→    }
  1036→
  1037→    DisposableEffect(url) {
  1038→        onDispose {
  1039→            mediaPlayer?.release()
  1040→            mediaPlayer = null
  1041→        }
  1042→    }
  1043→
  1044→    Surface(color = theme.cardBg, shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth(), shadowElevation = 1.dp) {
  1045→        Row(
  1046→            modifier = Modifier.padding(14.dp),
  1047→            verticalAlignment = Alignment.CenterVertically
  1048→        ) {
  1049→            // Audio icon
  1050→            Surface(
  1051→                color = if (disabled) Color(0xFFE2E8F0) else theme.primary.copy(alpha = 0.15f),
  1052→                shape = RoundedCornerShape(10.dp),
  1053→                modifier = Modifier.size(44.dp)
  1054→            ) {
  1055→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
  1056→                    Icon(
  1057→                        Icons.Default.Headphones,
  1058→                        null,
  1059→                        tint = if (disabled) Color(0xFF94A3B8) else theme.primary,
  1060→                        modifier = Modifier.size(22.dp)
  1061→                    )
  1062→                }
  1063→            }
  1064→            Spacer(Modifier.width(12.dp))
  1065→            // Play count info
  1066→            Column(modifier = Modifier.weight(1f)) {
  1067→                Text(
  1068→                    if (disabled) "Audio played ($maxPlays/$maxPlays)" else "Audio — Play $persistentCount / $maxPlays",
  1069→                    color = if (disabled) Color(0xFF94A3B8) else theme.darkText,
  1070→                    fontSize = 13.sp,
  1071→                    fontWeight = FontWeight.SemiBold
  1072→                )
  1073→                if (!disabled) {
  1074→                    Text(
  1075→                        if (maxPlays == 1) "Tap to listen (1 time only)" else "Tap to listen ($maxPlays times)",
  1076→                        color = theme.subText, fontSize = 11.sp
  1077→                    )
  1078→                } else {
  1079→                    Text("No more plays left", color = Color(0xFF94A3B8), fontSize = 11.sp)
  1080→                }
  1081→            }
  1082→            // Play/Pause button — DISABLED after all plays used
  1083→            IconButton(
  1084→                onClick = {
  1085→                    if (disabled) return@IconButton
  1086→                    sound.click()
  1087→                    if (isPlaying) {
  1088→                        // Don't pause — let it finish. Only allow play, not pause.
  1089→                        return@IconButton
  1090→                    } else {
  1091→                        try {
  1092→                            mediaPlayer?.release()
  1093→                            val mp = android.media.MediaPlayer().apply {
  1094→                                setDataSource(url)
  1095→                                setOnPreparedListener {
  1096→                                    start()
  1097→                                    isPlaying = true
  1098→                                    incrementPlayCount()
  1099→                                }
  1100→                                setOnCompletionListener {
  1101→                                    val currentCount = playCounts?.get(questionId) ?: 0
  1102→                                    if (currentCount < maxPlays) {
  1103→                                        scope.launch {
  1104→                                            if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
  1105→                                            val latestCount = playCounts?.get(questionId) ?: 0
  1106→                                            if (latestCount < maxPlays) {
  1107→                                                incrementPlayCount()
  1108→                                                start()
  1109→                                            } else {
  1110→                                                isPlaying = false
  1111→                                            }
  1112→                                        }
  1113→                                    } else {
  1114→                                        isPlaying = false
  1115→                                    }
  1116→                                }
  1117→                                setOnErrorListener { _, _, _ ->
  1118→                                    isPlaying = false
  1119→                                    true
  1120→                                }
  1121→                                prepareAsync()
  1122→                            }
  1123→                            mediaPlayer = mp
  1124→                        } catch (_: Exception) {
  1125→                            isPlaying = false
  1126→                        }
  1127→                    }
  1128→                },
  1129→                enabled = !disabled && !isPlaying
  1130→            ) {
  1131→                Icon(
  1132→                    when {
  1133→                        disabled -> Icons.Default.Lock
  1134→                        isPlaying -> Icons.Default.VolumeUp
  1135→                        else -> Icons.Default.PlayArrow
  1136→                    },
  1137→                    null,
  1138→                    tint = if (disabled) Color(0xFFCBD5E1) else theme.primary,
  1139→                    modifier = Modifier.size(32.dp)
  1140→                )
  1141→            }
  1142→        }
  1143→    }
  1144→}

  1910→/**
  1911→ * Builds an AnnotatedString where the [blankWord] is underlined within [text].
  1912→ * If blankWord is null/empty, returns the plain text.
  1913→ * The underlined word is case-insensitive matched.
  1914→ */
  1915→fun buildUnderlinedText(text: String, blankWord: String?): androidx.compose.ui.text.AnnotatedString {
  1916→    if (blankWord.isNullOrBlank()) return androidx.compose.ui.text.AnnotatedString(text)
  1917→    val idx = text.indexOf(blankWord, ignoreCase = true)
  1918→    if (idx < 0) return androidx.compose.ui.text.AnnotatedString(text)
  1919→    return androidx.compose.ui.text.buildAnnotatedString {
  1920→        append(text.substring(0, idx))
  1921→        withStyle(androidx.compose.ui.text.SpanStyle(textDecoration = androidx.compose.ui.text.style.TextDecoration.Underline)) {
  1922→            append(text.substring(idx, idx + blankWord.length))
  1923→        }
  1924→        append(text.substring(idx + blankWord.length))
  1925→    }
  1926→}

  1929→// CLIENT-SIDE SUBMIT FALLBACK — when the server's submit endpoint doesn't
  1930→// know how to handle combined exam IDs (qbank-combined / bundle-{id}), we
  1931→// grade the exam locally using the question data we already have loaded.
  1932→// ═══════════════════════════════════════════════════════════════════════════

  1934→/**
  1935→ * Grades a combined exam client-side. Used as a fallback when
  1936→ * /api/student/tests/[testId]/submit returns 404 for combined exam IDs.
  1937→ *
  1938→ * Grades SINGLE_CHOICE, TRUE_FALSE, ONE_WORD, FILL_BLANK, MULTIPLE_CHOICE.
  1939→ * Subjective types (SHORT_ANSWER, LONG_ANSWER, MATCHING) are marked as
  1940→ * incorrect (the student would need a teacher to review them).
  1941→ */
  1942→private fun gradeCombinedExamClientSide(
  1943→    test: TestDetail,
  1944→    answers: Map<String, Any>,
  1945→): SubmitResponse {
  1946→    var score = 0
  1947→    val review = mutableListOf<ReviewItem>()
  1948→
  1949→    // Normalize URL — strip origin so /api/files/... matches https://.../api/files/...
  1950→    fun normalizeUrl(val_: Any?): Any? {
  1951→        if (val_ is String) {
  1952→            return try {
  1953→                if (val_.startsWith("http://") || val_.startsWith("https://")) {
  1954→                    java.net.URI(val_).path
  1955→                } else val_
  1956→            } catch (_: Exception) { val_ }
  1957→        }
  1958→        return val_
  1959→    }
  1960→
  1961→    for (item in test.items) {
  1962→        val q = item.question
  1963→        val ans = normalizeUrl(answers[q.id])
  1964→        var isCorrect = false
