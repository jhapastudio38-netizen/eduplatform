    53→/**
    54→ * Exam taking screen — full flow:
    55→ * 1. Loading skeleton (with timeout + retry)
    56→ * 2. Question-by-question with audio (loops N times), image, options
    57→ * 3. Click sound on every interaction
    58→ * 4. Wrong → shows correct answer in red
    59→ * 5. Correct → green confirmation
    60→ * 6. Auto-advances to next question
    61→ * 7. Final score + review screen
    62→ * 8. Stats auto-update via /api/student/tests/[id]/submit
    63→ */
    64→@Composable
    65→fun ExamScreen(theme: AppTheme, testId: String, onExit: () -> Unit) {
    66→    val sound = rememberSoundManager()
    67→    val scope = rememberCoroutineScope()
    68→    val context = LocalContext.current
    69→    var test by remember { mutableStateOf<TestDetail?>(null) }
    70→    var loading by remember { mutableStateOf(true) }
    71→    var error by remember { mutableStateOf("") }
    72→    var currentIdx by remember { mutableStateOf(0) }
    73→    val answers = remember { mutableStateMapOf<String, Any>() }
    74→    // Persistent audio play counts per question ID — survives navigation.
    75→    // Prevents cheat where student navigates away and back to reset plays.
    76→    val audioPlayCounts = remember { mutableStateMapOf<String, Int>() }
    77→    // When audio is playing, disable navigation buttons
    78→    var audioPlaying by remember { mutableStateOf(false) }
    79→    var currentlyPlayingId by remember { mutableStateOf<String?>(null) }
    80→    var submitResult by remember { mutableStateOf<SubmitResponse?>(null) }
    81→    var submitting by remember { mutableStateOf(false) }
    82→    // Per-question feedback (after answering, before moving on)
    83→    var questionFeedback by remember { mutableStateOf<QuestionFeedback?>(null) }
    84→    // Timer
    85→    var timeLeft by remember { mutableStateOf(0) }
    86→
    87→    // ── ORIENTATION ── FORCE LANDSCAPE. No onDispose PORTRAIT —
    88→    // MainScreen handles portrait when screen changes to non-exam.
    89→    DisposableEffect(Unit) {
    90→        val activity = context as? Activity
    91→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    92→        onDispose { }
    93→    }
    94→    LaunchedEffect(Unit) {
    95→        kotlinx.coroutines.delay(100)
    96→        val activity = context as? Activity
    97→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    98→    }
    99→    // Retry trigger — increment to force reload
   100→    var retryCount by remember { mutableStateOf(0) }
   101→
   102→    // Load test detail — uses LaunchedEffect's own scope, finally block guarantees loading cleanup
   103→    // ─── Back button handler — show exit warning during exam ──────────────
   104→    var showExitDialog by remember { mutableStateOf(false) }
   105→    androidx.activity.compose.BackHandler(enabled = !loading && error.isEmpty()) {
   106→        showExitDialog = true
   107→    }
   108→    if (showExitDialog) {
   109→        androidx.compose.material3.AlertDialog(
   110→            onDismissRequest = { showExitDialog = false },
   111→            title = { Text("Exit Exam?") },
   112→            text = { Text("Your exam is in progress. Are you sure you want to exit? Your answers will be lost.") },
   113→            confirmButton = { Button(onClick = { showExitDialog = false; onExit() }) { Text("Exit") } },
   114→            dismissButton = { OutlinedButton(onClick = { showExitDialog = false }) { Text("Continue Exam") } }
   115→        )
   116→    }
   117→
   118→    LaunchedEffect(testId, retryCount) {
   119→        loading = true
   120→        error = ""
   121→        try {
   122→            // 30-second timeout — combined exams may need multiple requests
   123→            val result = withTimeoutOrNull(30_000L) {
   124→                when {
   125→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   126→                    testId == "qbank-combined" -> {
   127→                        try {
   128→                            AppState.api.getQBankCombined().test
   129→                        } catch (e: retrofit2.HttpException) {
   130→                            if (e.code() == 404) buildQBankCombinedClientSide()
   131→                            else throw e
   132→                        }
   133→                    }
   134→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   135→                    testId.startsWith("bundle-") -> {
   136→                        val bundleId = testId.removePrefix("bundle-")
   137→                        try {
   138→                            AppState.api.getBundleCombined(bundleId).test
   139→                        } catch (e: retrofit2.HttpException) {
   140→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   141→                            else throw e
   142→                        }
   143→                    }
   144→                    // Normal test — fetch by ID
   145→                    else -> AppState.api.getTestDetail(testId).test
   146→                }
   147→            }
   148→            if (result != null) {
   149→                test = result
   150→                timeLeft = (if (result.durationMin > 0) result.durationMin else 50) * 60
   151→            } else {
   152→                error = "The request timed out. Check your internet connection and try again."
   153→            }
   154→        } catch (e: retrofit2.HttpException) {
   155→            // Try to read the actual error message from the response body
   156→            val rawBody = try { e.response()?.errorBody()?.string() } catch (_: Exception) { null }
   157→            error = when (e.code()) {
   158→                401 -> "Your session has expired. Please log out and sign in again."
   159→                403 -> "This exam is not available right now."
   160→                404 -> "This test could not be found. It may have been removed."
   161→                500 -> "Server error. Please try again in a moment.${if (rawBody != null) " ($rawBody)" else ""}"
   162→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   163→            }
   164→        } catch (e: java.net.UnknownHostException) {
   165→            error = "No internet connection. Please check your network and try again."
   166→        } catch (e: java.net.SocketTimeoutException) {
   167→            error = "The request timed out. Please try again."
   168→        } catch (e: java.io.IOException) {
   169→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   170→        } catch (e: Exception) {
   171→            error = "Unexpected error: ${e.message ?: "Please try again."}"
   172→        } finally {
   173→            // GUARANTEED: loading is always reset, even if the coroutine is cancelled
   174→            loading = false
   175→        }
   176→    }
   177→
   178→    // Countdown timer
   179→    LaunchedEffect(test, submitResult) {
   180→        if (test != null && submitResult == null) {
   181→            while (timeLeft > 0) {
   182→                delay(1000)
   183→                timeLeft--
   184→            }
   185→            // Auto-submit when timer reaches zero
   186→            val currentTest = test
   187→            if (timeLeft == 0 && currentTest != null && submitResult == null && !submitting) {
   188→                submitting = true
   189→                sound.swoosh()
   190→                try {
   191→                    // Use fallback for combined exams (qbank-combined / bundle-{id})
   192→                    submitResult = if (currentTest.id == "qbank-combined" || currentTest.id.startsWith("bundle-")) {
   193→                        submitCombinedExamWithFallback(currentTest, answers.toMap())
   194→                    } else {
   195→                        submitExamWithFallback(currentTest, answers.toMap())
   196→                    }
   197→                    sound.success()
   198→                } catch (e: java.net.UnknownHostException) {
   199→                    error = "No internet connection. Could not submit your answers."
   200→                } catch (e: java.io.IOException) {
   201→                    error = "Network error. Could not submit your answers."
   202→                } catch (e: Exception) {
   203→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   204→                }
   205→                submitting = false
   206→            }
   207→        }
   208→    }
   209→
   210→    if (loading) {
   211→        // Loading skeleton with a subtle "Loading..." label so users know it's working
   212→        Column(Modifier.fillMaxSize()) {
   213→            LinearProgressIndicator(
   214→                modifier = Modifier.fillMaxWidth(),
   215→                color = theme.primary,
   216→                trackColor = theme.primary.copy(alpha = 0.1f),
   217→            )
   218→            SkeletonListScreen(theme, itemCount = 4)
   219→        }
   220→        return
   221→    }
   222→
   223→    if (error.isNotEmpty()) {
   224→        Column(
   225→            Modifier.fillMaxSize().padding(24.dp),
   226→            horizontalAlignment = Alignment.CenterHorizontally,
   227→            verticalArrangement = Arrangement.Center
   228→        ) {
   229→            Icon(
   230→                Icons.Default.CloudOff,
   231→                null,
   232→                tint = theme.errorRed.copy(alpha = 0.7f),
   233→                modifier = Modifier.size(56.dp)
   234→            )
   235→            Spacer(Modifier.height(16.dp))
   236→            Text(
   237→                "Couldn't load the test",
   238→                color = theme.darkText,
   239→                fontSize = 18.sp,
   240→                fontWeight = FontWeight.Bold
   241→            )
   242→            Spacer(Modifier.height(8.dp))
   243→            Text(
   244→                error,
   245→                color = theme.subText,
   246→                fontSize = 13.sp,
   247→                textAlign = TextAlign.Center,
   248→                modifier = Modifier.padding(horizontal = 16.dp)
   249→            )
   250→            Spacer(Modifier.height(8.dp))
   251→            Text(
   252→                "Test ID: $testId",
   253→                color = theme.subText.copy(alpha = 0.5f),
   254→                fontSize = 10.sp,
   255→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   256→            )
   257→            Spacer(Modifier.height(24.dp))
   258→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   259→                OutlinedButton(
   260→                    onClick = onExit,
   261→                    shape = RoundedCornerShape(10.dp)
   262→                ) { Text("Go back") }
   263→                Button(
   264→                    onClick = {
   265→                        sound.click()
   266→                        retryCount++ // triggers LaunchedEffect reload
   267→                    },
   268→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   269→                    shape = RoundedCornerShape(10.dp)
   270→                ) {
   271→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   272→                    Spacer(Modifier.width(6.dp))
   273→                    Text("Retry")
   274→                }
   275→            }
   276→        }
   277→        return
   278→    }
   279→
   280→    val t = test ?: return
   281→    // Guard: if the test has no questions, show a friendly message instead of crashing
   282→    if (t.items.isEmpty()) {
   283→        Column(
   284→            Modifier.fillMaxSize().padding(20.dp),
   285→            horizontalAlignment = Alignment.CenterHorizontally,
   286→            verticalArrangement = Arrangement.Center
   287→        ) {
   288→            Icon(Icons.Default.Quiz, null, tint = theme.subText, modifier = Modifier.size(48.dp))
   289→            Spacer(Modifier.height(12.dp))
   290→            Text("This test has no questions yet.", color = theme.darkText, fontSize = 14.sp, textAlign = TextAlign.Center)
   291→            Spacer(Modifier.height(16.dp))
   292→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }
   293→        }
   294→        return
   295→    }
   296→
   297→    val currentQuestion = t.items.getOrNull(currentIdx)
   298→
   299→    // ─── Result screen ──────────────────────────────────────────────────────
   300→    if (submitResult != null) {
   301→        ExamResultScreen(theme, submitResult!!, onExit, sound, examTitle = t.title, examDescription = t.description)
   302→        return
   303→    }
   304→
   305→    // ═══ EXAM UI — spec-compliant landscape layout ═══
   306→    // Sort items: Reading (text) first, then Listening (audio)
   307→    val sortedItems = t.items.sortedWith(compareBy(
   308→        { if (it.question.blockType == "audio") 1 else 0 },
   309→        { it.question.blockNumber }
   310→    ))
   311→    val item = sortedItems.getOrNull(currentIdx) ?: return
   312→    val q = item.question
   313→    val options = q.options ?: emptyList()
   314→    val textItems = sortedItems.filter { it.question.blockType != "audio" }
   315→    val readingCount = if (textItems.isNotEmpty()) textItems.size else sortedItems.size
   316→    val listeningCount = if (textItems.isNotEmpty()) sortedItems.size - textItems.size else 0
   317→    val answeredCount = answers.size
   318→    val remainingCount = sortedItems.size - answeredCount
   319→    // Start with the grid view so the student can pick which question to answer
   320→    // first. They tap a number → answer → return to grid → pick next → submit.
   321→    var showGrid by remember { mutableStateOf(true) }
   322→
   323→    Column(modifier = Modifier.fillMaxSize().background(Color(0xFFF8FAFC))) {
   324→        // ── 1. TOP STATUS HEADER ── compact white bar with thin border
   325→        Surface(
   326→            color = Color.White,
   327→            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   328→        ) {
   329→            Row(modifier = Modifier.fillMaxWidth().height(36.dp), verticalAlignment = Alignment.CenterVertically) {
   330→                // Section type
   331→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   332→                    val sectionLabel = if (q.blockType == "audio") "Listening ($listeningCount)" else "Reading ($readingCount)"
   333→                    Text(sectionLabel, color = Color(0xFF003478), fontSize = 10.sp, fontWeight = FontWeight.Bold)
   334→                }
   335→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   336→                // Total
   337→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   338→                    Text("Total: ${t.items.size}", color = Color(0xFF64748B), fontSize = 10.sp)
   339→                }
   340→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   341→                // Remaining
   342→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   343→                    Text("Left: $remainingCount", color = Color(0xFF64748B), fontSize = 10.sp)
   344→                }
   345→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   346→                // Timer
   347→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   348→                    val mm = timeLeft / 60; val ss = timeLeft % 60
   349→                    Text(String.format("%02d:%02d", mm, ss), color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   350→                }
   351→            }
   352→        }
   353→
   354→        // ── 2. INSTRUCTION ROW ── compact question number + title + FREE badge
   355→        Surface(modifier = Modifier.fillMaxWidth().padding(horizontal = 6.dp, vertical = 2.dp)) {
   356→            Row(verticalAlignment = Alignment.Top) {
   357→                Text("${currentIdx + 1}. ", color = Color(0xFF003478), fontSize = 16.sp, fontWeight = FontWeight.Bold)
   358→                val displayText = q.stem.ifBlank { q.mediaText ?: "" }
   359→                Text(displayText, color = Color(0xFF1E293B), fontSize = 15.sp, fontWeight = FontWeight.SemiBold, modifier = Modifier.weight(1f, fill = true))
   360→                if (q.isFree) {
   361→                    Spacer(Modifier.width(4.dp))
   362→                    Surface(color = Color(0xFF22C55E), shape = RoundedCornerShape(3.dp)) {
   363→                        Text("FREE", color = Color.White, fontSize = 9.sp, fontWeight = FontWeight.Bold, modifier = Modifier.padding(horizontal = 5.dp, vertical = 2.dp))
   364→                    }
   365→                }
   366→                IconButton(onClick = { showGrid = true }, modifier = Modifier.size(28.dp)) {
   367→                    Icon(Icons.Default.GridView, null, tint = Color(0xFF64748B), modifier = Modifier.size(18.dp))
   368→                }
   369→            }
   370→        }
   371→        Box(modifier = Modifier.fillMaxWidth().height(2.dp).background(Color.Black))
   372→
   373→        // ── 3. MAIN CONTENT ── 60% question (left, scrollable) | 40% answers (right, scrollable)
   374→        // DreamKorea logo watermark in background (faded, centered)
   375→        Box(modifier = Modifier.weight(1f).fillMaxWidth()) {
   376→            // Watermark logo at the center of the background
   377→            Image(
   378→                painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.dreamkorea_logo),
   379→                contentDescription = null,
   380→                modifier = Modifier.align(Alignment.Center).size(280.dp).alpha(0.08f),
   381→                contentScale = ContentScale.Fit
   382→            )
   383→            Row(modifier = Modifier.fillMaxSize()) {
   384→            // LEFT: Question content (60%) — uses BoxWithConstraints so it always
   385→            // fills the available height, centering its contents vertically.
   386→            BoxWithConstraints(
   387→                modifier = Modifier.weight(0.6f).fillMaxHeight().padding(8.dp)
   388→            ) {
   389→                val boxMaxHeight = this.maxHeight
   390→                Column(
   391→                    modifier = Modifier.fillMaxWidth().verticalScroll(rememberScrollState()).heightIn(min = boxMaxHeight),
   392→                    verticalArrangement = Arrangement.Center,
   393→                    horizontalAlignment = Alignment.CenterHorizontally
   394→                ) {
   395→                    // Description TEXT — shown in a centered card (17sp)
   396→                    if (q.descType == "text" && !q.descText.isNullOrBlank()) {
   397→                        Surface(
   398→                            color = Color.White,
   399→                            shape = RoundedCornerShape(14.dp),
   400→                            shadowElevation = 2.dp,
   401→                            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   402→                            modifier = Modifier.fillMaxWidth(0.92f).padding(vertical = 4.dp)
   403→                        ) {
   404→                            Text(
   405→                                q.descText!!,
   406→                                color = Color(0xFF1E293B),
   407→                                fontSize = 17.sp,
   408→                                fontWeight = FontWeight.Medium,
   409→                                modifier = Modifier.padding(14.dp)
   410→                            )
   411→                        }
   412→                    }
   413→                    // Description IMAGE — shown when descType == "image"
   414→                    if (q.descType == "image" && !q.descImageUrl.isNullOrBlank()) {
   415→                        val url = q.descImageUrl!!.toAbsoluteUrl()
   416→                        coil.compose.AsyncImage(
   417→                            model = url, contentDescription = null,
   418→                            modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 200.dp).padding(vertical = 4.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(url) },
   419→                            contentScale = ContentScale.Fit
   420→                        )
   421→                    }
   422→                    // Media TEXT — shown in a centered card (17sp)
   423→                    if (q.mediaType == "text" && !q.mediaText.isNullOrBlank()) {
   424→                        Surface(
   425→                            color = Color.White,
   426→                            shape = RoundedCornerShape(14.dp),
   427→                            shadowElevation = 2.dp,
   428→                            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   429→                            modifier = Modifier.fillMaxWidth(0.92f).padding(vertical = 4.dp)
   430→                        ) {
   431→                            Text(
   432→                                q.mediaText!!,
   433→                                color = Color(0xFF1E293B),
   434→                                fontSize = 17.sp,
   435→                                fontWeight = FontWeight.Medium,
   436→                                modifier = Modifier.padding(14.dp)
   437→                            )
   438→                        }
   439→                    }
   440→                    // Media IMAGE — main media image
   441→                    val mediaImgUrl = (q.mediaImageUrl ?: q.imageUrl)?.toAbsoluteUrl()
   442→                    if (!mediaImgUrl.isNullOrBlank()) {
   443→                        coil.compose.AsyncImage(
   444→                            model = mediaImgUrl, contentDescription = null,
   445→                            modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 220.dp).padding(vertical = 4.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(mediaImgUrl) },
   446→                            contentScale = ContentScale.Fit
   447→                        )
   448→                    }
   449→                    // Media AUDIO — single play button (36dp), blocked when another audio is playing
   450→                    val mediaAudUrl = (q.mediaAudioUrl ?: q.audioUrl)?.toAbsoluteUrl()
   451→                    if (!mediaAudUrl.isNullOrBlank()) {
   452→                        // key(q.id) ensures the AudioPlayerCard is FULLY RECREATED when
   453→                        // the question changes — state (playCount, disabled) resets completely.
   454→                        key(q.id) {
   455→                            AudioPlayerCard(
   456→                                theme = theme, url = mediaAudUrl,
   457→                                loopCount = q.audioLoop,
   458→                                loopDelaySec = if (q.audioLoopDelay > 0) q.audioLoopDelay else 2,
   459→                                sound = sound, questionId = q.id, playCounts = audioPlayCounts,
   460→                                blocked = currentlyPlayingId != null && currentlyPlayingId != q.id,
   461→                                onPlayingChange = { playing ->
   462→                                    audioPlaying = playing
   463→                                    currentlyPlayingId = if (playing) q.id else null
   464→                                }
   465→                            )
   466→                        }
   467→                    }
   468→                }
   469→            }
   470→
   471→            // Vertical divider
   472→            Box(modifier = Modifier.width(2.dp).fillMaxHeight().background(Color.Black))
   473→
   474→            // RIGHT: Answer options (40%) — scrollable so long options don't get cut
   475→            Column(
   476→                modifier = Modifier.weight(0.4f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   477→                verticalArrangement = Arrangement.Center
   478→            ) {
   479→                when (q.answerType) {
   480→                    "text", "choose" -> {

   877→/// Reference-style 4-column grid of PERFECT SQUARE cells with 2px black borders.
   878→/// Solved = blue fill + white text. Current = blue border + glow.
   879→/// Uses continuous numbering: Reading 1-20, Listening 21-40 (from globalIdx).
   880→/// showAllBlocks: true = pad grid to expected count (20) with blank cells so
   881→/// the user sees which questions are added vs blank.
   882→/// false = only show cells for questions that actually exist.
   883→/// Cells ALWAYS maintain 1:1 aspect ratio (perfect square) regardless of count.
   884→/// Grid is scrollable so all blocks are reachable.
   885→@Composable
   886→private fun QuestionGridRef(
   887→    test: TestDetail,
   888→    items: List<TestItemDetail>,
   889→    answers: SnapshotStateMap<String, Any>,
   890→    currentIdx: Int,
   891→    sound: SoundManager,
   892→    haptic: androidx.compose.ui.hapticfeedback.HapticFeedback,
   893→    filterMode: Boolean?,
   894→    accentBlue: Color,
   895→    showAllBlocks: Boolean,
   896→    onPick: (Int) -> Unit,
   897→) {
   898→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   899→    val cols = 4  // 4 columns per the HTML reference
   900→    val gridScrollState = rememberScrollState()
   901→
   902→    // Determine total cells to render
   903→    val expectedTotal = if (showAllBlocks) {
   904→        val isAudio = items.isNotEmpty() && items[0].question.blockType == "audio"
   905→        val blockCount = if (isAudio) test.audioBlockCount else test.textBlockCount
   906→        maxOf(blockCount, items.size)
   907→    } else {
   908→        items.size
   909→    }
   910→    val rowsCount = (expectedTotal + cols - 1) / cols
   911→
   912→    // Scrollable grid — cells are PERFECT SQUARES (aspectRatio 1f)
   913→    Column(
   914→        modifier = Modifier
   915→            .fillMaxWidth()
   916→            .verticalScroll(gridScrollState),
   917→        verticalArrangement = Arrangement.spacedBy(8.dp)
   918→    ) {
   919→        for (rowIdx in 0 until rowsCount) {
   920→            Row(
   921→                modifier = Modifier.fillMaxWidth(),
   922→                horizontalArrangement = Arrangement.spacedBy(8.dp)
   923→            ) {
   924→                for (colIdx in 0 until cols) {
   925→                    val localIdx = rowIdx * cols + colIdx
   926→                    if (localIdx < items.size) {
   927→                        val globalIdx = globalIndices[localIdx]
   928→                        // DISPLAY NUMBER: Reading shows 1-20, Listening shows 21-40
   929→                        // Uses blockNumber (1-20 within each block) + 20 offset for audio
   930→                        val q = items[localIdx].question
   931→                        val displayNum = if (q.blockType == "audio") {
   932→                            (q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)) + 20
   933→                        } else {
   934→                            q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)
   935→                        }
   936→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   937→                        val isCurrent = globalIdx == currentIdx
   938→                        val isFilteredOut = when (filterMode) {
   939→                            true -> !isAnswered
   940→                            false -> isAnswered
   941→                            null -> false
   942→                        }
   943→                        Box(
   944→                            modifier = Modifier
   945→                                .weight(1f)
   946→                                .aspectRatio(1f)  // PERFECT SQUARE — always 1:1
   947→                                .clip(RoundedCornerShape(6.dp))
   948→                                .border(
   949→                                    width = if (isCurrent) 2.5.dp else 1.5.dp,
   950→                                    color = when {
   951→                                        isCurrent -> accentBlue
   952→                                        isAnswered -> accentBlue
   953→                                        else -> Color(0xFF111111)
   954→                                    },
   955→                                    shape = RoundedCornerShape(6.dp)
   956→                                )
   957→                                .background(if (isAnswered) accentBlue else Color.White)
   958→                                .alpha(if (isFilteredOut) 0.15f else 1f)
   959→                                .clickable(enabled = !isFilteredOut) {
   960→                                    sound.click()
   961→                                    haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
   962→                                    onPick(globalIdx)
   963→                                },
   964→                            contentAlignment = Alignment.Center
   965→                        ) {
   966→                            Text(
   967→                                "$displayNum",

  1035→// ─── Audio player with loop support ───────────────────────────────────────────
  1036→// Compact play button (36dp) — no card, no label, just an icon.
  1037→// loopCount = total number of times to play the audio:
  1038→//   0 or 1 = plays once
  1039→//   2 = plays twice
  1040→//   N = plays N times
  1041→//  -1 = infinite loop
  1042→// `blocked` = another audio is currently playing — show as disabled.
  1043→// `unlimited` = review mode — no play limit (e.g. teacher reviewing audio).
  1044→@Composable
  1045→fun AudioPlayerCard(
  1046→    theme: AppTheme,
  1047→    url: String,
  1048→    loopCount: Int,
  1049→    loopDelaySec: Int,
  1050→    sound: SoundManager,
  1051→    questionId: String? = null,
  1052→    playCounts: SnapshotStateMap<String, Int>? = null,
  1053→    onPlayingChange: ((Boolean) -> Unit)? = null,
  1054→    blocked: Boolean = false,
  1055→    unlimited: Boolean = false,
  1056→) {
  1057→    val context = LocalContext.current
  1058→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
  1059→    var isPlaying by remember { mutableStateOf(false) }
  1060→    var localPlayCount by remember { mutableStateOf(0) }
  1061→    val effectiveCount = if (playCounts != null && questionId != null) playCounts[questionId] ?: 0 else localPlayCount
  1062→    LaunchedEffect(isPlaying) { onPlayingChange?.invoke(isPlaying) }
  1063→    val maxPlays = if (unlimited) Int.MAX_VALUE else (if (loopCount <= 0) 2 else loopCount)
  1064→    val disabled = if (unlimited) false else effectiveCount >= maxPlays
  1065→    val scope = rememberCoroutineScope()
  1066→    fun incrementPlayCount() { if (questionId != null && playCounts != null) playCounts[questionId] = (playCounts[questionId] ?: 0) + 1 else localPlayCount++ }
  1067→    fun currentCount(): Int = if (playCounts != null && questionId != null) playCounts[questionId] ?: 0 else localPlayCount
  1068→    DisposableEffect(url) { onDispose { mediaPlayer?.release(); mediaPlayer = null } }
  1069→    val isBlocked = disabled || isPlaying || blocked
  1070→    Surface(
  1071→        color = if (disabled || blocked) Color(0xFFF1F5F9) else theme.primary.copy(alpha = 0.08f),
  1072→        shape = RoundedCornerShape(8.dp),
  1073→        modifier = Modifier.size(36.dp)
  1074→    ) {
  1075→        Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize().clickable(enabled = !isBlocked) {
  1076→            if (disabled || blocked) return@clickable
  1077→            if (isPlaying) return@clickable
  1078→            sound.click()
  1079→            isPlaying = true
  1080→            onPlayingChange?.invoke(true)
  1081→            try {
  1082→                mediaPlayer?.release()
  1083→                val mp = android.media.MediaPlayer().apply {
  1084→                    setDataSource(url)
  1085→                    setOnPreparedListener { start(); incrementPlayCount() }
  1086→                    setOnCompletionListener {
  1087→                        if (unlimited) { isPlaying = false }
  1088→                        else {
  1089→                            val cc = currentCount()
  1090→                            if (cc < maxPlays) {
  1091→                                scope.launch {
  1092→                                    if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
  1093→                                    val latestCount = currentCount()
  1094→                                    if (latestCount < maxPlays) { incrementPlayCount(); start() }
  1095→                                    else { isPlaying = false }
  1096→                                }
  1097→                            } else { isPlaying = false }
  1098→                        }
  1099→                    }
  1100→                    setOnErrorListener { _, _, _ -> isPlaying = false; true }
  1101→                    prepareAsync()
  1102→                }
  1103→                mediaPlayer = mp
  1104→            } catch (_: Exception) { isPlaying = false }
  1105→        }) {
  1106→            Icon(Icons.Default.PlayArrow, null,
  1107→                tint = if (disabled || blocked) Color(0xFFCBD5E1) else if (isPlaying) theme.primary.copy(alpha = 0.4f) else theme.primary,
  1108→                modifier = Modifier.size(20.dp)
  1109→            )
  1110→        }
  1111→    }
  1112→}

  1832→    if (answer == null) return null
  1833→    return when (answer) {
  1834→        is String -> answer
  1835→        is List<*> -> answer.joinToString(", ")
  1836→        else -> answer.toString()
  1837→    }

  1840→/// Renders an answer — shows media (image/audio) for URL answers, text otherwise.
  1841→@Composable
  1842→private fun AnswerDisplay(answer: Any?, theme: AppTheme, sound: SoundManager? = null) {
  1843→    if (answer == null) {
  1844→        Text("—", color = theme.subText, fontSize = 12.sp)
  1845→        return
  1846→    }
  1847→    when (answer) {
  1848→        is String -> {
  1849→            val url = answer.trim()
  1850→            if (url.startsWith("http") && (url.contains(".jpg") || url.contains(".jpeg") || url.contains(".png") || url.contains(".gif") || url.contains(".webp"))) {
  1851→                coil.compose.AsyncImage(
  1852→                    model = url, contentDescription = "Answer image",
  1853→                    modifier = Modifier.fillMaxWidth().heightIn(max = 80.dp).clip(RoundedCornerShape(6.dp)),
  1854→                    contentScale = ContentScale.Fit
  1855→                )
  1856→            } else if (url.startsWith("http") && (url.contains(".mp3") || url.contains(".wav") || url.contains(".ogg") || url.contains(".m4a"))) {
  1857→                AudioPlayerCard(theme = theme, url = url, loopCount = 1, loopDelaySec = 0, sound = sound ?: rememberSoundManager())
  1858→            } else {
  1859→                Text(url, color = theme.darkText, fontSize = 12.sp, fontWeight = FontWeight.Medium)
  1860→            }
  1861→        }
  1862→        is List<*> -> {
  1863→            Column { answer.forEach { Text(it?.toString() ?: "", color = theme.darkText, fontSize = 12.sp) } }
  1864→        }
  1865→        else -> Text(answer.toString(), color = theme.darkText, fontSize = 12.sp)
  1866→    }
  1867→}
