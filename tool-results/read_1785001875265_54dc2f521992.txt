   140→                  </div>

   184→function CreateExamDialog({ open, onOpenChange, onCreated }: {
   185→  open: boolean;
   186→  onOpenChange: (v: boolean) => void;
   187→  onCreated: (t: Test) => void;
   188→}) {
   189→  const [busy, setBusy] = useState(false);
   190→  const [form, setForm] = useState({
   191→    title: "",
   192→    description: "",
   193→    durationMin: 60,
   194→    examType: "UBT",
   195→    category: "",
   196→    price: "",
   197→    featuredImage: "",
   198→    audioPlayMode: "single" as "single" | "double",
   199→    audioGapSec: 2,
   200→    textBlockCount: 20,
   201→    audioBlockCount: 20,
   202→  });
   203→
   204→  async function uploadFile(file: File, folder: string): Promise<string> {
   205→    const fd = new FormData();
   206→    fd.append("file", file);
   207→    fd.append("folder", folder);
   208→    const res = await fetch("/api/admin/file-upload", { method: "POST", body: fd });
   209→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || "Upload failed"); }
   210→    const d = await res.json();
   211→    return d.url;
   212→  }
   213→
   214→  async function create() {
   215→    if (!form.title.trim()) { toast.error("Exam name required"); return; }
   216→    setBusy(true);
   217→    try {
   218→      const res = await fetch("/api/admin/tests", {
   219→        method: "POST",
   220→        headers: { "Content-Type": "application/json" },
   221→        body: JSON.stringify({
   222→          ...form,
   223→          price: form.price ? parseFloat(form.price) : undefined,
   224→          isExam: true,
   225→          isPublished: true,
   226→        }),
   227→      });
   228→      if (!res.ok) { const d = await res.json(); toast.error(d.error || "Failed"); return; }
   229→      const d = await res.json();
   230→      toast.success("Exam created — now add questions");
   231→      onCreated(d.test);
   232→    } finally { setBusy(false); }
   233→  }
   234→
   235→  return (
   236→    <Dialog open={open} onOpenChange={onOpenChange}>
   237→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   238→        <DialogHeader><DialogTitle>Create New Exam</DialogTitle></DialogHeader>
   239→        <div className="space-y-3">
   240→          {/* Exam Name */}
   241→          <div>
   242→            <Label>Exam Name *</Label>
   243→            <Input value={form.title} onChange={(e) => setForm(f => ({ ...f, title: e.target.value }))} placeholder="e.g. UBT Mock Test 1" />
   244→          </div>
   245→
   246→          {/* Exam Details */}
   247→          <div>
   248→            <Label>Exam Details</Label>
   249→            <Textarea rows={2} value={form.description} onChange={(e) => setForm(f => ({ ...f, description: e.target.value }))} placeholder="Brief description of the exam" />
   250→          </div>
   251→
   252→          {/* Exam Time + Price */}
   253→          <div className="grid grid-cols-2 gap-3">
   254→            <div>
   255→              <Label>Exam Time (minutes) *</Label>
   256→              <Input type="number" value={form.durationMin} onChange={(e) => setForm(f => ({ ...f, durationMin: parseInt(e.target.value) || 60 }))} min={1} />
   257→            </div>
   258→            <div>
   259→              <Label>Exam Price (optional)</Label>

   262→          </div>
   263→
   264→          {/* Exam Type + Category */}

   266→            <div>
   267→              <Label>Exam Type *</Label>
   268→              <Select value={form.examType} onValueChange={(v) => setForm(f => ({ ...f, examType: v }))}>
   269→                <SelectTrigger><SelectValue /></SelectTrigger>
   270→                <SelectContent>
   271→                  <SelectItem value="UBT">UBT</SelectItem>
   272→                  <SelectItem value="CBT">CBT</SelectItem>
   273→                  <SelectItem value="CHAPTER">Chapter</SelectItem>
   274→                </SelectContent>

   276→            </div>

   278→              <Label>Category (shows in app)</Label>

   281→          </div>
   282→
   283→          {/* Featured Image */}

   285→            <Label>Featured Image</Label>

   344→// EXAM EDITOR — Step 2: block-based question editor
   345→// ═══════════════════════════════════════════════════════════════════════════
   346→
   347→function ExamEditor({ test, onClose }: { test: Test; onClose: () => void }) {
   348→  const [activeBlock, setActiveBlock] = useState<"text" | "audio">("text");
   349→  const [activeNumber, setActiveNumber] = useState(1);
   350→  const [questions, setQuestions] = useState<Record<string, QuestionData>>({});
   351→  const [loading, setLoading] = useState(true);
   352→  const [clipboard, setClipboard] = useState<string>("");
   353→  const [showPasteDialog, setShowPasteDialog] = useState(false);
   354→  const [pasteCode, setPasteCode] = useState("");
   355→  const [pushing, setPushing] = useState(false);
   356→  const [isPublished, setIsPublished] = useState(test.isPublished);
   357→
   358→  const textCount = test.textBlockCount || 20;
   359→  const audioCount = test.audioBlockCount || 20;
   360→
   361→  function key(blockType: string, blockNumber: number) {
   362→    return `${blockType}-${blockNumber}`;
   363→  }
   364→
   365→  useEffect(() => {
   366→    // Load existing questions
   367→    fetch(`/api/admin/tests/${test.id}/questions`)
   368→      .then((r) => r.json())
   369→      .then((d) => {
   370→        const map: Record<string, QuestionData> = {};
   371→        for (const q of d.questions || []) {
   372→          map[key(q.blockType, q.blockNumber)] = q;
   373→        }
   374→        setQuestions(map);
   375→      })
   376→      .finally(() => setLoading(false));
   377→  }, [test.id]);
   378→
   379→  const currentKey = key(activeBlock, activeNumber);
   380→  const currentQuestion = questions[currentKey] || emptyQuestion(activeBlock, activeNumber);
   381→
   382→  function updateQuestion(q: QuestionData) {
   383→    setQuestions((prev) => ({ ...prev, [currentKey]: q }));
   384→  }
   385→
   386→  async function saveQuestion() {
   387→    const q = currentQuestion;
   388→    if (!q.stem.trim()) { toast.error("Question text required"); return; }
   389→    try {
   390→      const res = await fetch(`/api/admin/tests/${test.id}/questions`, {
   391→        method: "POST",
   392→        headers: { "Content-Type": "application/json" },
   393→        body: JSON.stringify(q),
   394→      });
   395→      if (!res.ok) { const d = await res.json(); toast.error(d.error || "Save failed"); return; }
   396→      toast.success(`Question ${q.blockNumber} saved`);
   397→    } catch {
   398→      toast.error("Save failed");
   399→    }
   400→  }
   401→
   402→  function copyQuestion() {
   403→    const q = currentQuestion;
   404→    if (!q.stem.trim()) { toast.error("Nothing to copy — question is empty"); return; }
   405→    // Generate a copy code — admin can define their own prefix
   406→    const code = `DK-${activeBlock.toUpperCase()}-${q.blockNumber}-${Date.now().toString(36).toUpperCase()}`;
   407→    const data = JSON.stringify({ code, question: q });
   408→    // Store in localStorage so admin can paste later
   409→    const allCopies = JSON.parse(localStorage.getItem("dk_copies") || "{}");
   410→    allCopies[code] = data;
   411→    localStorage.setItem("dk_copies", JSON.stringify(allCopies));
   412→    setClipboard(code);
   413→    // Copy code to clipboard
   414→    navigator.clipboard.writeText(code);
   415→    toast.success(`Copied! Code: ${code}`);
   416→  }
   417→
   418→  function pasteQuestion() {
   419→    setShowPasteDialog(true);
   420→  }
   421→
   422→  async function pushToApp() {
   423→    const filledCount = Object.values(questions).filter(q => q.stem.trim()).length;
   424→    if (filledCount === 0) {
   425→      toast.error("Cannot push: add at least one question first");
   426→      return;
   427→    }
   428→    if (!confirm(`Push this exam to all student apps?\n\n${filledCount} question(s) will be live.\n\nStudents will see it in their Exams page immediately.`)) {
   429→      return;
   430→    }
   431→    setPushing(true);
   432→    try {
   433→      const res = await fetch(`/api/admin/tests/${test.id}/publish`, { method: "POST" });
   434→      const d = await res.json();
   435→      if (!res.ok) { toast.error(d.error || "Push failed"); return; }
   436→      setIsPublished(true);
   437→      toast.success(d.message || "Pushed to app — students can now see this exam");
   438→    } catch {
   439→      toast.error("Push failed — check your connection");
   440→    } finally {
   441→      setPushing(false);
   442→    }
   443→  }
   444→
   445→  function doPaste() {
   446→    const allCopies = JSON.parse(localStorage.getItem("dk_copies") || "{}");
   447→    const data = allCopies[pasteCode.trim()];
   448→    if (!data) { toast.error("Invalid paste code"); return; }
   449→    const parsed = JSON.parse(data);
   450→    const q = parsed.question as QuestionData;
   451→    // Paste into current slot — keep current block number/type
   452→    const pasted: QuestionData = {
   453→      ...q,
   454→      blockType: activeBlock,
   455→      blockNumber: activeNumber,
   456→    };
   457→    updateQuestion(pasted);
   458→    toast.success("Question pasted — click Save to persist");
   459→    setShowPasteDialog(false);
   460→    setPasteCode("");
   461→  }
   462→
   463→  const blockNumbers = activeBlock === "text"
   464→    ? Array.from({ length: textCount }, (_, i) => i + 1)
   465→    : Array.from({ length: audioCount }, (_, i) => i + 1);
   466→
   467→  return (
   468→    <Dialog open={true} onOpenChange={(v) => { if (!v) onClose(); }}>
   469→      <DialogContent className="max-w-5xl max-h-[95vh] overflow-hidden flex flex-col">
   470→        <DialogHeader>
   471→          <DialogTitle className="flex items-center gap-2">
   472→            <span>{test.title}</span>
   473→            <Badge variant="outline">{test.examType}</Badge>
   474→            {test.category && <Badge variant="secondary">{test.category}</Badge>}
   475→          </DialogTitle>
   476→        </DialogHeader>
   477→
   478→        {/* Block tabs */}
   479→        <div className="flex gap-2 border-b pb-2">
   480→          <Button
   481→            variant={activeBlock === "text" ? "default" : "outline"}
   482→            size="sm"
   483→            onClick={() => { setActiveBlock("text"); setActiveNumber(1); }}
   484→          >
   485→            <FileText className="w-4 h-4 mr-1" /> Text Block (1-{textCount})
   486→          </Button>
   487→          <Button
   488→            variant={activeBlock === "audio" ? "default" : "outline"}
   489→            size="sm"
   490→            onClick={() => { setActiveBlock("audio"); setActiveNumber(1); }}
   491→          >
   492→            <Headphones className="w-4 h-4 mr-1" /> Audio Block ({textCount + 1}-{textCount + audioCount})
   493→          </Button>
   494→        </div>
   495→
   496→        {/* Block number selector — grid of numbered buttons */}
   497→        <div className="grid grid-cols-10 gap-1 max-h-24 overflow-y-auto p-1 bg-slate-50 rounded">
   498→          {blockNumbers.map((num) => {
   499→            const k = key(activeBlock, num);
   500→            const isFilled = questions[k] && questions[k].stem.trim();
   501→            const isActive = num === activeNumber;
   502→            return (
   503→              <button
   504→                key={num}
   505→                onClick={() => setActiveNumber(num)}
   506→                className={`h-8 rounded text-xs font-medium transition-colors ${
   507→                  isActive ? "bg-primary text-primary-foreground" :
   508→                  isFilled ? "bg-green-100 text-green-700 border border-green-300" :
   509→                  "bg-white border hover:bg-slate-100"
   510→                }`}
   511→              >
   512→                {num}
   513→              </button>
   514→            );
   515→          })}
   516→        </div>
   517→
   518→        {/* Question editor */}
   519→        <div className="flex-1 overflow-y-auto pr-1">
   520→          {loading ? (
   521→            <p className="text-center py-8 text-muted-foreground">Loading questions…</p>
   522→          ) : (
   523→            <QuestionEditor
   524→              question={currentQuestion}
   525→              onChange={updateQuestion}
   526→              blockLabel={`Question ${activeBlock === "audio" ? textCount + activeNumber : activeNumber}`}
   527→              isAudioBlock={activeBlock === "audio"}
   528→            />
   529→          )}
   530→        </div>
   531→
   532→        {/* Action buttons — Done, Copy, Paste, Push to App */}
   533→        <div className="flex items-center justify-between border-t pt-3">
   534→          <div className="flex gap-2 flex-wrap">
   535→            <Button onClick={saveQuestion} variant="default">
   536→              <Save className="w-4 h-4 mr-1" /> Done
   537→            </Button>
   538→            <Button onClick={copyQuestion} variant="outline">
   539→              <Copy className="w-4 h-4 mr-1" /> Copy
   540→            </Button>
   541→            <Button onClick={pasteQuestion} variant="outline">
   542→              <ClipboardPaste className="w-4 h-4 mr-1" /> Paste
   543→            </Button>
   544→          </div>
   545→          <div className="flex items-center gap-2">
   546→            {isPublished && (
   547→              <Badge className="bg-green-500">
   548→                <CheckCircle2 className="w-3 h-3 mr-1" /> Live in App
   549→              </Badge>
   550→            )}
   551→            <Button
   552→              onClick={pushToApp}
   553→              disabled={pushing || isPublished}
   554→              variant={isPublished ? "outline" : "default"}
   555→              className={isPublished ? "" : "bg-green-600 hover:bg-green-700"}
   556→            >
   557→              {pushing ? (
   558→                <><span className="animate-spin mr-1">⏳</span> Pushing…</>
   559→              ) : isPublished ? (
   560→                <><CheckCircle2 className="w-4 h-4 mr-1" /> Pushed</>
   561→              ) : (
   562→                <><span className="mr-1">🚀</span> Push to App</>
   563→              )}
   564→            </Button>
   565→            <Button variant="ghost" onClick={onClose}>Close</Button>
   566→          </div>
   567→        </div>
   568→      </DialogContent>
   569→
   570→      {/* Paste dialog */}
   571→      {showPasteDialog && (
   572→        <Dialog open={true} onOpenChange={setShowPasteDialog}>
   573→          <DialogContent className="max-w-md">
   574→            <DialogHeader><DialogTitle>Paste Question</DialogTitle></DialogHeader>
   575→            <div className="space-y-3">
   576→              <Label>Enter paste code</Label>
   577→              <Input value={pasteCode} onChange={(e) => setPasteCode(e.target.value)} placeholder="DK-TEXT-1-XXXX" />
   578→              <p className="text-xs text-muted-foreground">Paste the code you got from "Copy Question" to duplicate that question here.</p>
   579→            </div>
   580→            <div className="flex justify-end gap-2">
   581→              <Button variant="outline" onClick={() => setShowPasteDialog(false)}>Cancel</Button>
   582→              <Button onClick={doPaste}>Paste</Button>
   583→            </div>
   584→          </DialogContent>
   585→        </Dialog>
   586→      )}
   587→    </Dialog>
   588→  );
   589→}

   601→  async function uploadFile(file: File, folder: string): Promise<string> {
   602→    const fd = new FormData();

   607→    const d = await res.json();
   608→    return d.url;
   609→  }

   611→  function UploadButton({ onUpload, accept }: { onUpload: (url: string) => void; accept: string }) {

   614→        <input type="file" accept={accept} className="hidden" onChange={async (e) => {
   615→          const f = e.target.files?.[0]; if (!f) return;

   617→          catch (err: any) { toast.error(err.message); }
   618→        }} />
   619→        <span className="inline-flex items-center h-8 px-2 rounded-md bg-primary text-primary-foreground text-xs cursor-pointer hover:bg-primary/90">📁 Upload</span>
