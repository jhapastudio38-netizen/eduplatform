    53→/**
    54→ * Exam taking screen — full flow:
    55→ * 1. Loading skeleton (with timeout + retry)
    56→ * 2. Question-by-question with audio (loops N times), image, options
    57→ * 3. Click sound on every interaction
    58→ * 4. Wrong → shows correct answer in red
    59→ * 5. Correct → green confirmation
    60→ * 6. Auto-advances to next question
    61→ * 7. Final score + review screen
    62→ * 8. Stats auto-update via /api/student/tests/[id]/submit
    63→ */
    64→@Composable
    65→fun ExamScreen(theme: AppTheme, testId: String, onExit: () -> Unit) {
    66→    val sound = rememberSoundManager()
    67→    val scope = rememberCoroutineScope()
    68→    val context = LocalContext.current
    69→    var test by remember { mutableStateOf<TestDetail?>(null) }
    70→    var loading by remember { mutableStateOf(true) }
    71→    var error by remember { mutableStateOf("") }
    72→    var currentIdx by remember { mutableStateOf(0) }
    73→    val answers = remember { mutableStateMapOf<String, Any>() }
    74→    // Persistent audio play counts per question ID — survives navigation.
    75→    // Prevents cheat where student navigates away and back to reset plays.
    76→    val audioPlayCounts = remember { mutableStateMapOf<String, Int>() }
    77→    // When audio is playing, disable navigation buttons
    78→    var audioPlaying by remember { mutableStateOf(false) }
    79→    var submitResult by remember { mutableStateOf<SubmitResponse?>(null) }
    80→    var submitting by remember { mutableStateOf(false) }
    81→    // Per-question feedback (after answering, before moving on)
    82→    var questionFeedback by remember { mutableStateOf<QuestionFeedback?>(null) }
    83→    // Timer
    84→    var timeLeft by remember { mutableStateOf(0) }
    85→
    86→    // ── ORIENTATION ── FORCE LANDSCAPE. No onDispose PORTRAIT —
    87→    // MainScreen handles portrait when screen changes to non-exam.
    88→    DisposableEffect(Unit) {
    89→        val activity = context as? Activity
    90→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    91→        onDispose { }
    92→    }
    93→    LaunchedEffect(Unit) {
    94→        kotlinx.coroutines.delay(100)
    95→        val activity = context as? Activity
    96→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    97→    }
    98→    // Retry trigger — increment to force reload
    99→    var retryCount by remember { mutableStateOf(0) }
   100→
   101→    // Load test detail — uses LaunchedEffect's own scope, finally block guarantees loading cleanup
   102→    // ─── Back button handler — show exit warning during exam ──────────────
   103→    var showExitDialog by remember { mutableStateOf(false) }
   104→    androidx.activity.compose.BackHandler(enabled = !loading && error.isEmpty()) {
   105→        showExitDialog = true
   106→    }
   107→    if (showExitDialog) {
   108→        androidx.compose.material3.AlertDialog(
   109→            onDismissRequest = { showExitDialog = false },
   110→            title = { Text("Exit Exam?") },
   111→            text = { Text("Your exam is in progress. Are you sure you want to exit? Your answers will be lost.") },
   112→            confirmButton = { Button(onClick = { showExitDialog = false; onExit() }) { Text("Exit") } },
   113→            dismissButton = { OutlinedButton(onClick = { showExitDialog = false }) { Text("Continue Exam") } }
   114→        )
   115→    }
   116→
   117→    LaunchedEffect(testId, retryCount) {
   118→        loading = true
   119→        error = ""
   120→        try {
   121→            // 30-second timeout — combined exams may need multiple requests
   122→            val result = withTimeoutOrNull(30_000L) {
   123→                when {
   124→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   125→                    testId == "qbank-combined" -> {
   126→                        try {
   127→                            AppState.api.getQBankCombined().test
   128→                        } catch (e: retrofit2.HttpException) {
   129→                            if (e.code() == 404) buildQBankCombinedClientSide()
   130→                            else throw e
   131→                        }
   132→                    }
   133→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   134→                    testId.startsWith("bundle-") -> {
   135→                        val bundleId = testId.removePrefix("bundle-")
   136→                        try {
   137→                            AppState.api.getBundleCombined(bundleId).test
   138→                        } catch (e: retrofit2.HttpException) {
   139→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   140→                            else throw e
   141→                        }
   142→                    }
   143→                    // Normal test — fetch by ID
   144→                    else -> AppState.api.getTestDetail(testId).test
   145→                }
   146→            }
   147→            if (result != null) {
   148→                test = result
   149→                timeLeft = (result.durationMin.coerceAtLeast(1)) * 60
   150→            } else {
   151→                error = "The request timed out. Check your internet connection and try again."
   152→            }
   153→        } catch (e: retrofit2.HttpException) {
   154→            // Try to read the actual error message from the response body
   155→            val rawBody = try { e.response()?.errorBody()?.string() } catch (_: Exception) { null }
   156→            error = when (e.code()) {
   157→                401 -> "Your session has expired. Please log out and sign in again."
   158→                403 -> "This exam is not available right now."
   159→                404 -> "This test could not be found. It may have been removed."
   160→                500 -> "Server error. Please try again in a moment.${if (rawBody != null) " ($rawBody)" else ""}"
   161→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   162→            }
   163→        } catch (e: java.net.UnknownHostException) {
   164→            error = "No internet connection. Please check your network and try again."
   165→        } catch (e: java.net.SocketTimeoutException) {
   166→            error = "The request timed out. Please try again."
   167→        } catch (e: java.io.IOException) {
   168→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   169→        } catch (e: Exception) {
   170→            error = "Unexpected error: ${e.message ?: "Please try again."}"
   171→        } finally {
   172→            // GUARANTEED: loading is always reset, even if the coroutine is cancelled
   173→            loading = false
   174→        }
   175→    }
   176→
   177→    // Countdown timer
   178→    LaunchedEffect(test, submitResult) {
   179→        if (test != null && submitResult == null) {
   180→            while (timeLeft > 0) {
   181→                delay(1000)
   182→                timeLeft--
   183→            }
   184→            // Auto-submit when timer reaches zero
   185→            val currentTest = test
   186→            if (timeLeft == 0 && currentTest != null && submitResult == null && !submitting) {
   187→                submitting = true
   188→                sound.swoosh()
   189→                try {
   190→                    // Use fallback for combined exams (qbank-combined / bundle-{id})
   191→                    submitResult = if (currentTest.id == "qbank-combined" || currentTest.id.startsWith("bundle-")) {
   192→                        submitCombinedExamWithFallback(currentTest, answers.toMap())
   193→                    } else {
   194→                        submitExamWithFallback(currentTest, answers.toMap())
   195→                    }
   196→                    sound.success()
   197→                } catch (e: java.net.UnknownHostException) {
   198→                    error = "No internet connection. Could not submit your answers."
   199→                } catch (e: java.io.IOException) {
   200→                    error = "Network error. Could not submit your answers."
   201→                } catch (e: Exception) {
   202→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   203→                }
   204→                submitting = false
   205→            }
   206→        }
   207→    }
   208→
   209→    if (loading) {
   210→        // Loading skeleton with a subtle "Loading..." label so users know it's working
   211→        Column(Modifier.fillMaxSize()) {
   212→            LinearProgressIndicator(
   213→                modifier = Modifier.fillMaxWidth(),
   214→                color = theme.primary,
   215→                trackColor = theme.primary.copy(alpha = 0.1f),
   216→            )
   217→            SkeletonListScreen(theme, itemCount = 4)
   218→        }
   219→        return
   220→    }
   221→
   222→    if (error.isNotEmpty()) {
   223→        Column(
   224→            Modifier.fillMaxSize().padding(24.dp),
   225→            horizontalAlignment = Alignment.CenterHorizontally,
   226→            verticalArrangement = Arrangement.Center
   227→        ) {
   228→            Icon(
   229→                Icons.Default.CloudOff,
   230→                null,
   231→                tint = theme.errorRed.copy(alpha = 0.7f),
   232→                modifier = Modifier.size(56.dp)
   233→            )
   234→            Spacer(Modifier.height(16.dp))
   235→            Text(
   236→                "Couldn't load the test",
   237→                color = theme.darkText,
   238→                fontSize = 18.sp,
   239→                fontWeight = FontWeight.Bold
   240→            )
   241→            Spacer(Modifier.height(8.dp))
   242→            Text(
   243→                error,
   244→                color = theme.subText,
   245→                fontSize = 13.sp,
   246→                textAlign = TextAlign.Center,
   247→                modifier = Modifier.padding(horizontal = 16.dp)
   248→            )
   249→            Spacer(Modifier.height(8.dp))
   250→            Text(
   251→                "Test ID: $testId",
   252→                color = theme.subText.copy(alpha = 0.5f),
   253→                fontSize = 10.sp,
   254→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   255→            )
   256→            Spacer(Modifier.height(24.dp))
   257→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   258→                OutlinedButton(
   259→                    onClick = onExit,
   260→                    shape = RoundedCornerShape(10.dp)
   261→                ) { Text("Go back") }
   262→                Button(
   263→                    onClick = {
   264→                        sound.click()
   265→                        retryCount++ // triggers LaunchedEffect reload
   266→                    },
   267→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   268→                    shape = RoundedCornerShape(10.dp)
   269→                ) {
   270→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   271→                    Spacer(Modifier.width(6.dp))
   272→                    Text("Retry")
   273→                }
   274→            }
   275→        }
   276→        return
   277→    }
   278→
   279→    val t = test ?: return
   280→    // Guard: if the test has no questions, show a friendly message instead of crashing
   281→    if (t.items.isEmpty()) {
   282→        Column(
   283→            Modifier.fillMaxSize().padding(20.dp),
   284→            horizontalAlignment = Alignment.CenterHorizontally,
   285→            verticalArrangement = Arrangement.Center
   286→        ) {
   287→            Icon(Icons.Default.Quiz, null, tint = theme.subText, modifier = Modifier.size(48.dp))
   288→            Spacer(Modifier.height(12.dp))
   289→            Text("This test has no questions yet.", color = theme.darkText, fontSize = 14.sp, textAlign = TextAlign.Center)
   290→            Spacer(Modifier.height(16.dp))
   291→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }
   292→        }
   293→        return
   294→    }
   295→
   296→    val currentQuestion = t.items.getOrNull(currentIdx)
   297→
   298→    // ─── Result screen ──────────────────────────────────────────────────────
   299→    if (submitResult != null) {
   300→        ExamResultScreen(theme, submitResult!!, onExit, sound, examTitle = t.title, examDescription = t.description)
   301→        return
   302→    }
   303→
   304→    // ═══ EXAM UI — spec-compliant landscape layout ═══
   305→    // Sort items: Reading (text) first, then Listening (audio)
   306→    val sortedItems = t.items.sortedWith(compareBy(
   307→        { if (it.question.blockType == "audio") 1 else 0 },
   308→        { it.question.blockNumber }
   309→    ))
   310→    val item = sortedItems.getOrNull(currentIdx) ?: return
   311→    val q = item.question
   312→    val options = q.options ?: emptyList()
   313→    val textItems = sortedItems.filter { it.question.blockType != "audio" }
   314→    val readingCount = if (textItems.isNotEmpty()) textItems.size else sortedItems.size
   315→    val listeningCount = if (textItems.isNotEmpty()) sortedItems.size - textItems.size else 0
   316→    val answeredCount = answers.size
   317→    val remainingCount = sortedItems.size - answeredCount
   318→    // Start with the grid view so the student can pick which question to answer
   319→    // first. They tap a number → answer → return to grid → pick next → submit.
   320→    var showGrid by remember { mutableStateOf(true) }
   321→
   322→    Column(modifier = Modifier.fillMaxSize().background(Color(0xFFF8FAFC))) {
   323→        // ── 1. TOP STATUS HEADER ── compact white bar with thin border
   324→        Surface(
   325→            color = Color.White,
   326→            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   327→        ) {
   328→            Row(modifier = Modifier.fillMaxWidth().height(36.dp), verticalAlignment = Alignment.CenterVertically) {
   329→                // Section type
   330→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   331→                    val sectionLabel = if (q.blockType == "audio") "Listening ($listeningCount)" else "Reading ($readingCount)"
   332→                    Text(sectionLabel, color = Color(0xFF003478), fontSize = 10.sp, fontWeight = FontWeight.Bold)
   333→                }
   334→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   335→                // Total
   336→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   337→                    Text("Total: ${t.items.size}", color = Color(0xFF64748B), fontSize = 10.sp)
   338→                }
   339→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   340→                // Remaining
   341→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   342→                    Text("Left: $remainingCount", color = Color(0xFF64748B), fontSize = 10.sp)
   343→                }
   344→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   345→                // Timer
   346→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   347→                    val mm = timeLeft / 60; val ss = timeLeft % 60
   348→                    Text(String.format("%02d:%02d", mm, ss), color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   349→                }
   350→            }
   351→        }
   352→
   353→        // ── 2. INSTRUCTION ROW ── compact question number + title + FREE badge
   354→        Surface(modifier = Modifier.fillMaxWidth().padding(horizontal = 6.dp, vertical = 2.dp)) {
   355→            Row(verticalAlignment = Alignment.CenterVertically) {
   356→                Text("${currentIdx + 1}. ", color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   357→                val displayText = q.stem.take(80)
   358→                Text(displayText, color = Color(0xFF1E293B), fontSize = 11.sp, fontWeight = FontWeight.SemiBold, maxLines = 1, overflow = TextOverflow.Ellipsis, modifier = Modifier.weight(1f, fill = false))
   359→                if (q.isFree) {
   360→                    Spacer(Modifier.width(4.dp))
   361→                    Surface(color = Color(0xFF22C55E), shape = RoundedCornerShape(3.dp)) {
   362→                        Text("FREE", color = Color.White, fontSize = 8.sp, fontWeight = FontWeight.Bold, modifier = Modifier.padding(horizontal = 4.dp, vertical = 1.dp))
   363→                    }
   364→                }
   365→                Spacer(Modifier.weight(1f))
   366→                IconButton(onClick = { showGrid = true }, modifier = Modifier.size(24.dp)) {
   367→                    Icon(Icons.Default.GridView, null, tint = Color(0xFF64748B), modifier = Modifier.size(14.dp))
   368→                }
   369→            }
   370→        }
   371→        Box(modifier = Modifier.fillMaxWidth().height(2.dp).background(Color.Black))
   372→
   373→        // ── 3. MAIN CONTENT ── 60% question (left, scrollable) | 40% answers (right, scrollable)
   374→        // DreamKorea logo watermark in background (faded, centered)
   375→        Box(modifier = Modifier.weight(1f).fillMaxWidth()) {
   376→            // Watermark logo at the center of the background
   377→            Image(
   378→                painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.dreamkorea_logo),
   379→                contentDescription = null,
   380→                modifier = Modifier.align(Alignment.Center).size(280.dp).alpha(0.08f),
   381→                contentScale = ContentScale.Fit
   382→            )
   383→            Row(modifier = Modifier.fillMaxSize()) {
   384→            // LEFT: Question content (60%) — scrollable so long titles/stems/images don't get cut
   385→            Column(
   386→                modifier = Modifier.weight(0.6f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   387→                verticalArrangement = Arrangement.Center,
   388→                horizontalAlignment = Alignment.CenterHorizontally
   389→            ) {
   390→                // Question title (if set by admin) — shown at the top
   391→                if (!q.title.isNullOrBlank()) {
   392→                    Text(
   393→                        q.title,
   394→                        color = Color(0xFF003478),
   395→                        fontSize = 13.sp,
   396→                        fontWeight = FontWeight.Bold,
   397→                        modifier = Modifier.fillMaxWidth(0.92f).padding(bottom = 4.dp)
   398→                    )
   399→                }
   400→                // Question text (stem) — shown in a card with border/shadow
   401→                val questionText = q.stem.ifBlank { q.mediaText ?: "" }
   402→                if (questionText.isNotBlank()) {
   403→                    Surface(
   404→                        color = Color.White,
   405→                        shape = RoundedCornerShape(14.dp),
   406→                        shadowElevation = 2.dp,
   407→                        border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   408→                        modifier = Modifier.fillMaxWidth(0.92f)
   409→                    ) {
   410→                        Text(
   411→                            questionText,
   412→                            color = Color(0xFF1E293B),
   413→                            fontSize = 14.sp,
   414→                            fontWeight = FontWeight.Medium,
   415→                            modifier = Modifier.padding(14.dp)
   416→                        )
   417→                    }
   418→                    Spacer(Modifier.height(6.dp))
   419→                }
   420→                // Media images — ContentScale.Fit (contain, not cover)
   421→                if (q.descType == "image" && !q.descImageUrl.isNullOrBlank()) {
   422→                    val url = q.descImageUrl!!.toAbsoluteUrl()
   423→                    coil.compose.AsyncImage(
   424→                        model = url, contentDescription = null,
   425→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 120.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(url) },
   426→                        contentScale = ContentScale.Fit
   427→                    )
   428→                }
   429→                val mediaImgUrl = (q.mediaImageUrl ?: q.imageUrl)?.toAbsoluteUrl()
   430→                if (!mediaImgUrl.isNullOrBlank()) {
   431→                    coil.compose.AsyncImage(
   432→                        model = mediaImgUrl, contentDescription = null,
   433→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 140.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(mediaImgUrl) },
   434→                        contentScale = ContentScale.Fit
   435→                    )
   436→                }
   437→                val mediaAudUrl = (q.mediaAudioUrl ?: q.audioUrl)?.toAbsoluteUrl()
   438→                if (!mediaAudUrl.isNullOrBlank()) {
   439→                    // key(q.id) ensures the AudioPlayerCard is FULLY RECREATED when
   440→                    // the question changes — state (playCount, disabled) resets completely.
   441→                    // This fixes the bug where audio was locked from the previous question.
   442→                    key(q.id) {
   443→                        AudioPlayerCard(theme = theme, url = mediaAudUrl, loopCount = q.audioLoop, loopDelaySec = if (q.audioLoopDelay > 0) q.audioLoopDelay else 2, sound = sound, questionId = q.id, playCounts = audioPlayCounts, onPlayingChange = { audioPlaying = it })
   444→                    }
   445→                }
   446→            }
   447→
   448→            // Vertical divider
   449→            Box(modifier = Modifier.width(2.dp).fillMaxHeight().background(Color.Black))
   450→
   451→            // RIGHT: Answer options (40%) — scrollable so long options don't get cut
   452→            Column(
   453→                modifier = Modifier.weight(0.4f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   454→                verticalArrangement = Arrangement.Center
   455→            ) {
   456→                when (q.answerType) {
   457→                    "text", "choose" -> {
   458→                        (0 until minOf(4, options.size)).forEach { i ->
   459→                            val isSelected = answers[q.id] == options.getOrNull(i)
   460→                            val optText = options.getOrNull(i) ?: ""
   461→                            val blankWord = q.optionBlanks.getOrNull(i)?.takeIf { it.isNotBlank() }
   462→                            Row(
   463→                                modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp).clickable { sound.click(); answers[q.id] = optText },
   464→                                verticalAlignment = Alignment.CenterVertically
   465→                            ) {
   466→                                Surface(
   467→                                    color = if (isSelected) theme.primary else Color.White,
   468→                                    border = androidx.compose.foundation.BorderStroke(2.dp, Color.Black),
   469→                                    shape = androidx.compose.foundation.shape.CircleShape,
   470→                                    modifier = Modifier.size(40.dp)
   471→                                ) {
   472→                                    Box(contentAlignment = Alignment.Center) { Text("${i+1}", color = if (isSelected) Color.White else Color.Black, fontSize = 15.sp, fontWeight = FontWeight.Bold) }
   473→                                }
   474→                                Spacer(Modifier.width(8.dp))

   842→/// Reference-style 4-column grid of PERFECT SQUARE cells with 2px black borders.
   843→/// Solved = blue fill + white text. Current = blue border + glow.
   844→/// Uses continuous numbering: Reading 1-20, Listening 21-40 (from globalIdx).
   845→/// showAllBlocks: true = pad grid to expected count (20) with blank cells so
   846→/// the user sees which questions are added vs blank.
   847→/// false = only show cells for questions that actually exist.
   848→/// Cells ALWAYS maintain 1:1 aspect ratio (perfect square) regardless of count.
   849→/// Grid is scrollable so all blocks are reachable.
   850→@Composable
   851→private fun QuestionGridRef(
   852→    test: TestDetail,
   853→    items: List<TestItemDetail>,
   854→    answers: SnapshotStateMap<String, Any>,
   855→    currentIdx: Int,
   856→    sound: SoundManager,
   857→    haptic: androidx.compose.ui.hapticfeedback.HapticFeedback,
   858→    filterMode: Boolean?,
   859→    accentBlue: Color,
   860→    showAllBlocks: Boolean,
   861→    onPick: (Int) -> Unit,
   862→) {
   863→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   864→    val cols = 4  // 4 columns per the HTML reference
   865→    val gridScrollState = rememberScrollState()
   866→
   867→    // Determine total cells to render
   868→    val expectedTotal = if (showAllBlocks) {
   869→        val isAudio = items.isNotEmpty() && items[0].question.blockType == "audio"
   870→        val blockCount = if (isAudio) test.audioBlockCount else test.textBlockCount
   871→        maxOf(blockCount, items.size)
   872→    } else {
   873→        items.size
   874→    }
   875→    val rowsCount = (expectedTotal + cols - 1) / cols
   876→
   877→    // Scrollable grid — cells are PERFECT SQUARES (aspectRatio 1f)
   878→    Column(
   879→        modifier = Modifier
   880→            .fillMaxWidth()
   881→            .verticalScroll(gridScrollState),
   882→        verticalArrangement = Arrangement.spacedBy(8.dp)
   883→    ) {
   884→        for (rowIdx in 0 until rowsCount) {
   885→            Row(
   886→                modifier = Modifier.fillMaxWidth(),
   887→                horizontalArrangement = Arrangement.spacedBy(8.dp)
   888→            ) {
   889→                for (colIdx in 0 until cols) {
   890→                    val localIdx = rowIdx * cols + colIdx
   891→                    if (localIdx < items.size) {
   892→                        val globalIdx = globalIndices[localIdx]
   893→                        // DISPLAY NUMBER: Reading shows 1-20, Listening shows 21-40
   894→                        // Uses blockNumber (1-20 within each block) + 20 offset for audio
   895→                        val q = items[localIdx].question
   896→                        val displayNum = if (q.blockType == "audio") {
   897→                            (q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)) + 20
   898→                        } else {
   899→                            q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)
   900→                        }
   901→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   902→                        val isCurrent = globalIdx == currentIdx
   903→                        val isFilteredOut = when (filterMode) {
   904→                            true -> !isAnswered
   905→                            false -> isAnswered
   906→                            null -> false
   907→                        }
   908→                        Box(
   909→                            modifier = Modifier
   910→                                .weight(1f)
   911→                                .aspectRatio(1f)  // PERFECT SQUARE — always 1:1
   912→                                .clip(RoundedCornerShape(6.dp))
   913→                                .border(
   914→                                    width = if (isCurrent) 2.5.dp else 1.5.dp,
   915→                                    color = when {
   916→                                        isCurrent -> accentBlue
   917→                                        isAnswered -> accentBlue
   918→                                        else -> Color(0xFF111111)
   919→                                    },
   920→                                    shape = RoundedCornerShape(6.dp)
   921→                                )
   922→                                .background(if (isAnswered) accentBlue else Color.White)
   923→                                .alpha(if (isFilteredOut) 0.15f else 1f)
   924→                                .clickable(enabled = !isFilteredOut) {
   925→                                    sound.click()
   926→                                    haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
   927→                                    onPick(globalIdx)
   928→                                },
   929→                            contentAlignment = Alignment.Center
   930→                        ) {
   931→                            Text(
   932→                                "$displayNum",
   933→                                color = if (isAnswered) Color.White else Color(0xFF111111),
   934→                                fontSize = 15.sp,
   935→                                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Medium,
   936→                            )
   937→                        }
   938→                    } else if (showAllBlocks) {
   939→                        // Empty placeholder cell — perfect square
   940→                        // Reading placeholders: 1-20, Listening placeholders: 21-40
   941→                        val isAudioGrid = items.isNotEmpty() && items[0].question.blockType == "audio"
   942→                        val placeholderNum = if (isAudioGrid) localIdx + 21 else localIdx + 1
   943→                        Box(
   944→                            modifier = Modifier
   945→                                .weight(1f)
   946→                                .aspectRatio(1f)  // PERFECT SQUARE
   947→                                .clip(RoundedCornerShape(6.dp))
   948→                                .border(1.5.dp, Color(0xFFEEEEEE), RoundedCornerShape(6.dp))
   949→                                .background(Color(0xFFFAFAFA)),
   950→                            contentAlignment = Alignment.Center
   951→                        ) {
   952→                            Text(
   953→                                "$placeholderNum",

  1000→// ─── Audio player with loop support ───────────────────────────────────────────
  1001→// loopCount = total number of times to play the audio:
  1002→//   0 or 1 = plays once
  1003→//   2 = plays twice
  1004→//   N = plays N times
  1005→//  -1 = infinite loop
  1006→//
  1007→// isReview: when true the button never dims and the student can replay as
  1008→// many times as they want (used on the post-exam Review screen).
  1009→@Composable
  1010→fun AudioPlayerCard(
  1011→    theme: AppTheme,
  1012→    url: String,
  1013→    loopCount: Int,
  1014→    loopDelaySec: Int,
  1015→    sound: SoundManager,
  1016→    questionId: String? = null,
  1017→    playCounts: SnapshotStateMap<String, Int>? = null,
  1018→    onPlayingChange: ((Boolean) -> Unit)? = null,
  1019→    isReview: Boolean = false,
  1020→) {
  1021→    val context = LocalContext.current
  1022→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
  1023→    var isPlaying by remember { mutableStateOf(false) }
  1024→
  1025→    // PERSISTENT play count — stored in parent map (survives navigation).
  1026→    // Prevents cheat where student navigates away and back to reset plays.
  1027→    val persistentCount = playCounts?.get(questionId) ?: 0
  1028→    // Sync isPlaying with parent callback
  1029→    LaunchedEffect(isPlaying) {
  1030→        onPlayingChange?.invoke(isPlaying)
  1031→    }
  1032→
  1033→    val maxPlays = if (loopCount <= 0) 2 else loopCount
  1034→    // In review mode we never disable — the student may replay indefinitely.
  1035→    val disabled = !isReview && persistentCount >= maxPlays
  1036→    val scope = rememberCoroutineScope()
  1037→
  1038→    // ── Pulse animation while playing (scale 1.0 → 1.2 → 1.0, 600ms repeat) ──
  1039→    val infiniteTransition = rememberInfiniteTransition(label = "audioPulse")
  1040→    val pulseScale by infiniteTransition.animateFloat(
  1041→        initialValue = 1.0f,
  1042→        targetValue = 1.2f,
  1043→        animationSpec = infiniteRepeatable(
  1044→            animation = tween(300, easing = LinearEasing),
  1045→            repeatMode = RepeatMode.Reverse
  1046→        ),
  1047→        label = "pulseScale"
  1048→    )
  1049→    val buttonScale = if (isPlaying) pulseScale else 1.0f
  1050→
  1051→    fun incrementPlayCount() {
  1052→        if (questionId != null && playCounts != null) {
  1053→            playCounts[questionId] = (playCounts[questionId] ?: 0) + 1
  1054→        }
  1055→    }
  1056→
  1057→    DisposableEffect(url) {
  1058→        onDispose {
  1059→            mediaPlayer?.release()
  1060→            mediaPlayer = null
  1061→        }
  1062→    }
  1063→
  1064→    Surface(color = theme.cardBg, shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth(), shadowElevation = 1.dp) {
  1065→        Row(
  1066→            modifier = Modifier.padding(14.dp),
  1067→            verticalAlignment = Alignment.CenterVertically
  1068→        ) {
  1069→            // Audio icon
  1070→            Surface(
  1071→                color = if (disabled) Color(0xFFE2E8F0) else theme.primary.copy(alpha = 0.15f),
  1072→                shape = RoundedCornerShape(10.dp),
  1073→                modifier = Modifier.size(44.dp)
  1074→            ) {
  1075→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
  1076→                    Icon(
  1077→                        Icons.Default.Headphones,
  1078→                        null,
  1079→                        tint = if (disabled) Color(0xFF94A3B8) else theme.primary,
  1080→                        modifier = Modifier.size(22.dp)
  1081→                    )
  1082→                }
  1083→            }
  1084→            Spacer(Modifier.width(12.dp))
  1085→            // Simple label — no play count shown
  1086→            Column(modifier = Modifier.weight(1f)) {
  1087→                Text(
  1088→                    if (disabled) "Audio locked" else if (isPlaying) "Playing…" else "Tap to play",
  1089→                    color = if (disabled) Color(0xFF94A3B8) else theme.darkText,
  1090→                    fontSize = 13.sp,
  1091→                    fontWeight = FontWeight.SemiBold
  1092→                )
  1093→            }
  1094→            // ── Circular play button (FIX-3) ───────────────────────────────
  1095→            // 48dp circle with colored background. While playing: pulsing
  1096→            // scale animation + GraphicEq icon. When disabled (exam mode,
  1097→            // limit reached): dim to 0.3 alpha + PlayArrow (no lock icon).
  1098→            // When isReview: never dim, always full color.
  1099→            Box(
  1100→                modifier = Modifier
  1101→                    .size(48.dp)
  1102→                    .scale(buttonScale)
  1103→                    .alpha(if (disabled) 0.3f else 1.0f)
  1104→                    .clip(androidx.compose.foundation.shape.CircleShape)
  1105→                    .background(if (isPlaying) theme.primary else theme.primary)
  1106→                    .clickable {
  1107→                        if (disabled) return@clickable
  1108→                        sound.click()
  1109→                        if (isPlaying) {
  1110→                            // Don't pause — let it finish. Only allow play, not pause.
  1111→                            return@clickable
  1112→                        }
  1113→                        try {
  1114→                            mediaPlayer?.release()
  1115→                            val mp = android.media.MediaPlayer().apply {
  1116→                                setDataSource(url)
  1117→                                setOnPreparedListener {
  1118→                                    start()
  1119→                                    isPlaying = true
  1120→                                    incrementPlayCount()
  1121→                                }
  1122→                                setOnCompletionListener {
  1123→                                    val currentCount = playCounts?.get(questionId) ?: 0
  1124→                                    if (currentCount < maxPlays || isReview) {
  1125→                                        scope.launch {
  1126→                                            if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
  1127→                                            val latestCount = playCounts?.get(questionId) ?: 0
  1128→                                            if (latestCount < maxPlays || isReview) {
  1129→                                                incrementPlayCount()
  1130→                                                start()
  1131→                                            } else {
  1132→                                                isPlaying = false
  1133→                                            }
  1134→                                        }
  1135→                                    } else {
  1136→                                        isPlaying = false
  1137→                                    }
  1138→                                }
  1139→                                setOnErrorListener { _, _, _ ->
  1140→                                    isPlaying = false
  1141→                                    true
  1142→                                }
  1143→                                prepareAsync()
  1144→                            }
  1145→                            mediaPlayer = mp
  1146→                        } catch (_: Exception) {
  1147→                            isPlaying = false
  1148→                        }
  1149→                    },
  1150→                contentAlignment = Alignment.Center
  1151→            ) {
  1152→                Icon(
  1153→                    if (isPlaying) Icons.Default.GraphicEq else Icons.Default.PlayArrow,
  1154→                    null,
  1155→                    tint = Color.White,
  1156→                    modifier = Modifier.size(24.dp)
  1157→                )
  1158→            }
  1159→        }
  1160→    }
  1161→}

  1775→private fun formatAnswer(answer: Any?): String? {
  1776→    if (answer == null) return null
  1777→    return when (answer) {
  1778→        is String -> answer
  1779→        is List<*> -> answer.joinToString(", ")
  1780→        else -> answer.toString()
  1781→    }
  1782→}

  1784→/// Renders an answer — shows media (image/audio) for URL answers, text otherwise.
  1785→@Composable
  1786→private fun AnswerDisplay(answer: Any?, theme: AppTheme, sound: SoundManager? = null) {
  1787→    if (answer == null) {
  1788→        Text("—", color = theme.subText, fontSize = 12.sp)
  1789→        return
  1790→    }
  1791→    when (answer) {
  1792→        is String -> {
  1793→            val url = answer.trim()
  1794→            if (url.startsWith("http") && (url.contains(".jpg") || url.contains(".jpeg") || url.contains(".png") || url.contains(".gif") || url.contains(".webp"))) {
  1795→                coil.compose.AsyncImage(
  1796→                    model = url, contentDescription = "Answer image",
  1797→                    modifier = Modifier.fillMaxWidth().heightIn(max = 80.dp).clip(RoundedCornerShape(6.dp)),
  1798→                    contentScale = ContentScale.Fit
  1799→                )
  1800→            } else if (url.startsWith("http") && (url.contains(".mp3") || url.contains(".wav") || url.contains(".ogg") || url.contains(".m4a"))) {
  1801→                AudioPlayerCard(theme = theme, url = url, loopCount = 1, loopDelaySec = 0, sound = sound ?: rememberSoundManager(), isReview = true)
  1802→            } else {
  1803→                Text(url, color = theme.darkText, fontSize = 12.sp, fontWeight = FontWeight.Medium)
  1804→            }
  1805→        }
  1806→        is List<*> -> {
  1807→            Column { answer.forEach { Text(it?.toString() ?: "", color = theme.darkText, fontSize = 12.sp) } }
  1808→        }
  1809→        else -> Text(answer.toString(), color = theme.darkText, fontSize = 12.sp)
  1810→    }
  1811→}
