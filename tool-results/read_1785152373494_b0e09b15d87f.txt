   105→            is Screen.Exam -> BottomTab.Exams

   107→            is Screen.Learn -> BottomTab.Tools
   108→            else -> activeTab
   109→        }
   110→    }
   111→
   112→    Surface(modifier = Modifier.fillMaxSize(), color = BgGray) {
   113→        Column(modifier = Modifier.fillMaxSize()) {

   116→                is Screen.Exam -> false
   117→                is Screen.BookReader -> false
   118→                is Screen.Home -> true

   210→// ─── Bottom Navigation Bar — clean, full-width ────────────────────────────────
   211→@Composable
   212→fun BottomNavBar(activeTab: BottomTab, onTabClick: (BottomTab) -> Unit) {
   213→    Surface(
   214→        color = CardWhite,
   215→        shadowElevation = 8.dp,
   216→        tonalElevation = 0.dp,
   217→    ) {
   218→        Row(
   219→            modifier = Modifier
   220→                .fillMaxWidth()
   221→                .navigationBarsPadding()
   222→                .padding(vertical = 8.dp),
   223→            horizontalArrangement = Arrangement.SpaceEvenly,
   224→            verticalAlignment = Alignment.CenterVertically,
   225→        ) {
   226→            BottomTab.values().forEach { tab ->
   227→                val isActive = tab == activeTab
   228→                val iconColor = if (isActive) NavyBlue else Color(0xFF94A3B8)
   229→                val textColor = if (isActive) NavyBlue else Color(0xFF94A3B8)
   230→
   231→                // Animated scale on press
   232→                var pressed by remember { mutableStateOf(false) }
   233→                val scale by animateFloatAsState(
   234→                    targetValue = if (pressed) 0.85f else 1f,
   235→                    animationSpec = spring(dampingRatio = Spring.DampingRatioMediumBouncy),
   236→                    label = "navScale"
   237→                )
   238→                // Animated background for active tab
   239→                val bgAlpha by animateFloatAsState(
   240→                    targetValue = if (isActive) 1f else 0f,
   241→                    animationSpec = tween(300, easing = FastOutSlowInEasing),
   242→                    label = "navBgAlpha"
   243→                )
   244→
   245→                Box(
   246→                    modifier = Modifier
   247→                        .clip(RoundedCornerShape(16.dp))
   248→                        .background(NavyBlue.copy(alpha = 0.08f * bgAlpha))
   249→                        .clickable { onTabClick(tab) }
   250→                        .padding(horizontal = 16.dp, vertical = 6.dp)
   251→                        .scale(scale),
   252→                    contentAlignment = Alignment.Center,
   253→                ) {
   254→                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
   255→                        Icon(
   256→                            tab.icon,
   257→                            tab.label,
   258→                            tint = iconColor,
   259→                            modifier = Modifier.size(24.dp),
   260→                        )
   261→                        Spacer(Modifier.height(3.dp))
   262→                        Text(
   263→                            tab.label,
   264→                            color = textColor,
   265→                            fontSize = 10.sp,
   266→                            fontWeight = if (isActive) FontWeight.SemiBold else FontWeight.Normal,
   267→                        )
   268→                    }
   269→                }
   270→            }
   271→        }
   272→    }
   273→}

   303→// Async image loader — uses Coil
   304→@Composable
   305→fun AsyncImageLoader(url: String, modifier: Modifier = Modifier) {
   306→    val absoluteUrl = if (url.isBlank()) "" else if (url.startsWith("http")) url else "https://my-project-five-sepia.vercel.app$url"
   307→    if (absoluteUrl.isBlank()) {
   308→        Box(modifier = modifier.background(NavyBlue.copy(alpha = 0.05f)), contentAlignment = Alignment.Center) {
   309→            Icon(Icons.Default.Image, null, tint = NavyBlue.copy(alpha = 0.2f), modifier = Modifier.size(32.dp))
   310→        }
   311→        return
   312→    }
   313→    // Use Crop so the image fills the entire card — no empty space, no "cut" look
   314→    coil.compose.AsyncImage(model = absoluteUrl, contentDescription = null, modifier = modifier, contentScale = ContentScale.Crop)
   315→}

   317→// ─── HOME SCREEN — Professional EdTech design ─────────────────────────────────
   318→@Composable
   319→fun HomeScreen(theme: AppTheme, sound: SoundManager, userName: String, onNavigate: (Screen) -> Unit) {
   320→    var stats by remember { mutableStateOf<UserStats?>(null) }
   321→    var homeCards by remember {
   322→        mutableStateOf<List<HomeCard>>(AppState.getCachedNow<List<HomeCard>>(AppState.KEY_HOME_CARDS) ?: emptyList())
   323→    }
   324→    var loading by remember { mutableStateOf(homeCards.isEmpty()) }
   325→
   326→    LaunchedEffect(Unit) {
   327→        if (homeCards.isNotEmpty()) loading = false
   328→        try {
   329→            homeCards = AppState.getCachedHomeCards()
   330→            stats = AppState.api.getStats().stats
   331→        } catch (_: Exception) {}
   332→        finally { loading = false }
   333→    }
   334→
   335→    val effectiveCards = remember(homeCards, loading) {
   336→        if (loading) emptyList()
   337→        else if (homeCards.isNotEmpty()) {
   338→            // Merge server cards with guaranteed QBank + Batch cards
   339→            val serverKeys = homeCards.map { it.key }.toSet()
   340→            val serverRoutes = homeCards.map { it.route }.toSet()
   341→            val extras = mutableListOf<HomeCard>()
   342→            // Only add QBank card if server doesn't have one (by key OR by route)
   343→            if (!serverKeys.contains("qbank_packages") && !serverRoutes.contains("questionbank")) {
   344→                extras.add(HomeCard(key = "qbank_packages", title = "Question Bank", section = "test", sortOrder = 99, route = "questionbank", imageUrl = ""))
   345→            }
   346→            // Only add Batch card if server doesn't have one (by key OR by route)
   347→            if (!serverKeys.contains("batch_packages") && !serverRoutes.contains("batch")) {
   348→                extras.add(HomeCard(key = "batch_packages", title = "Batch", section = "test", sortOrder = 100, route = "batch", imageUrl = ""))
   349→            }
   350→            homeCards + extras
   351→        } else listOf(
   352→            HomeCard(key = "ubt_test", title = "UBT Test", section = "test", sortOrder = 0, route = "tests", imageUrl = ""),
   353→            HomeCard(key = "demo_exam", title = "Demo Exam", section = "test", sortOrder = 1, route = "tests", imageUrl = ""),
   354→            HomeCard(key = "qbank_packages", title = "Question Bank", section = "test", sortOrder = 2, route = "questionbank", imageUrl = ""),
   355→            HomeCard(key = "batch_packages", title = "Batch", section = "test", sortOrder = 3, route = "batch", imageUrl = ""),
   356→            HomeCard(key = "results", title = "Results", section = "test", sortOrder = 5, route = "results", imageUrl = ""),
   357→            HomeCard(key = "all_books", title = "Books", section = "resources", sortOrder = 0, route = "books", imageUrl = ""),
   358→            HomeCard(key = "eye_vision", title = "Eye Vision", section = "resources", sortOrder = 2, route = "eyevision", imageUrl = ""),
   359→            HomeCard(key = "join", title = "Join Live", section = "resources", sortOrder = 3, route = "join", imageUrl = "")
   360→        )
   361→    }
   362→
   363→    if (loading) {
   364→        Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
   365→            CircularProgressIndicator(color = NavyBlue)
   366→        }
   367→        return
   368→    }
   369→
   370→    val sections = effectiveCards.groupBy { it.section }
   371→
   372→    LazyColumn(
   373→        modifier = Modifier.fillMaxSize(),
   374→        contentPadding = PaddingValues(bottom = 100.dp)
   375→    ) {
   376→        // ─── Hero card — with tree image background + greeting + progress ─────
   377→        item {
   378→            Surface(
   379→                shape = RoundedCornerShape(20.dp),
   380→                modifier = Modifier.fillMaxWidth().padding(16.dp),
   381→                shadowElevation = 6.dp,
   382→            ) {
   383→                Box(modifier = Modifier.fillMaxWidth().height(160.dp)) {
   384→                    // Tree image as background — Crop to fill the entire card
   385→                    // so it blends with the gradient overlay nicely
   386→                    Image(
   387→                        painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.hero_bg),
   388→                        contentDescription = null,
   389→                        modifier = Modifier.fillMaxSize(),
   390→                        contentScale = ContentScale.Crop
   391→                    )
   392→                    // Gradient overlay — pink/magenta to match the tree image
   393→                    Box(
   394→                        modifier = Modifier.fillMaxSize().background(
   395→                            Brush.verticalGradient(
   396→                                listOf(Color(0xFF8B2252).copy(alpha = 0.75f), Color(0xFF1E3A8A).copy(alpha = 0.85f))
   397→                            )
   398→                        )
   399→                    )

   402→                        Text("Hello, $userName", color = Color.White, fontSize = 22.sp, fontWeight = FontWeight.Bold)

   405→                        Row(verticalAlignment = Alignment.CenterVertically) {
   406→                            Surface(color = Color.White.copy(alpha = 0.2f), shape = RoundedCornerShape(8.dp)) {
   407→                                Text("${stats?.totalExamsTaken ?: 0} exams", color = Color.White, fontSize = 11.sp, modifier = Modifier.padding(horizontal = 8.dp, vertical = 3.dp))

   443→        item {
   444→            Surface(

   760→@Composable
   761→fun TestsScreen(theme: AppTheme, sound: SoundManager, filter: String = "all", title: String = "All Exams", onBack: () -> Unit, onStartExam: (String) -> Unit, onOpenPackages: () -> Unit = {}) {
   762→    // Local filter state — allows in-screen filtering via tabs
   763→    var activeFilter by remember { mutableStateOf(filter) }
   764→    var tests by remember {
   765→        mutableStateOf<List<TestItem>>(AppState.getCachedNow<List<TestItem>>(AppState.keyTests(activeFilter)) ?: emptyList())
   766→    }
   767→    var loading by remember { mutableStateOf(tests.isEmpty()) }
   768→    var error by remember { mutableStateOf("") }
   769→    var retryCount by remember { mutableStateOf(0) }
   770→
   771→    LaunchedEffect(activeFilter, retryCount) {
   772→        if (tests.isNotEmpty()) loading = false else loading = true
   773→        error = ""
   774→        try {
   775→            // Always fetch fresh — admin may have just published a new exam
   776→            AppState.invalidateCache(AppState.keyTests(activeFilter))
   777→            val result = kotlinx.coroutines.withTimeoutOrNull(20_000L) { AppState.getCachedTests(activeFilter) }
   778→            if (result != null) tests = result
   779→            else if (tests.isEmpty()) error = "The request timed out. Check your internet and try again."
   780→        } catch (e: retrofit2.HttpException) {
   781→            if (tests.isEmpty()) error = when (e.code()) { 401 -> "Your session has expired." else -> "Could not load (HTTP ${e.code()})." }
   782→        } catch (e: java.net.UnknownHostException) { if (tests.isEmpty()) error = "No internet connection." }
   783→        catch (e: java.net.SocketTimeoutException) { if (tests.isEmpty()) error = "The request timed out." }
   784→        catch (e: java.io.IOException) { if (tests.isEmpty()) error = "Network error." }
   785→        catch (e: Exception) { if (tests.isEmpty()) error = "Unexpected error." }
   786→        finally { loading = false }
   787→    }
   788→
   789→    if (loading) {
   790→        Column(Modifier.fillMaxSize()) {
   791→            LinearProgressIndicator(modifier = Modifier.fillMaxWidth(), color = NavyBlue, trackColor = NavyBlue.copy(alpha = 0.1f))

   812→        return
   813→    }
   814→
   815→    LazyColumn(modifier = Modifier.fillMaxSize(), contentPadding = PaddingValues(bottom = 100.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
   816→        item {

   829→                    horizontalArrangement = Arrangement.spacedBy(8.dp)
   830→                ) {

   832→                    filters.forEach { (key, label) ->
   833→                        val isActive = activeFilter == key
