   567→// ─── Tests Screen ─────────────────────────────────────────────────────────────
   568→@Composable
   569→fun TestsScreen(theme: AppTheme, sound: SoundManager, filter: String = "all", title: String = "Tests & Exams", onBack: () -> Unit, onStartExam: (String) -> Unit) {
   570→    // Stale-while-revalidate: show cached data instantly
   571→    var tests by remember {
   572→        mutableStateOf<List<TestItem>>(AppState.getCachedNow<List<TestItem>>(AppState.keyTests(filter)) ?: emptyList())
   573→    }
   574→    var loading by remember { mutableStateOf(tests.isEmpty()) }
   575→    var error by remember { mutableStateOf("") }
   576→    var retryCount by remember { mutableStateOf(0) }
   577→
   578→    LaunchedEffect(filter, retryCount) {
   579→        if (tests.isNotEmpty()) loading = false
   580→        else loading = true
   581→        error = ""
   582→        try {
   583→            if (retryCount > 0) AppState.invalidateCache(AppState.keyTests(filter))
   584→            val result = kotlinx.coroutines.withTimeoutOrNull(20_000L) {
   585→                AppState.getCachedTests(filter)
   586→            }
   587→            if (result != null) {
   588→                tests = result
   589→            } else if (tests.isEmpty()) {
   590→                error = "The request timed out. Check your internet and try again."
   591→            }
   592→        } catch (e: retrofit2.HttpException) {
   593→            if (tests.isEmpty()) {
   594→                error = when (e.code()) {
   595→                    401 -> "Your session has expired. Please log out and sign in again."
   596→                    else -> "Could not load tests (HTTP ${e.code()})."
   597→                }
   598→            }
   599→        } catch (e: java.net.UnknownHostException) {
   600→            if (tests.isEmpty()) error = "No internet connection. Please check your network."
   601→        } catch (e: java.net.SocketTimeoutException) {
   602→            if (tests.isEmpty()) error = "The request timed out. Please try again."
   603→        } catch (e: java.io.IOException) {
   604→            if (tests.isEmpty()) error = "Network error: ${e.message ?: "Could not connect."}"
   605→        } catch (e: Exception) {
   606→            if (tests.isEmpty()) error = "Unexpected error: ${e.message ?: "Please try again."}"
   607→        } finally {
   608→            loading = false
   609→        }
   610→    }
   611→
   612→    if (loading) {
   613→        Column(Modifier.fillMaxSize()) {
   614→            LinearProgressIndicator(
   615→                modifier = Modifier.fillMaxWidth(),
   616→                color = theme.primary,
   617→                trackColor = theme.primary.copy(alpha = 0.1f),
   618→            )
   619→            ScreenHeader(theme, sound, title, "Tap a test to start.", onBack)
   620→            SkeletonListScreen(theme, itemCount = 5)
   621→        }
   622→        return
   623→    }
   624→
   625→    if (error.isNotEmpty()) {
   626→        Column(
   627→            Modifier.fillMaxSize().padding(24.dp),
   628→            horizontalAlignment = Alignment.CenterHorizontally,
   629→            verticalArrangement = Arrangement.Center
   630→        ) {
   631→            Icon(Icons.Default.CloudOff, null, tint = theme.errorRed.copy(alpha = 0.7f), modifier = Modifier.size(56.dp))
   632→            Spacer(Modifier.height(16.dp))
   633→            Text("Couldn't load tests", color = theme.darkText, fontSize = 18.sp, fontWeight = FontWeight.Bold)
   634→            Spacer(Modifier.height(8.dp))
   635→            Text(error, color = theme.subText, fontSize = 13.sp, textAlign = TextAlign.Center, modifier = Modifier.padding(horizontal = 16.dp))
   636→            Spacer(Modifier.height(24.dp))
   637→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   638→                OutlinedButton(onClick = onBack, shape = RoundedCornerShape(10.dp)) { Text("Go back") }
   639→                Button(
   640→                    onClick = { sound.click(); retryCount++ },
   641→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   642→                    shape = RoundedCornerShape(10.dp)
   643→                ) {
   644→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   645→                    Spacer(Modifier.width(6.dp))
   646→                    Text("Retry")
   647→                }
   648→            }
   649→        }
   650→        return
   651→    }
   652→
   653→    LazyColumn(
   654→        modifier = Modifier.fillMaxSize(),
   655→        contentPadding = PaddingValues(16.dp),
   656→        verticalArrangement = Arrangement.spacedBy(10.dp)
   657→    ) {
   658→        item { ScreenHeader(theme, sound, title, if (tests.isEmpty()) "No tests in this category yet." else "Tap a test to start. Stats update automatically.", onBack) }
   659→        if (tests.isEmpty()) {
   660→            item { EmptyState(theme, "Nothing here yet", "Your teacher will add content to this section soon.", Icons.Default.Quiz) }
   661→        } else {
   662→            itemsIndexed(tests) { i, t ->
   663→                AnimatedListItem(index = i, theme = theme) {
   664→                    TestCard(theme, sound, t, onClick = { onStartExam(t.id) })
   665→                }
   666→            }
   667→        }
   668→    }
   669→}
