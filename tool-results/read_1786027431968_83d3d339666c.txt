    53→    53→/**
    54→    54→ * Exam taking screen — full flow:
    55→    55→ * 1. Loading skeleton (with timeout + retry)

    64→    64→@Composable

    66→    66→    val sound = rememberSoundManager()

    71→    71→    var error by remember { mutableStateOf("") }

    73→    73→    val answers = remember { mutableStateMapOf<String, Any>() }
    74→    74→    // Persistent audio play counts per question ID — survives navigation.
    75→    75→    // Prevents cheat where student navigates away and back to reset plays.
    76→    76→    val audioPlayCounts = remember { mutableStateMapOf<String, Int>() }
    77→    77→    // When audio is playing, disable navigation buttons

    82→    82→    var questionFeedback by remember { mutableStateOf<QuestionFeedback?>(null) }

    85→    85→
    86→    86→    // ── ORIENTATION ── FORCE LANDSCAPE. No onDispose PORTRAIT —

    89→    89→        val activity = context as? Activity

    92→    92→    }

   115→   115→    }

   117→   117→    LaunchedEffect(testId, retryCount) {

   119→   119→        error = ""

   121→   121→            // 30-second timeout — combined exams may need multiple requests
   122→   122→            val result = withTimeoutOrNull(30_000L) {
   123→   123→                when {
   124→   124→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   125→   125→                    testId == "qbank-combined" -> {
   126→   126→                        try {
   127→   127→                            AppState.api.getQBankCombined().test
   128→   128→                        } catch (e: retrofit2.HttpException) {
   129→   129→                            if (e.code() == 404) buildQBankCombinedClientSide()
   130→   130→                            else throw e
   131→   131→                        }
   132→   132→                    }
   133→   133→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   134→   134→                    testId.startsWith("bundle-") -> {
   135→   135→                        val bundleId = testId.removePrefix("bundle-")
   136→   136→                        try {
   137→   137→                            AppState.api.getBundleCombined(bundleId).test
   138→   138→                        } catch (e: retrofit2.HttpException) {
   139→   139→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   140→   140→                            else throw e
   141→   141→                        }
   142→   142→                    }
   143→   143→                    // Normal test — fetch by ID
   144→   144→                    else -> AppState.api.getTestDetail(testId).test
   145→   145→                }

   151→   151→                error = "The request timed out. Check your internet connection and try again."
   152→   152→            }
   153→   153→        } catch (e: retrofit2.HttpException) {
   154→   154→            // Try to read the actual error message from the response body
   155→   155→            val rawBody = try { e.response()?.errorBody()?.string() } catch (_: Exception) { null }

   157→   157→                401 -> "Your session has expired. Please log out and sign in again."
   158→   158→                403 -> "This exam is not available right now."
   159→   159→                404 -> "This test could not be found. It may have been removed."
   160→   160→                500 -> "Server error. Please try again in a moment.${if (rawBody != null) " ($rawBody)" else ""}"
   161→   161→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   162→   162→            }
   163→   163→        } catch (e: java.net.UnknownHostException) {
   164→   164→            error = "No internet connection. Please check your network and try again."
   165→   165→        } catch (e: java.net.SocketTimeoutException) {
   166→   166→            error = "The request timed out. Please try again."
   167→   167→        } catch (e: java.io.IOException) {
   168→   168→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   169→   169→        } catch (e: Exception) {
   170→   170→            error = "Unexpected error: ${e.message ?: "Please try again."}"
   171→   171→        } finally {
   172→   172→            // GUARANTEED: loading is always reset, even if the coroutine is cancelled
   173→   173→            loading = false
   174→   174→        }
   175→   175→    }

   183→   183→            }

   192→   192→                        submitCombinedExamWithFallback(currentTest, answers.toMap())
   193→   193→                    } else {
   194→   194→                        submitExamWithFallback(currentTest, answers.toMap())
   195→   195→                    }
   196→   196→                    sound.success()
   197→   197→                } catch (e: java.net.UnknownHostException) {
   198→   198→                    error = "No internet connection. Could not submit your answers."
   199→   199→                } catch (e: java.io.IOException) {
   200→   200→                    error = "Network error. Could not submit your answers."
   201→   201→                } catch (e: Exception) {
   202→   202→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   203→   203→                }
   204→   204→                submitting = false
   205→   205→            }
   206→   206→        }
   207→   207→    }
   208→   208→
   209→   209→    if (loading) {
   210→   210→        // Loading skeleton with a subtle "Loading..." label so users know it's working
   211→   211→        Column(Modifier.fillMaxSize()) {
   212→   212→            LinearProgressIndicator(
   213→   213→                modifier = Modifier.fillMaxWidth(),
   214→   214→                color = theme.primary,
   215→   215→                trackColor = theme.primary.copy(alpha = 0.1f),
   216→   216→            )
   217→   217→            SkeletonListScreen(theme, itemCount = 4)
   218→   218→        }
   219→   219→        return
   220→   220→    }
   221→   221→
   222→   222→    if (error.isNotEmpty()) {
   223→   223→        Column(
   224→   224→            Modifier.fillMaxSize().padding(24.dp),
   225→   225→            horizontalAlignment = Alignment.CenterHorizontally,
   226→   226→            verticalArrangement = Arrangement.Center
   227→   227→        ) {
   228→   228→            Icon(
   229→   229→                Icons.Default.CloudOff,
   230→   230→                null,
   231→   231→                tint = theme.errorRed.copy(alpha = 0.7f),
   232→   232→                modifier = Modifier.size(56.dp)
   233→   233→            )
   234→   234→            Spacer(Modifier.height(16.dp))
   235→   235→            Text(
   236→   236→                "Couldn't load the test",
   237→   237→                color = theme.darkText,
   238→   238→                fontSize = 18.sp,
   239→   239→                fontWeight = FontWeight.Bold
   240→   240→            )
   241→   241→            Spacer(Modifier.height(8.dp))
   242→   242→            Text(
   243→   243→                error,
   244→   244→                color = theme.subText,
   245→   245→                fontSize = 13.sp,
   246→   246→                textAlign = TextAlign.Center,
   247→   247→                modifier = Modifier.padding(horizontal = 16.dp)
   248→   248→            )
   249→   249→            Spacer(Modifier.height(8.dp))
   250→   250→            Text(
   251→   251→                "Test ID: $testId",
   252→   252→                color = theme.subText.copy(alpha = 0.5f),
   253→   253→                fontSize = 10.sp,
   254→   254→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   255→   255→            )
   256→   256→            Spacer(Modifier.height(24.dp))
   257→   257→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   258→   258→                OutlinedButton(
   259→   259→                    onClick = onExit,
   260→   260→                    shape = RoundedCornerShape(10.dp)
   261→   261→                ) { Text("Go back") }
   262→   262→                Button(
   263→   263→                    onClick = {
   264→   264→                        sound.click()
   265→   265→                        retryCount++ // triggers LaunchedEffect reload
   266→   266→                    },
   267→   267→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   268→   268→                    shape = RoundedCornerShape(10.dp)
   269→   269→                ) {
   270→   270→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   271→   271→                    Spacer(Modifier.width(6.dp))
   272→   272→                    Text("Retry")

   274→   274→            }
   275→   275→        }
   276→   276→        return

   278→   278→
   279→   279→    val t = test ?: return
   280→   280→    // Guard: if the test has no questions, show a friendly message instead of crashing
   281→   281→    if (t.items.isEmpty()) {

   283→   283→            Modifier.fillMaxSize().padding(20.dp),
   284→   284→            horizontalAlignment = Alignment.CenterHorizontally,
   285→   285→            verticalArrangement = Arrangement.Center
   286→   286→        ) {
   287→   287→            Icon(Icons.Default.Quiz, null, tint = theme.subText, modifier = Modifier.size(48.dp))

   290→   290→            Spacer(Modifier.height(16.dp))
   291→   291→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }
   292→   292→        }
   293→   293→        return
   294→   294→    }
   295→   295→
   296→   296→    val currentQuestion = t.items.getOrNull(currentIdx)
   297→   297→
   298→   298→    // ─── Result screen ──────────────────────────────────────────────────────
   299→   299→    if (submitResult != null) {
   300→   300→        ExamResultScreen(theme, submitResult!!, onExit, sound, examTitle = t.title, examDescription = t.description)

   321→   321→
   322→   322→    Column(modifier = Modifier.fillMaxSize().background(Color(0xFFF8FAFC))) {

   365→   365→                Spacer(Modifier.weight(1f))
   366→   366→                IconButton(onClick = { showGrid = true }, modifier = Modifier.size(24.dp)) {

   371→   371→        Box(modifier = Modifier.fillMaxWidth().height(2.dp).background(Color.Black))

   373→   373→        // ── 3. MAIN CONTENT ── 60% question (left, scrollable) | 40% answers (right, scrollable)

   410→   410→                        Text(

   842→   842→/// Reference-style 4-column grid of PERFECT SQUARE cells with 2px black borders.
   843→   843→/// Solved = blue fill + white text. Current = blue border + glow.
   844→   844→/// Uses continuous numbering: Reading 1-20, Listening 21-40 (from globalIdx).
   845→   845→/// showAllBlocks: true = pad grid to expected count (20) with blank cells so
   846→   846→/// the user sees which questions are added vs blank.
   847→   847→/// false = only show cells for questions that actually exist.
   848→   848→/// Cells ALWAYS maintain 1:1 aspect ratio (perfect square) regardless of count.
   849→   849→/// Grid is scrollable so all blocks are reachable.
   850→   850→@Composable
   851→   851→private fun QuestionGridRef(
   852→   852→    test: TestDetail,
   853→   853→    items: List<TestItemDetail>,
   854→   854→    answers: SnapshotStateMap<String, Any>,
   855→   855→    currentIdx: Int,
   856→   856→    sound: SoundManager,
   857→   857→    haptic: androidx.compose.ui.hapticfeedback.HapticFeedback,
   858→   858→    filterMode: Boolean?,
   859→   859→    accentBlue: Color,
   860→   860→    showAllBlocks: Boolean,
   861→   861→    onPick: (Int) -> Unit,
   862→   862→) {
   863→   863→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   864→   864→    val cols = 4  // 4 columns per the HTML reference
   865→   865→    val gridScrollState = rememberScrollState()
   866→   866→
   867→   867→    // Determine total cells to render
   868→   868→    val expectedTotal = if (showAllBlocks) {
   869→   869→        val isAudio = items.isNotEmpty() && items[0].question.blockType == "audio"
   870→   870→        val blockCount = if (isAudio) test.audioBlockCount else test.textBlockCount
   871→   871→        maxOf(blockCount, items.size)
   872→   872→    } else {
   873→   873→        items.size
   874→   874→    }
   875→   875→    val rowsCount = (expectedTotal + cols - 1) / cols
   876→   876→
   877→   877→    // Scrollable grid — cells are PERFECT SQUARES (aspectRatio 1f)
   878→   878→    Column(
   879→   879→        modifier = Modifier
   880→   880→            .fillMaxWidth()
   881→   881→            .verticalScroll(gridScrollState),
   882→   882→        verticalArrangement = Arrangement.spacedBy(8.dp)
   883→   883→    ) {
   884→   884→        for (rowIdx in 0 until rowsCount) {
   885→   885→            Row(
   886→   886→                modifier = Modifier.fillMaxWidth(),
   887→   887→                horizontalArrangement = Arrangement.spacedBy(8.dp)
   888→   888→            ) {

   890→   890→                    val localIdx = rowIdx * cols + colIdx
   891→   891→                    if (localIdx < items.size) {
   892→   892→                        val globalIdx = globalIndices[localIdx]
   893→   893→                        // DISPLAY NUMBER: Reading shows 1-20, Listening shows 21-40
   894→   894→                        // Uses blockNumber (1-20 within each block) + 20 offset for audio
   895→   895→                        val q = items[localIdx].question
   896→   896→                        val displayNum = if (q.blockType == "audio") {
   897→   897→                            (q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)) + 20

   900→   900→                        }
   901→   901→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   902→   902→                        val isCurrent = globalIdx == currentIdx
   903→   903→                        val isFilteredOut = when (filterMode) {
   904→   904→                            true -> !isAnswered
   905→   905→                            false -> isAnswered
   906→   906→                            null -> false

   912→   912→                                .clip(RoundedCornerShape(6.dp))

   915→   915→                                    color = when {

   918→   918→                                        else -> Color(0xFF111111)
   919→   919→                                    },
   920→   920→                                    shape = RoundedCornerShape(6.dp)
   921→   921→                                )
   922→   922→                                .background(if (isAnswered) accentBlue else Color.White)
   923→   923→                                .alpha(if (isFilteredOut) 0.15f else 1f)
   924→   924→                                .clickable(enabled = !isFilteredOut) {
   925→   925→                                    sound.click()
   926→   926→                                    haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
   927→   927→                                    onPick(globalIdx)
   928→   928→                                },
   929→   929→                            contentAlignment = Alignment.Center
   930→   930→                        ) {
   931→   931→                            Text(
   932→   932→                                "$displayNum",
   933→   933→                                color = if (isAnswered) Color.White else Color(0xFF111111),
   934→   934→                                fontSize = 15.sp,
   935→   935→                                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Medium,
   936→   936→                            )
   937→   937→                        }
   938→   938→                    } else if (showAllBlocks) {
   939→   939→                        // Empty placeholder cell — perfect square
   940→   940→                        // Reading placeholders: 1-20, Listening placeholders: 21-40
   941→   941→                        val isAudioGrid = items.isNotEmpty() && items[0].question.blockType == "audio"
   942→   942→                        val placeholderNum = if (isAudioGrid) localIdx + 21 else localIdx + 1

   958→   958→                        }
   959→   959→                    } else {

   961→   961→                        Spacer(modifier = Modifier.weight(1f).aspectRatio(1f))

   963→   963→                }
   964→   964→            }

  1005→  1005→//  -1 = infinite loop
  1006→  1006→@Composable
  1007→  1007→fun AudioPlayerCard(
  1008→  1008→    theme: AppTheme,
  1009→  1009→    url: String,
  1010→  1010→    loopCount: Int,
  1011→  1011→    loopDelaySec: Int,
  1012→  1012→    sound: SoundManager,
  1013→  1013→    questionId: String? = null,
  1014→  1014→    playCounts: SnapshotStateMap<String, Int>? = null,
  1015→  1015→    onPlayingChange: ((Boolean) -> Unit)? = null,
  1016→  1016→) {
  1017→  1017→    val context = LocalContext.current
  1018→  1018→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
  1019→  1019→    var isPlaying by remember { mutableStateOf(false) }
  1020→  1020→
  1021→  1021→    // PERSISTENT play count — stored in parent map (survives navigation).
  1022→  1022→    // Prevents cheat where student navigates away and back to reset plays.
  1023→  1023→    val persistentCount = playCounts?.get(questionId) ?: 0
  1024→  1024→    // Sync isPlaying with parent callback
  1025→  1025→    LaunchedEffect(isPlaying) {
  1026→  1026→        onPlayingChange?.invoke(isPlaying)
  1027→  1027→    }
  1028→  1028→
  1029→  1029→    val maxPlays = if (loopCount <= 0) 2 else loopCount
  1030→  1030→    val disabled = persistentCount >= maxPlays
  1031→  1031→    val scope = rememberCoroutineScope()
  1032→  1032→
  1033→  1033→    fun incrementPlayCount() {
  1034→  1034→        if (questionId != null && playCounts != null) {
  1035→  1035→            playCounts[questionId] = (playCounts[questionId] ?: 0) + 1
  1036→  1036→        }
  1037→  1037→    }
  1038→  1038→
  1039→  1039→    DisposableEffect(url) {
  1040→  1040→        onDispose {
  1041→  1041→            mediaPlayer?.release()
  1042→  1042→            mediaPlayer = null
  1043→  1043→        }
  1044→  1044→    }
  1045→  1045→
  1046→  1046→    Surface(color = theme.cardBg, shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth(), shadowElevation = 1.dp) {
  1047→  1047→        Row(
  1048→  1048→            modifier = Modifier.padding(14.dp),
  1049→  1049→            verticalAlignment = Alignment.CenterVertically
  1050→  1050→        ) {
  1051→  1051→            // Audio icon
  1052→  1052→            Surface(
  1053→  1053→                color = if (disabled) Color(0xFFE2E8F0) else theme.primary.copy(alpha = 0.15f),
  1054→  1054→                shape = RoundedCornerShape(10.dp),
  1055→  1055→                modifier = Modifier.size(44.dp)
  1056→  1056→            ) {
  1057→  1057→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
  1058→  1058→                    Icon(
  1059→  1059→                        Icons.Default.Headphones,
  1060→  1060→                        null,
  1061→  1061→                        tint = if (disabled) Color(0xFF94A3B8) else theme.primary,
  1062→  1062→                        modifier = Modifier.size(22.dp)
  1063→  1063→                    )
  1064→  1064→                }
  1065→  1065→            }
  1066→  1066→            Spacer(Modifier.width(12.dp))
  1067→  1067→            // Simple label — no play count shown
  1068→  1068→            Column(modifier = Modifier.weight(1f)) {
  1069→  1069→                Text(
  1070→  1070→                    if (disabled) "Audio locked" else "Tap to play",
  1071→  1071→                    color = if (disabled) Color(0xFF94A3B8) else theme.darkText,
  1072→  1072→                    fontSize = 13.sp,
  1073→  1073→                    fontWeight = FontWeight.SemiBold
  1074→  1074→                )
  1075→  1075→            }
  1076→  1076→            // Play/Pause button — DISABLED after all plays used
  1077→  1077→            IconButton(
  1078→  1078→                onClick = {
  1079→  1079→                    if (disabled) return@IconButton
  1080→  1080→                    sound.click()
  1081→  1081→                    if (isPlaying) {
  1082→  1082→                        // Don't pause — let it finish. Only allow play, not pause.
  1083→  1083→                        return@IconButton
  1084→  1084→                    } else {
  1085→  1085→                        try {
  1086→  1086→                            mediaPlayer?.release()
  1087→  1087→                            val mp = android.media.MediaPlayer().apply {
  1088→  1088→                                setDataSource(url)
  1089→  1089→                                setOnPreparedListener {
  1090→  1090→                                    start()
  1091→  1091→                                    isPlaying = true
  1092→  1092→                                    incrementPlayCount()
  1093→  1093→                                }
  1094→  1094→                                setOnCompletionListener {
  1095→  1095→                                    val currentCount = playCounts?.get(questionId) ?: 0
  1096→  1096→                                    if (currentCount < maxPlays) {
  1097→  1097→                                        scope.launch {
  1098→  1098→                                            if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
  1099→  1099→                                            val latestCount = playCounts?.get(questionId) ?: 0
  1100→  1100→                                            if (latestCount < maxPlays) {
  1101→  1101→                                                incrementPlayCount()
  1102→  1102→                                                start()
  1103→  1103→                                            } else {
  1104→  1104→                                                isPlaying = false
  1105→  1105→                                            }
  1106→  1106→                                        }
  1107→  1107→                                    } else {

  1112→  1112→                                    isPlaying = false

  1115→  1115→                                prepareAsync()
  1116→  1116→                            }
  1117→  1117→                            mediaPlayer = mp
  1118→  1118→                        } catch (_: Exception) {
  1119→  1119→                            isPlaying = false
  1120→  1120→                        }
  1121→  1121→                    }
  1122→  1122→                },
  1123→  1123→                enabled = !disabled && !isPlaying
  1124→  1124→            ) {

  1135→  1135→            }
  1136→  1136→        }
  1137→  1137→    }
