    47→/**
    48→ * Exam taking screen — full flow:
    49→ * 1. Loading skeleton (with timeout + retry)
    50→ * 2. Question-by-question with audio (loops N times), image, options
    51→ * 3. Click sound on every interaction
    52→ * 4. Wrong → shows correct answer in red
    53→ * 5. Correct → green confirmation
    54→ * 6. Auto-advances to next question
    55→ * 7. Final score + review screen
    56→ * 8. Stats auto-update via /api/student/tests/[id]/submit
    57→ */
    58→@Composable
    59→fun ExamScreen(theme: AppTheme, testId: String, onExit: () -> Unit) {
    60→    val sound = rememberSoundManager()
    61→    val scope = rememberCoroutineScope()
    62→    val context = LocalContext.current
    63→    var test by remember { mutableStateOf<TestDetail?>(null) }
    64→    var loading by remember { mutableStateOf(true) }
    65→    var error by remember { mutableStateOf("") }
    66→    var currentIdx by remember { mutableStateOf(0) }
    67→    val answers = remember { mutableStateMapOf<String, Any>() }
    68→    var submitResult by remember { mutableStateOf<SubmitResponse?>(null) }
    69→    var submitting by remember { mutableStateOf(false) }
    70→    // Per-question feedback (after answering, before moving on)
    71→    var questionFeedback by remember { mutableStateOf<QuestionFeedback?>(null) }
    72→    // Timer
    73→    var timeLeft by remember { mutableStateOf(0) }
    74→
    75→    // ── ORIENTATION ── FORCE LANDSCAPE. No onDispose PORTRAIT —
    76→    // MainScreen handles portrait when screen changes to non-exam.
    77→    DisposableEffect(Unit) {
    78→        val activity = context as? Activity
    79→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    80→        onDispose { }
    81→    }
    82→    LaunchedEffect(Unit) {
    83→        kotlinx.coroutines.delay(100)
    84→        val activity = context as? Activity
    85→        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE
    86→    }
    87→    // Retry trigger — increment to force reload
    88→    var retryCount by remember { mutableStateOf(0) }
    89→
    90→    // Load test detail — uses LaunchedEffect's own scope, finally block guarantees loading cleanup
    91→    // ─── Back button handler — show exit warning during exam ──────────────
    92→    var showExitDialog by remember { mutableStateOf(false) }
    93→    androidx.activity.compose.BackHandler(enabled = !loading && error.isEmpty()) {
    94→        showExitDialog = true
    95→    }
    96→    if (showExitDialog) {
    97→        androidx.compose.material3.AlertDialog(
    98→            onDismissRequest = { showExitDialog = false },
    99→            title = { Text("Exit Exam?") },
   100→            text = { Text("Your exam is in progress. Are you sure you want to exit? Your answers will be lost.") },
   101→            confirmButton = { Button(onClick = { showExitDialog = false; onExit() }) { Text("Exit") } },
   102→            dismissButton = { OutlinedButton(onClick = { showExitDialog = false }) { Text("Continue Exam") } }
   103→        )
   104→    }
   105→
   106→    LaunchedEffect(testId, retryCount) {
   107→        loading = true
   108→        error = ""
   109→        try {
   110→            // 30-second timeout — combined exams may need multiple requests
   111→            val result = withTimeoutOrNull(30_000L) {
   112→                when {
   113→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   114→                    testId == "qbank-combined" -> {
   115→                        try {
   116→                            AppState.api.getQBankCombined().test
   117→                        } catch (e: retrofit2.HttpException) {
   118→                            if (e.code() == 404) buildQBankCombinedClientSide()
   119→                            else throw e
   120→                        }
   121→                    }
   122→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   123→                    testId.startsWith("bundle-") -> {
   124→                        val bundleId = testId.removePrefix("bundle-")
   125→                        try {
   126→                            AppState.api.getBundleCombined(bundleId).test
   127→                        } catch (e: retrofit2.HttpException) {
   128→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   129→                            else throw e
   130→                        }
   131→                    }
   132→                    // Normal test — fetch by ID
   133→                    else -> AppState.api.getTestDetail(testId).test
   134→                }
   135→            }
   136→            if (result != null) {
   137→                test = result
   138→                timeLeft = (result.durationMin.coerceAtLeast(1)) * 60
   139→            } else {
   140→                error = "The request timed out. Check your internet connection and try again."
   141→            }
   142→        } catch (e: retrofit2.HttpException) {
   143→            // Try to read the actual error message from the response body
   144→            val rawBody = try { e.response()?.errorBody()?.string() } catch (_: Exception) { null }
   145→            error = when (e.code()) {
   146→                401 -> "Your session has expired. Please log out and sign in again."
   147→                403 -> "This exam is not available right now."
   148→                404 -> "This test could not be found. It may have been removed."
   149→                500 -> "Server error. Please try again in a moment.${if (rawBody != null) " ($rawBody)" else ""}"
   150→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   151→            }
   152→        } catch (e: java.net.UnknownHostException) {
   153→            error = "No internet connection. Please check your network and try again."
   154→        } catch (e: java.net.SocketTimeoutException) {
   155→            error = "The request timed out. Please try again."
   156→        } catch (e: java.io.IOException) {
   157→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   158→        } catch (e: Exception) {
   159→            error = "Unexpected error: ${e.message ?: "Please try again."}"
   160→        } finally {
   161→            // GUARANTEED: loading is always reset, even if the coroutine is cancelled
   162→            loading = false
   163→        }
   164→    }
   165→
   166→    // Countdown timer
   167→    LaunchedEffect(test, submitResult) {
   168→        if (test != null && submitResult == null) {
   169→            while (timeLeft > 0) {
   170→                delay(1000)
   171→                timeLeft--
   172→            }
   173→            // Auto-submit when timer reaches zero
   174→            val currentTest = test
   175→            if (timeLeft == 0 && currentTest != null && submitResult == null && !submitting) {
   176→                submitting = true
   177→                sound.swoosh()
   178→                try {
   179→                    // Use fallback for combined exams (qbank-combined / bundle-{id})
   180→                    submitResult = if (currentTest.id == "qbank-combined" || currentTest.id.startsWith("bundle-")) {
   181→                        submitCombinedExamWithFallback(currentTest, answers.toMap())
   182→                    } else {
   183→                        submitExamWithFallback(currentTest, answers.toMap())
   184→                    }
   185→                    sound.success()
   186→                } catch (e: java.net.UnknownHostException) {
   187→                    error = "No internet connection. Could not submit your answers."
   188→                } catch (e: java.io.IOException) {
   189→                    error = "Network error. Could not submit your answers."
   190→                } catch (e: Exception) {
   191→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   192→                }
   193→                submitting = false
   194→            }
   195→        }
   196→    }
   197→
   198→    if (loading) {
   199→        // Loading skeleton with a subtle "Loading..." label so users know it's working
   200→        Column(Modifier.fillMaxSize()) {
   201→            LinearProgressIndicator(
   202→                modifier = Modifier.fillMaxWidth(),
   203→                color = theme.primary,
   204→                trackColor = theme.primary.copy(alpha = 0.1f),
   205→            )
   206→            SkeletonListScreen(theme, itemCount = 4)
   207→        }
   208→        return
   209→    }
   210→
   211→    if (error.isNotEmpty()) {
   212→        Column(
   213→            Modifier.fillMaxSize().padding(24.dp),
   214→            horizontalAlignment = Alignment.CenterHorizontally,
   215→            verticalArrangement = Arrangement.Center
   216→        ) {
   217→            Icon(
   218→                Icons.Default.CloudOff,
   219→                null,
   220→                tint = theme.errorRed.copy(alpha = 0.7f),
   221→                modifier = Modifier.size(56.dp)
   222→            )
   223→            Spacer(Modifier.height(16.dp))
   224→            Text(
   225→                "Couldn't load the test",
   226→                color = theme.darkText,
   227→                fontSize = 18.sp,
   228→                fontWeight = FontWeight.Bold
   229→            )
   230→            Spacer(Modifier.height(8.dp))
   231→            Text(
   232→                error,
   233→                color = theme.subText,
   234→                fontSize = 13.sp,
   235→                textAlign = TextAlign.Center,
   236→                modifier = Modifier.padding(horizontal = 16.dp)
   237→            )
   238→            Spacer(Modifier.height(8.dp))
   239→            Text(
   240→                "Test ID: $testId",
   241→                color = theme.subText.copy(alpha = 0.5f),
   242→                fontSize = 10.sp,
   243→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   244→            )
   245→            Spacer(Modifier.height(24.dp))
   246→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   247→                OutlinedButton(
   248→                    onClick = onExit,
   249→                    shape = RoundedCornerShape(10.dp)
   250→                ) { Text("Go back") }
   251→                Button(
   252→                    onClick = {
   253→                        sound.click()
   254→                        retryCount++ // triggers LaunchedEffect reload
   255→                    },
   256→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   257→                    shape = RoundedCornerShape(10.dp)
   258→                ) {
   259→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))
   260→                    Spacer(Modifier.width(6.dp))
   261→                    Text("Retry")
   262→                }
   263→            }
   264→        }
   265→        return
   266→    }
   267→
   268→    val t = test ?: return
   269→    // Guard: if the test has no questions, show a friendly message instead of crashing
   270→    if (t.items.isEmpty()) {
   271→        Column(
   272→            Modifier.fillMaxSize().padding(20.dp),
   273→            horizontalAlignment = Alignment.CenterHorizontally,
   274→            verticalArrangement = Arrangement.Center
   275→        ) {
   276→            Icon(Icons.Default.Quiz, null, tint = theme.subText, modifier = Modifier.size(48.dp))
   277→            Spacer(Modifier.height(12.dp))
   278→            Text("This test has no questions yet.", color = theme.darkText, fontSize = 14.sp, textAlign = TextAlign.Center)
   279→            Spacer(Modifier.height(16.dp))
   280→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }
   281→        }
   282→        return
   283→    }
   284→
   285→    val currentQuestion = t.items.getOrNull(currentIdx)
   286→
   287→    // ─── Result screen ──────────────────────────────────────────────────────
   288→    if (submitResult != null) {
   289→        ExamResultScreen(theme, submitResult!!, onExit, sound, examTitle = t.title, examDescription = t.description)
   290→        return
   291→    }
   292→
   293→    // ═══ EXAM UI — spec-compliant landscape layout ═══
   294→    val item = t.items.getOrNull(currentIdx) ?: return
   295→    val q = item.question
   296→    val options = q.options ?: emptyList()
   297→    val textItems = t.items.filter { it.question.blockType == "text" }
   298→    val readingCount = if (textItems.isNotEmpty()) textItems.size else t.items.size
   299→    val listeningCount = if (textItems.isNotEmpty()) t.items.size - textItems.size else 0
   300→    val answeredCount = answers.size
   301→    val remainingCount = t.items.size - answeredCount
   302→    // Start with the grid view so the student can pick which question to answer
   303→    // first. They tap a number → answer → return to grid → pick next → submit.
   304→    var showGrid by remember { mutableStateOf(true) }
   305→
   306→    Column(modifier = Modifier.fillMaxSize().background(Color(0xFFF8FAFC))) {
   307→        // ── 1. TOP STATUS HEADER ── compact white bar with thin border
   308→        Surface(
   309→            color = Color.White,
   310→            border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   311→        ) {
   312→            Row(modifier = Modifier.fillMaxWidth().height(36.dp), verticalAlignment = Alignment.CenterVertically) {
   313→                // Section type
   314→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   315→                    val sectionLabel = if (q.blockType == "audio") "Listening ($listeningCount)" else "Reading ($readingCount)"
   316→                    Text(sectionLabel, color = Color(0xFF003478), fontSize = 10.sp, fontWeight = FontWeight.Bold)
   317→                }
   318→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   319→                // Total
   320→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   321→                    Text("Total: ${t.items.size}", color = Color(0xFF64748B), fontSize = 10.sp)
   322→                }
   323→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   324→                // Remaining
   325→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   326→                    Text("Left: $remainingCount", color = Color(0xFF64748B), fontSize = 10.sp)
   327→                }
   328→                Box(modifier = Modifier.width(1.dp).fillMaxHeight(0.6f).background(Color(0xFFE2E8F0)))
   329→                // Timer
   330→                Box(modifier = Modifier.weight(1f).fillMaxHeight(), contentAlignment = Alignment.Center) {
   331→                    val mm = timeLeft / 60; val ss = timeLeft % 60
   332→                    Text(String.format("%02d:%02d", mm, ss), color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   333→                }
   334→            }
   335→        }
   336→
   337→        // ── 2. INSTRUCTION ROW ── compact question number + title + FREE badge
   338→        Surface(modifier = Modifier.fillMaxWidth().padding(horizontal = 6.dp, vertical = 2.dp)) {
   339→            Row(verticalAlignment = Alignment.CenterVertically) {
   340→                Text("${currentIdx + 1}. ", color = Color(0xFF003478), fontSize = 12.sp, fontWeight = FontWeight.Bold)
   341→                val displayText = if (!q.title.isNullOrBlank()) q.title else q.stem.take(80)
   342→                Text(displayText, color = Color(0xFF1E293B), fontSize = 11.sp, fontWeight = FontWeight.SemiBold, maxLines = 1, overflow = TextOverflow.Ellipsis, modifier = Modifier.weight(1f, fill = false))
   343→                if (q.isFree) {
   344→                    Spacer(Modifier.width(4.dp))
   345→                    Surface(color = Color(0xFF22C55E), shape = RoundedCornerShape(3.dp)) {
   346→                        Text("FREE", color = Color.White, fontSize = 8.sp, fontWeight = FontWeight.Bold, modifier = Modifier.padding(horizontal = 4.dp, vertical = 1.dp))
   347→                    }
   348→                }
   349→                Spacer(Modifier.weight(1f))
   350→                IconButton(onClick = { showGrid = true }, modifier = Modifier.size(24.dp)) {
   351→                    Icon(Icons.Default.GridView, null, tint = Color(0xFF64748B), modifier = Modifier.size(14.dp))
   352→                }
   353→            }
   354→        }
   355→        Box(modifier = Modifier.fillMaxWidth().height(2.dp).background(Color.Black))
   356→
   357→        // ── 3. MAIN CONTENT ── 60% question (left, scrollable) | 40% answers (right, scrollable)
   358→        // The DreamKorea logo is rendered as a faded watermark in the background
   359→        // (centered, low alpha) so it blends in nicely without taking screen space.
   360→        Box(modifier = Modifier.weight(1f).fillMaxWidth()) {
   361→            // Watermark logo at the center of the background
   362→            Image(
   363→                painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.dreamkorea_logo),
   364→                contentDescription = null,
   365→                modifier = Modifier.align(Alignment.Center).size(280.dp).alpha(0.08f),
   366→                contentScale = ContentScale.Fit
   367→            )
   368→            Row(modifier = Modifier.fillMaxSize()) {
   369→            // LEFT: Question content (60%) — scrollable so long titles/stems/images don't get cut
   370→            Column(
   371→                modifier = Modifier.weight(0.6f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   372→                verticalArrangement = Arrangement.Center,
   373→                horizontalAlignment = Alignment.CenterHorizontally
   374→            ) {
   375→                // Question title (if set by admin) — prominent header
   376→                if (!q.title.isNullOrBlank()) {
   377→                    Surface(
   378→                        color = Color(0xFF003478),
   379→                        shape = RoundedCornerShape(10.dp),
   380→                        modifier = Modifier.fillMaxWidth(0.92f)
   381→                    ) {
   382→                        Text(
   383→                            q.title,
   384→                            color = Color.White,
   385→                            fontSize = 14.sp,
   386→                            fontWeight = FontWeight.Bold,
   387→                            textAlign = TextAlign.Center,
   388→                            modifier = Modifier.padding(vertical = 8.dp, horizontal = 12.dp),
   389→                            maxLines = 2,
   390→                        )
   391→                    }
   392→                    Spacer(Modifier.height(6.dp))
   393→                }
   394→                // Question card (stem) — white with subtle shadow
   395→                Surface(
   396→                    color = Color.White,
   397→                    shape = RoundedCornerShape(14.dp),
   398→                    shadowElevation = 2.dp,
   399→                    border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFE2E8F0)),
   400→                    modifier = Modifier.fillMaxWidth(0.92f)
   401→                ) {
   402→                    Text(
   403→                        q.stem,
   404→                        color = Color(0xFF1E293B),
   405→                        fontSize = 14.sp,
   406→                        fontWeight = FontWeight.Medium,
   407→                        modifier = Modifier.padding(14.dp)
   408→                    )
   409→                }
   410→                Spacer(Modifier.height(6.dp))
   411→                // Media images — ContentScale.Fit (contain, not cover)
   412→                if (q.descType == "image" && !q.descImageUrl.isNullOrBlank()) {
   413→                    val url = q.descImageUrl!!.toAbsoluteUrl()
   414→                    coil.compose.AsyncImage(
   415→                        model = url, contentDescription = null,
   416→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 120.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(url) },
   417→                        contentScale = ContentScale.Fit
   418→                    )
   419→                }
   420→                val mediaImgUrl = (q.mediaImageUrl ?: q.imageUrl)?.toAbsoluteUrl()
   421→                if (!mediaImgUrl.isNullOrBlank()) {
   422→                    coil.compose.AsyncImage(
   423→                        model = mediaImgUrl, contentDescription = null,
   424→                        modifier = Modifier.fillMaxWidth(0.92f).heightIn(max = 140.dp).clip(RoundedCornerShape(8.dp)).clickable { FullScreenImageViewer.show(mediaImgUrl) },
   425→                        contentScale = ContentScale.Fit
   426→                    )
   427→                }
   428→                val mediaAudUrl = (q.mediaAudioUrl ?: q.audioUrl)?.toAbsoluteUrl()
   429→                if (!mediaAudUrl.isNullOrBlank()) {
   430→                    AudioPlayerCard(theme = theme, url = mediaAudUrl, loopCount = q.audioLoop, loopDelaySec = q.audioLoopDelay, sound = sound)
   431→                }
   432→            }
   433→
   434→            // Vertical divider
   435→            Box(modifier = Modifier.width(2.dp).fillMaxHeight().background(Color.Black))
   436→
   437→            // RIGHT: Answer options (40%) — scrollable so long options don't get cut
   438→            Column(
   439→                modifier = Modifier.weight(0.4f).fillMaxHeight().padding(8.dp).verticalScroll(rememberScrollState()),
   440→                verticalArrangement = Arrangement.Center
   441→            ) {
   442→                when (q.answerType) {
   443→                    "text", "choose" -> {
   444→                        (0 until minOf(4, options.size)).forEach { i ->
   445→                            val isSelected = answers[q.id] == options.getOrNull(i)
   446→                            val optText = options.getOrNull(i) ?: ""
   447→                            val blankWord = q.optionBlanks.getOrNull(i)?.takeIf { it.isNotBlank() }
   448→                            Row(
   449→                                modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp).clickable { sound.click(); answers[q.id] = optText },
   450→                                verticalAlignment = Alignment.CenterVertically
   451→                            ) {
   452→                                Surface(
   453→                                    color = if (isSelected) theme.primary else Color.White,
   454→                                    border = androidx.compose.foundation.BorderStroke(2.dp, Color.Black),
   455→                                    shape = androidx.compose.foundation.shape.CircleShape,
   456→                                    modifier = Modifier.size(40.dp)
   457→                                ) {
   458→                                    Box(contentAlignment = Alignment.Center) { Text("${i+1}", color = if (isSelected) Color.White else Color.Black, fontSize = 15.sp, fontWeight = FontWeight.Bold) }
   459→                                }
   460→                                Spacer(Modifier.width(8.dp))
   461→                                // Render option text with underlined blank word (if set by admin)
   462→                                Text(
   463→                                    text = buildUnderlinedText(optText, blankWord),
   464→                                    color = Color.Black,
   465→                                    fontSize = 13.sp,
   466→                                    modifier = Modifier.weight(1f)
   467→                                )
   468→                            }
   469→                        }
   470→                    }
   471→                    "image" -> {
   472→                        (0 until minOf(4, q.optionImages.size)).forEach { i ->
   473→                            val imgUrl = q.optionImages[i]; if (imgUrl.isBlank()) return@forEach
   474→                            val absUrl = imgUrl.toAbsoluteUrl()
   475→                            val isSelected = answers[q.id] == absUrl
   476→                            Row(
   477→                                modifier = Modifier.fillMaxWidth().padding(vertical = 3.dp).clickable { sound.click(); answers[q.id] = absUrl },

   636→/// 5×4 grid of question buttons. Left-to-right numbering (1-20 or 21-40).
   637→/// Thin borders (#4A4A4A), no rounded corners, white bg, dark text.
   638→/// Answered = dark blue bg + white text. Empty = gray border + gray text.
   639→@Composable
   640→private fun QuestionGridSection(
   641→    test: TestDetail,
   642→    items: List<TestItemDetail>,
   643→    answers: SnapshotStateMap<String, Any>,
   644→    currentIdx: Int,
   645→    sound: SoundManager,
   646→    onPick: (Int) -> Unit,
   647→) {
   648→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   649→    val cols = 5
   650→    // Always show 20 boxes (4 rows × 5 cols)
   651→    val totalBoxes = 20
   652→    val rowsCount = 4
   653→
   654→    Column(modifier = Modifier.fillMaxWidth(), verticalArrangement = Arrangement.spacedBy(5.dp)) {
   655→        for (rowIdx in 0 until rowsCount) {
   656→            Row(
   657→                modifier = Modifier.fillMaxWidth(),
   658→                horizontalArrangement = Arrangement.spacedBy(5.dp)
   659→            ) {
   660→                for (colIdx in 0 until cols) {
   661→                    val localIdx = rowIdx * cols + colIdx
   662→                    if (localIdx < items.size) {
   663→                        val globalIdx = globalIndices[localIdx]
   664→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   665→                        val isCurrent = globalIdx == currentIdx
   666→                        Box(
   667→                            modifier = Modifier
   668→                                .weight(1f)
   669→                                .aspectRatio(1.1f)
   670→                                .border(1.5.dp, if (isCurrent) Color(0xFF003F73) else Color(0xFF4A4A4A))
   671→                                .background(if (isAnswered) Color(0xFF003F73) else Color.White)
   672→                                .clickable { sound.click(); onPick(globalIdx) },
   673→                            contentAlignment = Alignment.Center
   674→                        ) {
   675→                            Text(
   676→                                "${globalIdx + 1}",
   677→                                color = if (isAnswered) Color.White else Color(0xFF111111),
   678→                                fontSize = 14.sp,
   679→                                fontWeight = FontWeight.Bold,
   680→                            )
   681→                        }
   682→                    } else {
   683→                        val baseNum = if (globalIndices.isNotEmpty()) globalIndices[0] else 0
   684→                        val displayNum = baseNum + localIdx + 1
   685→                        // Empty box — show number, not clickable, gray
   686→                        Box(
   687→                            modifier = Modifier
   688→                                .weight(1f)
   689→                                .aspectRatio(1.1f)
   690→                                .border(1.5.dp, Color(0xFFCCCCCC)),
   691→                            contentAlignment = Alignment.Center
   692→                        ) {
   693→                            Text(
   694→                                "$displayNum",
   695→                                color = Color(0xFFCCCCCC),
   696→                                fontSize = 14.sp,
   697→                                fontWeight = FontWeight.Bold,
   698→                            )
   699→                        }
   700→                    }
   701→                }
   702→            }
   703→        }
   704→    }

   738→// ─── Audio player with loop support ───────────────────────────────────────────
   739→// loopCount = total number of times to play the audio:
   740→//   0 or 1 = plays once
   741→//   2 = plays twice
   742→//   N = plays N times
   743→//  -1 = infinite loop
   744→@Composable
   745→fun AudioPlayerCard(theme: AppTheme, url: String, loopCount: Int, loopDelaySec: Int, sound: SoundManager) {
   746→    val context = LocalContext.current
   747→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
   748→    var isPlaying by remember { mutableStateOf(false) }
   749→    var playCount by remember { mutableStateOf(0) }
   750→    // disabled = all plays used up, can't replay
   751→    var disabled by remember { mutableStateOf(false) }
   752→    val maxPlays = loopCount.coerceAtLeast(1)
   753→    val scope = rememberCoroutineScope()
   754→
   755→    DisposableEffect(url) {
   756→        onDispose {
   757→            mediaPlayer?.release()
   758→            mediaPlayer = null
   759→        }
   760→    }
   761→
   762→    Surface(color = theme.cardBg, shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth(), shadowElevation = 1.dp) {
   763→        Row(
   764→            modifier = Modifier.padding(14.dp),
   765→            verticalAlignment = Alignment.CenterVertically
   766→        ) {
   767→            // Audio icon
   768→            Surface(
   769→                color = if (disabled) Color(0xFFE2E8F0) else theme.primary.copy(alpha = 0.15f),
   770→                shape = RoundedCornerShape(10.dp),
   771→                modifier = Modifier.size(44.dp)
   772→            ) {
   773→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
   774→                    Icon(
   775→                        Icons.Default.Headphones,
   776→                        null,
   777→                        tint = if (disabled) Color(0xFF94A3B8) else theme.primary,
   778→                        modifier = Modifier.size(22.dp)
   779→                    )
   780→                }
   781→            }
   782→            Spacer(Modifier.width(12.dp))
   783→            // Play count info
   784→            Column(modifier = Modifier.weight(1f)) {
   785→                Text(
   786→                    if (disabled) "Audio played ($maxPlays/${maxPlays})" else "Audio — Play $playCount / $maxPlays",
   787→                    color = if (disabled) Color(0xFF94A3B8) else theme.darkText,
   788→                    fontSize = 13.sp,
   789→                    fontWeight = FontWeight.SemiBold
   790→                )
   791→                if (!disabled) {
   792→                    Text(
   793→                        if (maxPlays == 1) "Tap to listen (1 time only)" else "Tap to listen ($maxPlays times)",
   794→                        color = theme.subText, fontSize = 11.sp
   795→                    )
   796→                } else {
   797→                    Text("No more plays left", color = Color(0xFF94A3B8), fontSize = 11.sp)
   798→                }
   799→            }
   800→            // Play/Pause button — DISABLED after all plays used
   801→            IconButton(
   802→                onClick = {
   803→                    if (disabled) return@IconButton
   804→                    sound.click()
   805→                    if (isPlaying) {
   806→                        // Don't pause — let it finish. Only allow play, not pause.
   807→                        return@IconButton
   808→                    } else {
   809→                        try {
   810→                            mediaPlayer?.release()
   811→                            val mp = android.media.MediaPlayer().apply {
   812→                                setDataSource(url)
   813→                                setOnPreparedListener {
   814→                                    start()
   815→                                    isPlaying = true
   816→                                    playCount = 1
   817→                                }
   818→                                setOnCompletionListener {
   819→                                    if (playCount < maxPlays) {
   820→                                        scope.launch {
   821→                                            if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
   822→                                            if (playCount < maxPlays) {
   823→                                                playCount++
   824→                                                start()
   825→                                            } else {
   826→                                                isPlaying = false
   827→                                                disabled = true
   828→                                            }
   829→                                        }
   830→                                    } else {
   831→                                        isPlaying = false
   832→                                        disabled = true
   833→                                    }
   834→                                }
   835→                                setOnErrorListener { _, _, _ ->
   836→                                    isPlaying = false
   837→                                    true
   838→                                }
   839→                                prepareAsync()
   840→                            }
   841→                            mediaPlayer = mp
   842→                        } catch (_: Exception) {
   843→                            isPlaying = false
   844→                        }
   845→                    }
   846→                },
   847→                enabled = !disabled && !isPlaying
   848→            ) {
   849→                Icon(
   850→                    when {
   851→                        disabled -> Icons.Default.Lock
   852→                        isPlaying -> Icons.Default.VolumeUp
   853→                        else -> Icons.Default.PlayArrow
   854→                    },
   855→                    null,
   856→                    tint = if (disabled) Color(0xFFCBD5E1) else theme.primary,
   857→                    modifier = Modifier.size(32.dp)
   858→                )
   859→            }
   860→        }
   861→    }
   862→}

   865→@Composable
   866→fun AnswerInput(
   867→    theme: AppTheme,

   928→                    val selected = selectedList.contains(opt)

   936→                        }
   937→                    ) {
   938→                        Row(modifier = Modifier.padding(14.dp), verticalAlignment = Alignment.CenterVertically) {
   939→                            Surface(
   940→                                color = if (selected) theme.primary else Color.Transparent,
   941→                                shape = RoundedCornerShape(4.dp),

  1628→/**
  1629→ * Builds an AnnotatedString where the [blankWord] is underlined within [text].
  1630→ * If blankWord is null/empty, returns the plain text.
  1631→ * The underlined word is case-insensitive matched.
  1632→ */
  1633→fun buildUnderlinedText(text: String, blankWord: String?): androidx.compose.ui.text.AnnotatedString {
  1634→    if (blankWord.isNullOrBlank()) return androidx.compose.ui.text.AnnotatedString(text)
  1635→    val idx = text.indexOf(blankWord, ignoreCase = true)
  1636→    if (idx < 0) return androidx.compose.ui.text.AnnotatedString(text)
  1637→    return androidx.compose.ui.text.buildAnnotatedString {
  1638→        append(text.substring(0, idx))
  1639→        withStyle(androidx.compose.ui.text.SpanStyle(textDecoration = androidx.compose.ui.text.style.TextDecoration.Underline)) {
  1640→            append(text.substring(idx, idx + blankWord.length))
  1641→        }
  1642→        append(text.substring(idx + blankWord.length))
  1643→    }
  1644→}

  1647→// CLIENT-SIDE SUBMIT FALLBACK — when the server's submit endpoint doesn't
  1648→// know how to handle combined exam IDs (qbank-combined / bundle-{id}), we
  1649→// grade the exam locally using the question data we already have loaded.

  1652→/**
  1653→ * Grades a combined exam client-side. Used as a fallback when
  1654→ * /api/student/tests/[testId]/submit returns 404 for combined exam IDs.
  1655→ *
  1656→ * Grades SINGLE_CHOICE, TRUE_FALSE, ONE_WORD, FILL_BLANK, MULTIPLE_CHOICE.
  1657→ * Subjective types (SHORT_ANSWER, LONG_ANSWER, MATCHING) are marked as
  1658→ * incorrect (the student would need a teacher to review them).
  1659→ */
  1660→private fun gradeCombinedExamClientSide(
  1661→    test: TestDetail,
  1662→    answers: Map<String, Any>,
  1663→): SubmitResponse {
  1664→    var score = 0
  1665→    val review = mutableListOf<ReviewItem>()
  1666→
  1667→    // Normalize URL — strip origin so /api/files/... matches https://.../api/files/...
  1668→    fun normalizeUrl(val_: Any?): Any? {
  1669→        if (val_ is String) {
  1670→            return try {
  1671→                if (val_.startsWith("http://") || val_.startsWith("https://")) {
  1672→                    java.net.URI(val_).path
  1673→                } else val_
  1674→            } catch (_: Exception) { val_ }
  1675→        }
  1676→        return val_
  1677→    }
  1678→
  1679→    for (item in test.items) {
  1680→        val q = item.question
  1681→        val ans = normalizeUrl(answers[q.id])
  1682→        var isCorrect = false
  1683→
  1684→        when (q.type) {
  1685→            "SINGLE_CHOICE", "TRUE_FALSE", "ONE_WORD", "FILL_BLANK" -> {
  1686→                val correctIdx = q.correctOption
  1687→                val correctAns = normalizeUrl(q.options?.getOrNull(correctIdx)
  1688→                    ?: q.options?.getOrNull(0)
  1689→                    ?: "") as? String ?: ""
  1690→                if (ans is String && correctAns.isNotEmpty() &&
  1691→                    ans.trim().equals(correctAns.trim(), ignoreCase = true)) {
  1692→                    isCorrect = true
  1693→                    score++
  1694→                }
  1695→            }
  1696→            "MULTIPLE_CHOICE" -> {
  1697→                val correctIdx = q.correctOption
  1698→                if (ans is String && ans.toIntOrNull() == correctIdx) {
  1699→                    isCorrect = true
  1700→                    score++
  1701→                }
  1702→            }
  1703→        }
  1704→
  1705→        review.add(
  1706→            ReviewItem(
  1707→                questionId = q.id,
  1708→                stem = q.stem,
  1709→                title = q.title,
  1710→                type = q.type,
  1711→                options = q.options,
  1712→                optionImages = q.optionImages,
  1713→                optionAudios = q.optionAudios,
  1714→                imageUrl = q.imageUrl,
  1715→                audioUrl = q.audioUrl,
  1716→                audioLoop = q.audioLoop,
  1717→                audioLoopDelay = q.audioLoopDelay,
  1718→                userAnswer = ans,
  1719→                correctAnswer = q.options?.getOrNull(q.correctOption),
  1720→                explanation = q.explanation,
  1721→                isCorrect = isCorrect,
  1722→            )
  1723→        )
  1724→    }
  1725→
  1726→    val maxScore = test.items.size
  1727→    val pct = if (maxScore > 0) (score.toDouble() / maxScore) * 100 else 0.0
  1728→
  1729→    // Eye vision recommendation — same logic as server
  1730→    val eyeVision = if (maxScore > 0) {
  1731→        val mistakesPct = 100 - pct
  1732→        when {
  1733→            mistakesPct > 30 -> {
  1734→                val count = if (mistakesPct >= 70) 5 else if (mistakesPct >= 50) 4 else 3
  1735→                EyeVisionRecommendation(
  1736→                    show = true,
  1737→                    count = count,
  1738→                    reason = "You made ${Math.round(mistakesPct)}% mistakes. Let's check your eye vision with $count quick tests.",
  1739→                )
  1740→            }
  1741→            mistakesPct >= 15 -> {
  1742→                EyeVisionRecommendation(
  1743→                    show = true,
  1744→                    count = 2,
  1745→                    reason = "You made ${Math.round(mistakesPct)}% mistakes. A quick eye vision check is recommended.",
  1746→                )
  1747→            }
  1748→            else -> EyeVisionRecommendation()
  1749→        }
  1750→    } else EyeVisionRecommendation()
  1751→
  1752→    return SubmitResponse(
  1753→        score = score,
  1754→        maxScore = maxScore,
  1755→        graded = true,
  1756→        submissionId = "client-graded-${System.currentTimeMillis()}",
  1757→        review = review,
  1758→        eyeVision = eyeVision,
  1759→        completed = true,
  1760→    )
  1761→}
  1762→
  1763→/**
  1764→ * Submits a combined exam with fallback: tries the server first, and if the
  1765→ * server returns 404 (endpoint not deployed for combined IDs), grades the
  1766→ * exam client-side instead.
  1767→ */
  1768→private suspend fun submitCombinedExamWithFallback(
  1769→    test: TestDetail,
  1770→    answers: Map<String, Any>,
  1771→): SubmitResponse {
  1772→    return try {
  1773→        AppState.api.submitTest(test.id, SubmitRequest(answers))
  1774→    } catch (e: retrofit2.HttpException) {

  1778→        // Network errors, timeouts, etc. → also fall back to client-side grading
  1779→        // so the student always gets a result.
  1780→        gradeCombinedExamClientSide(test, answers)
  1781→    }
  1782→}

  1789→private suspend fun submitExamWithFallback(
