    33→/**
    34→ * Login Screen — three tabs: Sign In / Sign Up / Forgot Password.
    35→ *
    36→ * Sign In: email + password → POST /api/auth/credentials
    37→ * Sign Up: name + email + phone + password → POST /api/auth/signup (mode=student)
    38→ *          — no OTP needed, password is set immediately
    39→ * Forgot:  email → POST /api/auth/request-reset → enter 6-digit code →
    40→ *          new password → POST /api/auth/reset-password
    41→ *
    42→ * Default tab: Sign In. The DK logo animates in on mount. Background is a
    43→ * navy gradient (matching the brand). Form card sits on a white surface with
    44→ * a strong shadow for separation.
    45→ */
    46→@Composable
    47→fun LoginScreen(onLoginSuccess: () -> Unit) {
    48→    val scope = rememberCoroutineScope()
    49→    val sound = rememberSoundManager()
    50→
    51→    // Tab: "login" | "signup" | "forgot"
    52→    var mode by remember { mutableStateOf("login") }
    53→
    54→    // Sign Up state
    55→    var suName by remember { mutableStateOf("") }
    56→    var suEmail by remember { mutableStateOf("") }
    57→    var suPhone by remember { mutableStateOf("") }
    58→    var suPassword by remember { mutableStateOf("") }
    59→
    60→    // Login state
    61→    var liEmail by remember { mutableStateOf("") }
    62→    var liPassword by remember { mutableStateOf("") }
    63→
    64→    // Forgot password state
    65→    var fpEmail by remember { mutableStateOf("") }
    66→    var fpCode by remember { mutableStateOf("") }
    67→    var fpNewPassword by remember { mutableStateOf("") }
    68→    var fpStep by remember { mutableStateOf(1) } // 1=enter email, 2=enter code+new password
    69→
    70→    var loading by remember { mutableStateOf(false) }
    71→    var error by remember { mutableStateOf("") }
    72→    var info by remember { mutableStateOf("") }
    73→    var passwordVisible by remember { mutableStateOf(false) }
    74→
    75→    // Animations
    76→    var logoVisible by remember { mutableStateOf(false) }
    77→    var formVisible by remember { mutableStateOf(false) }
    78→    LaunchedEffect(Unit) {
    79→        logoVisible = true
    80→        kotlinx.coroutines.delay(300)
    81→        formVisible = true
    82→    }
    83→    val logoScale by animateFloatAsState(
    84→        targetValue = if (logoVisible) 1f else 0.3f,
    85→        animationSpec = spring(dampingRatio = Spring.DampingRatioMediumBouncy, stiffness = Spring.StiffnessLow),
    86→        label = "logoScale"
    87→    )
    88→    val formAlpha by animateFloatAsState(
    89→        targetValue = if (formVisible) 1f else 0f,
    90→        animationSpec = tween(600, easing = FastOutSlowInEasing),
    91→        label = "formAlpha"
    92→    )
    93→
    94→    Box(modifier = Modifier.fillMaxSize().background(
    95→        Brush.verticalGradient(listOf(Color(0xFFF8FAFC), Color(0xFFEFF6FF), NavyBlue.copy(alpha = 0.3f), NavyBlue))
    96→    )) {
    97→        Column(
    98→            modifier = Modifier.fillMaxSize().padding(horizontal = 24.dp).verticalScroll(rememberScrollState()),
    99→            horizontalAlignment = Alignment.CenterHorizontally,
   100→            verticalArrangement = Arrangement.Center
   101→        ) {
   102→            Spacer(modifier = Modifier.height(40.dp))
   103→
   104→            // Logo
   105→            Surface(color = Color.White, shape = androidx.compose.foundation.shape.CircleShape,
   106→                modifier = Modifier.size(100.dp).scale(logoScale), shadowElevation = 12.dp) {
   107→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
   108→                    Image(painter = painterResource(id = app.dreamkorea.smartclass.R.drawable.dreamkorea_logo),
   109→                        contentDescription = "Logo", modifier = Modifier.size(64.dp), contentScale = ContentScale.Fit)
   110→                }
   111→            }
   112→
   113→            Spacer(modifier = Modifier.height(20.dp))
   114→            Text("DreamKorea SmartClass", color = Color.White, fontSize = 18.sp, fontWeight = FontWeight.Bold)
   115→            Text("Learn Korean anywhere", color = Color.White.copy(alpha = 0.8f), fontSize = 12.sp)
   116→            Spacer(modifier = Modifier.height(24.dp))
   117→
   118→            // Three-tab toggle: Sign In / Sign Up / Forgot
   119→            AnimatedVisibility(visible = logoVisible, enter = fadeIn(tween(500))) {
   120→                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.Center) {
   121→                    listOf("login" to "Sign In", "signup" to "Sign Up", "forgot" to "Forgot").forEach { (key, label) ->
   122→                        if (key != "login" && key != "signup") Spacer(Modifier.width(8.dp))
   123→                        if (key == "signup") Spacer(Modifier.width(8.dp))
   124→                        FilterChip(
   125→                            selected = mode == key,
   126→                            onClick = { sound.click(); mode = key; error = ""; info = ""; fpStep = 1 },
   127→                            label = { Text(label, fontSize = 12.sp) },
   128→                            colors = FilterChipDefaults.filterChipColors(
   129→                                selectedContainerColor = NavyBlue,
   130→                                selectedLabelColor = Color.White
   131→                            )
   132→                        )
   133→                    }
   134→                }
   135→            }
   136→
   137→            Spacer(modifier = Modifier.height(20.dp))
   138→
   139→            // Form card
   140→            Surface(color = Color.White, shape = RoundedCornerShape(20.dp),
   141→                modifier = Modifier.fillMaxWidth().alpha(formAlpha), shadowElevation = 8.dp) {
   142→                Column(modifier = Modifier.padding(24.dp)) {
   143→                    // Info banner
   144→                    if (info.isNotEmpty()) {
   145→                        Surface(color = Color(0xFFE8F5E9), shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth().padding(bottom = 8.dp)) {
   146→                            Row(modifier = Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
   147→                                Icon(Icons.Default.CheckCircle, null, tint = SuccessGreen, modifier = Modifier.size(18.dp))
   148→                                Spacer(Modifier.width(8.dp))
   149→                                Text(info, color = SuccessGreen, fontSize = 13.sp, modifier = Modifier.weight(1f))
   150→                            }
   151→                        }
   152→                    }
   153→                    // Error banner
   154→                    if (error.isNotEmpty()) {
   155→                        Surface(color = Color(0xFFFEE2E2), shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth().padding(bottom = 8.dp)) {
   156→                            Row(modifier = Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
   157→                                Icon(Icons.Default.Error, null, tint = Color(0xFFEF4444), modifier = Modifier.size(18.dp))
   158→                                Spacer(Modifier.width(8.dp))
   159→                                Text(error, color = Color(0xFFEF4444), fontSize = 13.sp, modifier = Modifier.weight(1f))
   160→                            }
   161→                        }
   162→                    }
   163→
   164→                    AnimatedContent(targetState = mode, transitionSpec = {
   165→                        (fadeIn(tween(300)) + slideInHorizontally(tween(300)) { it / 5 }) togetherWith
   166→                        (fadeOut(tween(200)) + slideOutHorizontally(tween(200)) { -it / 5 })
   167→                    }, label = "modeTransition") { currentMode ->
   168→                        when (currentMode) {
   169→                            "login" -> LoginTab(
   170→                                email = liEmail, password = liPassword, passwordVisible = passwordVisible, loading = loading,
   171→                                onEmailChange = { liEmail = it }, onPasswordChange = { liPassword = it },
   172→                                onTogglePassword = { passwordVisible = !passwordVisible },
   173→                                onSubmit = {
   174→                                    if (liEmail.isBlank() || liPassword.isBlank()) { error = "Enter email and password"; return@LoginTab }
   175→                                    loading = true; error = ""; info = ""
   176→                                    scope.launch {
   177→                                        try {
   178→                                            val resp = AppState.api.loginCredentials(mapOf("username" to liEmail.trim(), "password" to liPassword))
   179→                                            if (resp.ok) {
   180→                                                sound.success()
   181→                                                AppState.saveUserProfile(resp.user)
   182→                                                AppState.invalidateCache()
   183→                                                onLoginSuccess()
   184→                                            } else {
   185→                                                sound.error()
   186→                                                error = resp.error ?: "Wrong email or password."
   187→                                            }
   188→                                        } catch (e: retrofit2.HttpException) {
   189→                                            sound.error()
   190→                                            error = extractHttpError(e) ?: "Wrong email or password."
   191→                                        } catch (e: java.net.UnknownHostException) { sound.error(); error = "No internet connection." }
   192→                                        catch (e: java.io.IOException) { sound.error(); error = "Could not connect." }
   193→                                        catch (e: Exception) { sound.error(); error = "Login failed: ${e.message ?: "unknown"}" }
   194→                                        loading = false
   195→                                    }
   196→                                }
   197→                            )
   198→                            "signup" -> SignupTab(
   199→                                name = suName, email = suEmail, phone = suPhone, password = suPassword,
   200→                                passwordVisible = passwordVisible, loading = loading,
   201→                                onNameChange = { suName = it }, onEmailChange = { suEmail = it },
   202→                                onPhoneChange = { suPhone = it }, onPasswordChange = { suPassword = it },
   203→                                onTogglePassword = { passwordVisible = !passwordVisible },
   204→                                onSubmit = {
   205→                                    if (suName.isBlank()) { error = "Name is required"; return@SignupTab }
   206→                                    if (suEmail.isBlank() || !suEmail.contains("@")) { error = "Enter a valid email"; return@SignupTab }
   207→                                    if (suPassword.length < 6) { error = "Password must be at least 6 characters"; return@SignupTab }
   208→                                    loading = true; error = ""; info = ""
   209→                                    scope.launch {
   210→                                        try {
   211→                                            val resp = AppState.api.signup(mapOf(
   212→                                                "mode" to "student",
   213→                                                "name" to suName.trim(),
   214→                                                "email" to suEmail.trim().lowercase(),
   215→                                                "phone" to suPhone.trim(),
   216→                                                "password" to suPassword
   217→                                            ))
   218→                                            if (resp.ok) {
   219→                                                sound.success()
   220→                                                AppState.saveUserProfile(resp.user)
   221→                                                AppState.invalidateCache()
   222→                                                onLoginSuccess()
   223→                                            } else {
   224→                                                sound.error()
   225→                                                error = resp.error ?: "Signup failed."
   226→                                            }
   227→                                        } catch (e: retrofit2.HttpException) {
   228→                                            sound.error()
   229→                                            error = extractHttpError(e) ?: "Signup failed."
   230→                                        } catch (e: java.net.UnknownHostException) { sound.error(); error = "No internet connection." }
   231→                                        catch (e: java.io.IOException) { sound.error(); error = "Could not connect." }
   232→                                        catch (e: Exception) { sound.error(); error = "Signup failed: ${e.message ?: "unknown"}" }
   233→                                        loading = false
   234→                                    }
   235→                                }
   236→                            )
   237→                            "forgot" -> ForgotTab(
   238→                                email = fpEmail, code = fpCode, newPassword = fpNewPassword,
   239→                                passwordVisible = passwordVisible, loading = loading, step = fpStep,
   240→                                onEmailChange = { fpEmail = it }, onCodeChange = { fpCode = it },
   241→                                onNewPasswordChange = { fpNewPassword = it },
   242→                                onTogglePassword = { passwordVisible = !passwordVisible },
   243→                                onRequestCode = {
   244→                                    if (fpEmail.isBlank() || !fpEmail.contains("@")) { error = "Enter a valid email"; return@ForgotTab }
   245→                                    loading = true; error = ""; info = ""
   246→                                    scope.launch {
   247→                                        try {
   248→                                            AppState.api.requestReset(mapOf("email" to fpEmail.trim().lowercase()))
   249→                                            sound.success()
   250→                                            info = "If an account exists, a reset code was sent to ${fpEmail.trim()}."
   251→                                            fpStep = 2
   252→                                        } catch (e: java.net.UnknownHostException) { sound.error(); error = "No internet connection." }
   253→                                        catch (e: java.io.IOException) { sound.error(); error = "Could not connect." }
   254→                                        catch (e: Exception) { sound.error(); error = "Request failed." }
   255→                                        loading = false
   256→                                    }
   257→                                },
   258→                                onReset = {
   259→                                    if (fpCode.length != 6) { error = "Enter the 6-digit code"; return@ForgotTab }
   260→                                    if (fpNewPassword.length < 6) { error = "New password must be at least 6 characters"; return@ForgotTab }
   261→                                    loading = true; error = ""; info = ""
   262→                                    scope.launch {
   263→                                        try {
   264→                                            val resp = AppState.api.resetPassword(mapOf(
   265→                                                "email" to fpEmail.trim().lowercase(),
   266→                                                "code" to fpCode.trim(),
   267→                                                "newPassword" to fpNewPassword
   268→                                            ))
   269→                                            if (resp.ok) {
   270→                                                sound.success()
   271→                                                info = "Password reset! You can now sign in with your new password."
   272→                                                mode = "login"
   273→                                                liEmail = fpEmail
   274→                                                fpStep = 1
   275→                                                fpCode = ""; fpNewPassword = ""
   276→                                            } else {
   277→                                                sound.error()
   278→                                                error = resp.error ?: "Reset failed."
   279→                                            }
   280→                                        } catch (e: retrofit2.HttpException) {
   281→                                            sound.error()
   282→                                            error = extractHttpError(e) ?: "Reset failed."
   283→                                        } catch (e: java.net.UnknownHostException) { sound.error(); error = "No internet connection." }
   284→                                        catch (e: java.io.IOException) { sound.error(); error = "Could not connect." }
   285→                                        catch (e: Exception) { sound.error(); error = "Reset failed: ${e.message ?: "unknown"}" }
   286→                                        loading = false
   287→                                    }
   288→                                }
   289→                            )
   290→                        }
   291→                    }
   292→                }
   293→            }
   294→
   295→            Spacer(modifier = Modifier.height(40.dp))
   296→        }
   297→    }
   298→}

   368→// ─── Forgot Password tab ─────────────────────────────────────────────────
   369→@Composable
   370→private fun ForgotTab(
   371→    email: String, code: String, newPassword: String,
   372→    passwordVisible: Boolean, loading: Boolean, step: Int,
   373→    onEmailChange: (String) -> Unit, onCodeChange: (String) -> Unit,
   374→    onNewPasswordChange: (String) -> Unit, onTogglePassword: () -> Unit,
   375→    onRequestCode: () -> Unit, onReset: () -> Unit
   376→) {
   377→    Column(modifier = Modifier.fillMaxWidth()) {
   378→        Text(if (step == 1) "Reset Password" else "Enter Code & New Password", color = TextDark, fontSize = 20.sp, fontWeight = FontWeight.Bold)
   379→        Text(if (step == 1) "We'll send a 6-digit code to your email" else "Check your email for the 6-digit code", color = TextMid, fontSize = 13.sp)
   380→        Spacer(Modifier.height(16.dp))
   381→
   382→        if (step == 1) {
   383→            Field("Email", email, onEmailChange, Icons.Default.Email, KeyboardType.Email)
   384→            Spacer(Modifier.height(16.dp))
   385→            Button(
   386→                onClick = onRequestCode,
   387→                modifier = Modifier.fillMaxWidth().height(50.dp),
   388→                colors = ButtonDefaults.buttonColors(containerColor = NavyBlue),
   389→                shape = RoundedCornerShape(12.dp),
   390→                enabled = !loading
   391→            ) {
   392→                if (loading) CircularProgressIndicator(color = Color.White, modifier = Modifier.size(20.dp), strokeWidth = 2.dp)
   393→                else Text("Send Reset Code", color = Color.White, fontSize = 15.sp, fontWeight = FontWeight.SemiBold)
   394→            }
   395→        } else {
   396→            // Read-only email display
   397→            Surface(color = Color(0xFFF1F5F9), shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth().padding(bottom = 10.dp)) {
   398→                Row(modifier = Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
   399→                    Icon(Icons.Default.Email, null, tint = TextMid, modifier = Modifier.size(18.dp))
   400→                    Spacer(Modifier.width(8.dp))
   401→                    Text(email, color = TextDark, fontSize = 13.sp, fontWeight = FontWeight.Medium)
   402→                }
   403→            }
   404→            // Code field
   405→            OutlinedTextField(
   406→                value = code, onValueChange = { onCodeChange(it.filter { c -> c.isDigit() }.take(6)) },
   407→                label = { Text("6-digit code") },
   408→                modifier = Modifier.fillMaxWidth(),
   409→                singleLine = true,
   410→                colors = OutlinedTextFieldDefaults.colors(
   411→                    focusedTextColor = TextDark, unfocusedTextColor = TextDark,
   412→                    focusedBorderColor = NavyBlue, unfocusedBorderColor = DividerColor,
   413→                    cursorColor = NavyBlue, focusedLabelColor = NavyBlue, unfocusedLabelColor = TextMid
   414→                ),
   415→                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
   416→                shape = RoundedCornerShape(12.dp),
   417→                leadingIcon = { Icon(Icons.Default.Password, null, tint = TextMid, modifier = Modifier.size(20.dp)) }
   418→            )
   419→            Spacer(Modifier.height(10.dp))
   420→            PasswordField("New Password (min 6 chars)", newPassword, onNewPasswordChange, passwordVisible, onTogglePassword)
   421→            Spacer(Modifier.height(16.dp))
   422→            Button(
   423→                onClick = onReset,
   424→                modifier = Modifier.fillMaxWidth().height(50.dp),
   425→                colors = ButtonDefaults.buttonColors(containerColor = NavyBlue),
   426→                shape = RoundedCornerShape(12.dp),
   427→                enabled = !loading
   428→            ) {
   429→                if (loading) CircularProgressIndicator(color = Color.White, modifier = Modifier.size(20.dp), strokeWidth = 2.dp)
   430→                else Text("Reset Password", color = Color.White, fontSize = 15.sp, fontWeight = FontWeight.SemiBold)
   431→            }
   432→        }
   433→    }
   434→}

   436→// ─── Reusable field helpers ──────────────────────────────────────────────
   437→@Composable
   438→private fun Field(label: String, value: String, onChange: (String) -> Unit, icon: androidx.compose.ui.graphics.vector.ImageVector, keyboardType: KeyboardType, placeholder: String = "") {
   439→    OutlinedTextField(
   440→        value = value, onValueChange = onChange, label = { Text(label) },
   441→        modifier = Modifier.fillMaxWidth(), singleLine = true,
   442→        colors = OutlinedTextFieldDefaults.colors(
   443→            focusedTextColor = TextDark, unfocusedTextColor = TextDark,
   444→            focusedBorderColor = NavyBlue, unfocusedBorderColor = DividerColor,
   445→            cursorColor = NavyBlue, focusedLabelColor = NavyBlue, unfocusedLabelColor = TextMid
   446→        ),
   447→        keyboardOptions = KeyboardOptions(keyboardType = keyboardType),
   448→        shape = RoundedCornerShape(12.dp),
   449→        leadingIcon = { Icon(icon, null, tint = TextMid, modifier = Modifier.size(20.dp)) },
   450→        placeholder = if (placeholder.isNotEmpty()) { { Text(placeholder, color = TextLight, fontSize = 13.sp) } } else null
   451→    )
   452→}

   454→@Composable
   455→private fun PasswordField(label: String, value: String, onChange: (String) -> Unit, visible: Boolean, onToggle: () -> Unit) {
   456→    OutlinedTextField(
   457→        value = value, onValueChange = onChange, label = { Text(label) },
   458→        modifier = Modifier.fillMaxWidth(), singleLine = true,

   476→/**
   477→ * Extract the human-readable error message from a Retrofit HttpException.
   478→ *
   479→ * The server returns errors as JSON: {"error": "Email already registered"}
   480→ * We parse the response body to pull out the "error" field. If parsing fails
   481→ * or the body is empty, we return null so the caller can use a fallback.
   482→ */
   483→private fun extractHttpError(e: retrofit2.HttpException): String? {
   484→    return try {
   485→        val raw = e.response()?.errorBody()?.string()
   486→        if (raw.isNullOrBlank()) return null
   487→        val json = com.google.gson.JsonParser.parseString(raw).asJsonObject
   488→        json.get("error")?.asString
   489→    } catch (_: Exception) {
   490→        null
   491→    }
   492→}
