   100→  function load() {

   138→        <div className="grid gap-3">

   140→            <Card key={t.id} className="hover:shadow-md transition-shadow cursor-pointer" onClick={() => setEditingTest(t)}>

   145→                  <div className="w-16 h-16 rounded-lg bg-slate-100 flex items-center justify-center">

   148→                )}
   149→                <div className="flex-1 min-w-0">
   150→                  <div className="flex items-center gap-2 flex-wrap">

   153→                    {t.category && <Badge variant="secondary">{t.category}</Badge>}
   154→                    {t.price ? <Badge>₩{t.price}</Badge> : <Badge variant="outline">Free</Badge>}

   156→                  <p className="text-sm text-muted-foreground mt-1">
   157→                    {t.durationMin} min • {t._count?.items || 0} questions • Pass {t.passScore}%

   162→                    <Badge className="bg-green-500">🚀 Live</Badge>

   197→// ─── Featured Image Upload (drag & drop + instant preview) ──────────────────
   198→function FeaturedImageUpload({ url, onUpload, onClear, uploadFn }: {
   199→  url: string;
   200→  onUpload: (url: string) => void;
   201→  onClear: () => void;
   202→  uploadFn: (file: File) => Promise<string>;
   203→}) {
   204→  const [uploading, setUploading] = useState(false);
   205→  const [dragOver, setDragOver] = useState(false);
   206→  const [localPreview, setLocalPreview] = useState("");
   207→
   208→  async function compressImage(file: File): Promise<File> {
   209→    if (file.size < 200_000) return file;
   210→    return new Promise((resolve) => {
   211→      const reader = new FileReader();
   212→      reader.onload = (e) => {
   213→        const img = new Image();
   214→        img.onload = () => {
   215→          const canvas = document.createElement("canvas");
   216→          const maxDim = 800;
   217→          let { width, height } = img;
   218→          if (width > maxDim || height > maxDim) {
   219→            if (width > height) { height = (height / width) * maxDim; width = maxDim; }
   220→            else { width = (width / height) * maxDim; height = maxDim; }
   221→          }
   222→          canvas.width = width; canvas.height = height;
   223→          const ctx = canvas.getContext("2d")!;
   224→          ctx.drawImage(img, 0, 0, width, height);
   225→          canvas.toBlob((blob) => {
   226→            if (blob) resolve(new File([blob], "featured.jpg", { type: "image/jpeg" }));
   227→            else resolve(file);
   228→          }, "image/jpeg", 0.8);
   229→        };
   230→        img.src = e.target?.result as string;
   231→      };
   232→      reader.readAsDataURL(file);
   233→    });
   234→  }
   235→
   236→  async function handleFile(file: File) {
   237→    const localUrl = URL.createObjectURL(file);
   238→    setLocalPreview(localUrl);
   239→    setUploading(true);
   240→    try {
   241→      const compressed = await compressImage(file);
   242→      const u = await uploadFn(compressed);
   243→      onUpload(u);
   244→      toast.success("Image uploaded");
   245→    } catch (err: any) {
   246→      toast.error(err.message || "Upload failed");
   247→      setLocalPreview("");
   248→    } finally {
   249→      setUploading(false);
   250→      setTimeout(() => URL.revokeObjectURL(localUrl), 1000);
   251→    }
   252→  }
   253→
   254→  const displayUrl = uploading && localPreview ? localPreview : url;
   255→
   256→  return (
   257→    <div className="mt-1">
   258→      {displayUrl ? (
   259→        <div className="relative inline-block">
   260→          <img src={displayUrl} alt="Featured" className="w-24 h-24 rounded-lg object-cover border-2 border-slate-200" />
   261→          {uploading && (
   262→            <div className="absolute inset-0 bg-black/40 rounded-lg flex items-center justify-center">
   263→              <div className="w-6 h-6 border-2 border-white border-t-transparent rounded-full animate-spin" />
   264→            </div>
   265→          )}
   266→          {!uploading && (
   267→            <button
   268→              onClick={() => { onClear(); setLocalPreview(""); }}
   269→              className="absolute -top-2 -right-2 w-7 h-7 bg-red-500 text-white rounded-full flex items-center justify-center text-sm font-bold hover:bg-red-600 shadow"
   270→            >✕</button>
   271→          )}
   272→        </div>
   273→      ) : (
   274→        <div
   275→          onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   276→          onDragLeave={() => setDragOver(false)}
   277→          onDrop={(e) => {
   278→            e.preventDefault(); setDragOver(false);
   279→            const f = e.dataTransfer.files?.[0]; if (f) handleFile(f);
   280→          }}
   281→          className={`flex flex-col items-center justify-center w-24 h-24 border-2 border-dashed rounded-lg cursor-pointer transition-all ${
   282→            dragOver ? "border-primary bg-primary/10 scale-105" : "border-slate-300 hover:border-primary hover:bg-slate-50"
   283→          }`}
   284→        >
   285→          <input type="file" accept="image/*" className="hidden" onChange={(e) => {
   286→            const f = e.target.files?.[0]; if (!f) return;
   287→            handleFile(f); e.target.value = "";
   288→          }} />
   289→          <ImageIcon className="w-7 h-7 text-slate-400 mb-1" />
   290→          <span className="text-[10px] text-slate-500 text-center px-1">Drag & drop<br/>or click</span>
   291→        </div>
   292→      )}
   293→    </div>
   294→  );
   295→}

   301→function CreateExamDialog({ open, testCategory, onOpenChange, onCreated }: {
   302→  open: boolean;
   303→  testCategory: string;
   304→  onOpenChange: (v: boolean) => void;
   305→  onCreated: (t: Test) => void;
   306→}) {
   307→  const isSimple = SIMPLE_CATEGORIES.includes(testCategory);
   308→  const [busy, setBusy] = useState(false);
   309→  const [form, setForm] = useState({
   310→    title: "",
   311→    description: "",
   312→    durationMin: 60,
   313→    examType: "UBT",
   314→    category: "",
   315→    price: "",
   316→    featuredImage: "",
   317→    audioPlayMode: "single" as "single" | "double",
   318→    audioGapSec: 2,
   319→    textBlockCount: isSimple ? 0 : 20,
   320→    audioBlockCount: isSimple ? 0 : 20,
   321→  });
   322→
   323→  async function uploadFile(file: File, folder: string): Promise<string> {
   324→    const fd = new FormData();
   325→    fd.append("file", file);
   326→    fd.append("folder", folder);
   327→    const res = await fetch("/api/admin/file-upload", { method: "POST", body: fd });
   328→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || "Upload failed"); }
   329→    const d = await res.json();
   330→    return d.url;
   331→  }
   332→
   333→  async function create() {
   334→    if (!form.title.trim()) { toast.error("Exam name required"); return; }
   335→    setBusy(true);
   336→    try {
   337→      const payload = {
   338→        title: form.title,
   339→        description: form.description,
   340→        durationMin: form.durationMin,
   341→        examType: form.examType,
   342→        testCategory,
   343→        category: form.category || undefined,
   344→        price: form.price ? parseFloat(form.price) : undefined,
   345→        featuredImage: form.featuredImage || undefined,
   346→        audioPlayMode: form.audioPlayMode,
   347→        audioGapSec: form.audioGapSec,
   348→        textBlockCount: form.textBlockCount,
   349→        audioBlockCount: form.audioBlockCount,
   350→        isExam: true,
   351→        isPublished: false, // Draft — admin pushes when ready
   352→      };
   353→      const res = await fetch("/api/admin/tests", {
   354→        method: "POST",
   355→        headers: { "Content-Type": "application/json" },
   356→        body: JSON.stringify(payload),
   357→      });
   358→      const d = await res.json();
   359→      if (!res.ok) { toast.error(d.error || `Failed (HTTP ${res.status})`); return; }
   360→      toast.success("Created — now add questions");
   361→      onCreated(d.test);
   362→    } catch (e: any) {
   363→      toast.error("Create failed: " + (e.message || "network error"));
   364→    } finally { setBusy(false); }
   365→  }
   366→
   367→  const dialogTitle = isSimple
   368→    ? `Create New ${CATEGORY_META[testCategory]?.title || "Set"}`
   369→    : "Create New Exam";
   370→
   371→  return (
   372→    <Dialog open={open} onOpenChange={onOpenChange}>
   373→      <DialogContent className="sm:max-w-4xl max-h-[90vh] overflow-y-auto">
   374→        <DialogHeader><DialogTitle className="text-xl">{dialogTitle}</DialogTitle></DialogHeader>
   375→        <div className="space-y-4">
   376→          {/* Exam Name */}
   377→          <div>
   378→            <Label className="text-sm font-semibold">Name *</Label>
   379→            <Input
   380→              value={form.title}
   381→              onChange={(e) => setForm(f => ({ ...f, title: e.target.value }))}
   382→              placeholder="e.g. UBT Mock Test 1"
   383→              className="h-12 text-base"
   384→            />
   385→          </div>
   386→
   387→          {/* Exam Details */}
   388→          <div>
   389→            <Label className="text-sm font-semibold">Description</Label>
   390→            <Textarea
   391→              rows={2}
   392→              value={form.description}
   393→              onChange={(e) => setForm(f => ({ ...f, description: e.target.value }))}
   394→              placeholder="Brief description of the exam"
   395→            />
   396→          </div>
   397→
   398→          {/* Exam Time + Price */}
   399→          <div className="grid grid-cols-2 gap-3">
   400→            <div>
   401→              <Label className="text-sm font-semibold">Time (minutes) *</Label>
   402→              <Input type="number" value={form.durationMin} onChange={(e) => setForm(f => ({ ...f, durationMin: parseInt(e.target.value) || 60 }))} min={1} />
   403→            </div>
   404→            <div>
   405→              <Label className="text-sm font-semibold">Price (optional)</Label>
   406→              <Input type="number" value={form.price} onChange={(e) => setForm(f => ({ ...f, price: e.target.value }))} placeholder="0 = free" min={0} />
   407→            </div>
   408→          </div>
   409→
   410→          {/* Exam Type + Category */}
   411→          <div className="grid grid-cols-2 gap-3">
   412→            <div>

   448→                    <SelectTrigger><SelectValue /></SelectTrigger>
   449→                    <SelectContent>
   450→                      <SelectItem value="single">Single (play once)</SelectItem>
   451→                      <SelectItem value="double">Double (play twice)</SelectItem>
   452→                    </SelectContent>
   453→                  </Select>
   454→                </div>
   455→                <div>

   458→                </div>
   459→              </div>
   460→            </div>
   461→          )}
   462→
   463→          {/* Block Counts — only for exam & demo */}
   464→          {!isSimple && (
   465→            <div className="grid grid-cols-2 gap-3">
   466→              <div>
   467→                <Label className="text-sm font-semibold">Text Questions Count</Label>

   473→              </div>
   474→            </div>
   475→          )}
   476→
   477→          {/* Simple mode hint */}
   478→          {isSimple && (
   479→            <div className="p-4 border rounded-lg bg-emerald-50 border-emerald-200 text-sm text-emerald-800">
   480→              Questions can be added freely after creation — there is no fixed block layout for this category.
   481→            </div>
   482→          )}
   483→        </div>
   484→
   485→        <div className="flex justify-end gap-2 pt-2 border-t">
   486→          <Button variant="outline" size="lg" onClick={() => onOpenChange(false)}>Cancel</Button>
   487→          <Button size="lg" onClick={create} disabled={busy}>
   488→            {busy ? "Creating…" : <><Plus className="w-4 h-4 mr-1" /> Create</>}
   489→          </Button>
   490→        </div>
   491→      </DialogContent>
   492→    </Dialog>
   493→  );
   494→}

   500→function ExamEditor({ test, testCategory, onClose }: { test: Test; testCategory: string; onClose: () => void }) {
   501→  const isSimple = SIMPLE_CATEGORIES.includes(testCategory);
   502→  const [activeBlock, setActiveBlock] = useState<"text" | "audio">("text");
   503→  const [activeNumber, setActiveNumber] = useState(1);

   910→  async function uploadFile(file: File): Promise<string> {
   911→    const fd = new FormData();
   912→    fd.append("file", file);
   913→    fd.append("folder", "questions");
   914→    const res = await fetch("/api/admin/file-upload", { method: "POST", body: fd });
   915→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || "Upload failed"); }
   916→    const d = await res.json();
   917→    return d.url;
   918→  }

   920→  // ─── Reusable upload field with drag-drop + instant preview ───────────────
   921→  function MediaUpload({
   922→    label, accept, url, onUpload, onClear, type,
   923→  }: {
   924→    label: string;
   925→    accept: string;
   926→    url: string;
   927→    onUpload: (url: string) => void;
   928→    onClear: () => void;
   929→    type: "image" | "audio";
   930→  }) {
   931→    const [uploading, setUploading] = useState(false);
   932→    const [dragOver, setDragOver] = useState(false);
   933→    const [localPreview, setLocalPreview] = useState<string>(""); // instant blob URL
   934→
   935→    // Compress images before upload (max 800px, 80% quality) — much faster
   936→    async function compressImage(file: File): Promise<File> {
   937→      if (file.size < 200_000) return file; // skip small images
   938→      return new Promise((resolve) => {
   939→        const reader = new FileReader();
   940→        reader.onload = (e) => {
   941→          const img = new Image();
   942→          img.onload = () => {
   943→            const canvas = document.createElement("canvas");
   944→            const maxDim = 800;
   945→            let { width, height } = img;
   946→            if (width > maxDim || height > maxDim) {
   947→              if (width > height) { height = (height / width) * maxDim; width = maxDim; }
   948→              else { width = (width / height) * maxDim; height = maxDim; }
   949→            }
   950→            canvas.width = width; canvas.height = height;
   951→            const ctx = canvas.getContext("2d")!;
   952→            ctx.drawImage(img, 0, 0, width, height);
   953→            canvas.toBlob((blob) => {
   954→              if (blob) resolve(new File([blob], file.name.replace(/\.\w+$/, ".jpg"), { type: "image/jpeg" }));
   955→              else resolve(file);
   956→            }, "image/jpeg", 0.8);
   957→          };
   958→          img.src = e.target?.result as string;
   959→        };
   960→        reader.readAsDataURL(file);
   961→      });
   962→    }
   963→
   964→    async function handleFile(file: File) {
   965→      // Instant local preview (no waiting for upload)
   966→      const localUrl = URL.createObjectURL(file);
   967→      setLocalPreview(localUrl);
   968→
   969→      setUploading(true);
   970→      try {
   971→        // Compress images for faster upload
   972→        const fileToUpload = type === "image" ? await compressImage(file) : file;
   973→        const u = await uploadFile(fileToUpload);
   974→        onUpload(u);
   975→        toast.success("Uploaded");
   976→      } catch (err: any) {
   977→        toast.error(err.message || "Upload failed");
   978→        setLocalPreview(""); // clear preview on error
   979→      } finally {
   980→        setUploading(false);
   981→        // Clean up the blob URL after a delay (let React render the real URL)
   982→        setTimeout(() => URL.revokeObjectURL(localUrl), 1000);
   983→      }
   984→    }
   985→
   986→    // The display URL: local preview during upload, real URL after
   987→    const displayUrl = uploading && localPreview ? localPreview : url;
   988→
   989→    return (
   990→      <div className="space-y-2">
   991→        {label && <Label className="text-sm font-medium">{label}</Label>}
   992→
   993→        {displayUrl ? (
   994→          /* ─── Preview shown (either local or uploaded) ─── */
   995→          <div className="flex items-start gap-3">
   996→            {type === "image" ? (
   997→              <div className="relative">
   998→                <img src={displayUrl} alt={label} className="w-28 h-28 rounded-lg object-cover border-2 border-slate-200" />
   999→                {uploading && (
  1000→                  <div className="absolute inset-0 bg-black/40 rounded-lg flex items-center justify-center">
  1001→                    <div className="w-6 h-6 border-2 border-white border-t-transparent rounded-full animate-spin" />
  1002→                  </div>
  1003→                )}
  1004→                {!uploading && (
  1005→                  <button
  1006→                    onClick={() => { onClear(); setLocalPreview(""); }}
  1007→                    className="absolute -top-2 -right-2 w-7 h-7 bg-red-500 text-white rounded-full flex items-center justify-center text-sm font-bold hover:bg-red-600 shadow"
  1008→                    title="Remove"
  1009→                  >✕</button>
  1010→                )}
  1011→              </div>
  1012→            ) : (
  1013→              <div className="relative flex-1">
  1014→                <audio controls src={displayUrl} className="w-full" />
  1015→                {uploading && (
  1016→                  <div className="absolute inset-0 bg-black/20 rounded flex items-center justify-center">
  1017→                    <div className="bg-white px-3 py-1 rounded-full text-xs font-medium flex items-center gap-2">
  1018→                      <div className="w-3 h-3 border-2 border-primary border-t-transparent rounded-full animate-spin" />
  1019→                      Uploading…
  1020→                    </div>
  1021→                  </div>
  1022→                )}
  1023→                {!uploading && (
