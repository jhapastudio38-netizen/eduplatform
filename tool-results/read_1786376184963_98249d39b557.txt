    39→/**

   323→// ─── Google Sign-In button ──────────────────────────────────────────────
   324→// Opens https://my-project-five-sepia.vercel.app/api/auth/google-mobile in
   325→// a Chrome Custom Tab. The OAuth flow completes server-side, then redirects
   326→// to dreamkorea://auth-callback (handled by MainActivity's intent-filter).
   327→//
   328→// Shows the Google logo (R.drawable.google_logo) if present in res/drawable,
   329→// otherwise falls back to text-only ("Sign in with Google").
   330→@Composable
   331→private fun GoogleSignInButton(sound: SoundManager) {
   332→    val context = LocalContext.current
   333→    // Look up the google_logo drawable at runtime — avoids a compile-time
   334→    // dependency on a resource that may or may not exist.
   335→    val googleLogoId = remember {
   336→        context.resources.getIdentifier("google_logo", "drawable", context.packageName)
   337→    }
   338→
   339→    Surface(
   340→        color = Color.White,
   341→        shape = RoundedCornerShape(12.dp),
   342→        border = androidx.compose.foundation.BorderStroke(1.dp, Color(0xFFD1D5DB)),
   343→        modifier = Modifier
   344→            .fillMaxWidth()
   345→            .height(50.dp)
   346→            .clickable {
   347→                sound.click()
   348→                val url = "https://my-project-five-sepia.vercel.app/api/auth/google-mobile"
   349→                try {
   350→                    CustomTabsIntent.Builder()
   351→                        .build()
   352→                        .launchUrl(context, Uri.parse(url))
   353→                } catch (_: Exception) {
   354→                    // Fallback: open in any available browser via ACTION_VIEW
   355→                    try {
   356→                        context.startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(url)))
   357→                    } catch (_: Exception) { /* no browser available — silently ignore */ }
   358→                }
   359→            },
   360→        shadowElevation = 1.dp
   361→    ) {
   362→        Row(
   363→            modifier = Modifier.fillMaxSize().padding(horizontal = 16.dp),
   364→            verticalAlignment = Alignment.CenterVertically,
   365→            horizontalArrangement = Arrangement.Center
   366→        ) {
   367→            if (googleLogoId != 0) {
   368→                Image(
   369→                    painter = painterResource(id = googleLogoId),
   370→                    contentDescription = "Google logo",
   371→                    modifier = Modifier.size(20.dp)
   372→                )
   373→                Spacer(Modifier.width(10.dp))
   374→            }
   375→            Text(
   376→                "Sign in with Google",
   377→                color = Color(0xFF1F2937),
   378→                fontSize = 15.sp,
   379→                fontWeight = FontWeight.Medium
   380→            )
   381→        }
   382→    }
   383→}

   463→        Text(if (step == 1) "Reset Password" else "Enter Code & New Password", color = TextDark, fontSize = 20.sp, fontWeight = FontWeight.Bold)

   465→        Spacer(Modifier.height(16.dp))
   466→
   467→        if (step == 1) {
   468→            Field("Email", email, onEmailChange, Icons.Default.Email, KeyboardType.Email)
   469→            Spacer(Modifier.height(16.dp))
   470→            Button(
   471→                onClick = onRequestCode,

   473→                colors = ButtonDefaults.buttonColors(containerColor = NavyBlue),
   474→                shape = RoundedCornerShape(12.dp),
   475→                enabled = !loading

   561→/**
   562→ * Extract the human-readable error message from a Retrofit HttpException.
   563→ *
   564→ * The server returns errors as JSON: {"error": "Email already registered"}
   565→ * We parse the response body to pull out the "error" field. If parsing fails
   566→ * or the body is empty, we return null so the caller can use a fallback.
   567→ */
   568→private fun extractHttpError(e: retrofit2.HttpException): String? {
   569→    return try {
   570→        val raw = e.response()?.errorBody()?.string()
   571→        if (raw.isNullOrBlank()) return null
   572→        val json = com.google.gson.JsonParser.parseString(raw).asJsonObject
   573→        json.get("error")?.asString
   574→    } catch (_: Exception) {
   575→        null
   576→    }
   577→}
