     9→datasource db {
    10→  // In production (AWS RDS) we use Postgres. In local dev we use SQLite.
    11→  // The DATABASE_URL env var determines which — for dev it's `file:...`,
    12→  // for prod it's `postgresql://...`. Prisma accepts either provider
    13→  // value at runtime; the URL prefix wins.
    14→  provider = "postgresql"
    15→  url      = env("DATABASE_URL")
    16→}

   181→model Question {
   182→  id          String     @id @default(cuid())
   183→  chapterId   String?
   184→  type        QuestionType
   185→  difficulty  Difficulty @default(MEDIUM)
   186→  stem        String // question text (markdown)
   187→  options     String? // JSON array for MCQ
   188→  correctAnswer String? // JSON: string | string[] | {match: ...}
   189→  explanation String?
   190→  tags        String? // JSON array
   191→  imageUrl    String? // Optional image for visual questions
   192→  audioUrl    String? // Optional audio for listening questions
   193→  audioLoop   Int      @default(0) // 0 = no loop, N = loop N times, -1 = infinite
   194→  audioLoopDelay Int   @default(0) // delay (seconds) between loops
   195→  aiGenerated Boolean    @default(false)
   196→  sourceFile  String?
   197→  // Question Bank: standalone practice questions (not tied to a test)
   198→  inQuestionBank Boolean @default(false)
   199→  category    String?  // e.g. "Vocabulary", "Grammar", "Listening", "Reading"
   200→  // ─── New block-based exam fields ────────────────────────────────────────
   201→  blockType   String?  // "text" | "audio" — which block this question belongs to
   202→  blockNumber Int?     // 1-40 (1-20 text block, 21-40 audio block)
   203→  // Set number within a QBank test — admin groups questions into Set 1, Set 2,
   204→  // Set 3 etc. so the student can pick a set to practice. Default 1.
   205→  setNumber   Int?     @default(1)
   206→  // Description type: "none" | "text" | "image" | "audio"
   207→  descType    String?  @default("none")
   208→  descText    String?  // text description (if descType = "text")
   209→  descImageUrl String? // image description (if descType = "image")
   210→  descAudioUrl String? // audio description (if descType = "audio")
   211→  // Question media type: "none" | "text" | "image" | "audio"
   212→  mediaType   String?  @default("none")
   213→  mediaText   String?  // text media (if mediaType = "text")
   214→  mediaImageUrl String? // image media (if mediaType = "image")
   215→  mediaAudioUrl String? // audio media (if mediaType = "audio")
   216→  // Answer type: "text" | "image" | "audio" | "choose"
   217→  answerType  String?  @default("text")
   218→  // For image answers: JSON array of 4 image URLs
   219→  optionImages String? // JSON: ["url1", "url2", "url3", "url4"]
   220→  // For audio answers: JSON array of 4 audio URLs
   221→  optionAudios String? // JSON: ["url1", "url2", "url3", "url4"]
   222→  // Per-option blank/underline word — admin types a word for each option,
   223→  // it gets underlined in the actual option text shown to students.
   224→  // JSON: ["word1", "", "word3", ""] — empty string = no underline
   225→  optionBlanks String?
   226→  // Which option is correct (0-3 index)
   227→  correctOption Int?   @default(0)
   228→  createdAt   DateTime @default(now())
   229→  updatedAt   DateTime @updatedAt
   230→
   231→  chapter     Chapter?  @relation(fields: [chapterId], references: [id], onDelete: SetNull)
   232→  testItems   TestItem[]
   233→
   234→  @@index([chapterId])
   235→  @@index([type, difficulty])
   236→  @@index([inQuestionBank])
   237→  @@index([category])
   238→  @@index([blockType, blockNumber])
   239→}
