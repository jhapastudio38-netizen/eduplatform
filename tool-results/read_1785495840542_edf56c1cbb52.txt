    53→    53→/**

    64→    64→@Composable

    73→    73→    val answers = remember { mutableStateMapOf<String, Any>() }

   121→   121→                when {
   122→   122→                    // Combined QBank exam — fetches ALL published question_bank tests as one test
   123→   123→                    testId == "qbank-combined" -> {
   124→   124→                        try {
   125→   125→                            AppState.api.getQBankCombined().test
   126→   126→                        } catch (e: retrofit2.HttpException) {
   127→   127→                            if (e.code() == 404) buildQBankCombinedClientSide()
   128→   128→                            else throw e
   129→   129→                        }
   130→   130→                    }
   131→   131→                    // Combined bundle exam — fetches ALL tests in a specific bundle (qbank/batch)
   132→   132→                    testId.startsWith("bundle-") -> {
   133→   133→                        val bundleId = testId.removePrefix("bundle-")

   135→   135→                            AppState.api.getBundleCombined(bundleId).test
   136→   136→                        } catch (e: retrofit2.HttpException) {
   137→   137→                            if (e.code() == 404) buildBundleCombinedClientSide(bundleId)
   138→   138→                            else throw e
   139→   139→                        }
   140→   140→                    }

   142→   142→                    else -> AppState.api.getTestDetail(testId).test

   155→   155→                401 -> "Your session has expired. Please log out and sign in again."

   159→   159→                else -> "Could not load the test (HTTP ${e.code()}).${if (rawBody != null) " $rawBody" else ""}"
   160→   160→            }
   161→   161→        } catch (e: java.net.UnknownHostException) {
   162→   162→            error = "No internet connection. Please check your network and try again."

   164→   164→            error = "The request timed out. Please try again."
   165→   165→        } catch (e: java.io.IOException) {
   166→   166→            error = "Network error: ${e.message ?: "Could not connect to server."}"
   167→   167→        } catch (e: Exception) {
   168→   168→            error = "Unexpected error: ${e.message ?: "Please try again."}"

   191→   191→                    } else {
   192→   192→                        submitExamWithFallback(currentTest, answers.toMap())
   193→   193→                    }
   194→   194→                    sound.success()
   195→   195→                } catch (e: java.net.UnknownHostException) {
   196→   196→                    error = "No internet connection. Could not submit your answers."
   197→   197→                } catch (e: java.io.IOException) {
   198→   198→                    error = "Network error. Could not submit your answers."
   199→   199→                } catch (e: Exception) {
   200→   200→                    error = "Could not submit: ${e.message ?: "Unknown error"}"
   201→   201→                }
   202→   202→                submitting = false
   203→   203→            }
   204→   204→        }
   205→   205→    }

   214→   214→            )

   216→   216→        }
   217→   217→        return
   218→   218→    }
   219→   219→
   220→   220→    if (error.isNotEmpty()) {

   222→   222→            Modifier.fillMaxSize().padding(24.dp),
   223→   223→            horizontalAlignment = Alignment.CenterHorizontally,
   224→   224→            verticalArrangement = Arrangement.Center
   225→   225→        ) {
   226→   226→            Icon(
   227→   227→                Icons.Default.CloudOff,
   228→   228→                null,
   229→   229→                tint = theme.errorRed.copy(alpha = 0.7f),
   230→   230→                modifier = Modifier.size(56.dp)
   231→   231→            )
   232→   232→            Spacer(Modifier.height(16.dp))
   233→   233→            Text(
   234→   234→                "Couldn't load the test",
   235→   235→                color = theme.darkText,
   236→   236→                fontSize = 18.sp,
   237→   237→                fontWeight = FontWeight.Bold
   238→   238→            )
   239→   239→            Spacer(Modifier.height(8.dp))
   240→   240→            Text(
   241→   241→                error,
   242→   242→                color = theme.subText,
   243→   243→                fontSize = 13.sp,
   244→   244→                textAlign = TextAlign.Center,
   245→   245→                modifier = Modifier.padding(horizontal = 16.dp)
   246→   246→            )
   247→   247→            Spacer(Modifier.height(8.dp))
   248→   248→            Text(
   249→   249→                "Test ID: $testId",
   250→   250→                color = theme.subText.copy(alpha = 0.5f),
   251→   251→                fontSize = 10.sp,
   252→   252→                fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace
   253→   253→            )
   254→   254→            Spacer(Modifier.height(24.dp))
   255→   255→            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
   256→   256→                OutlinedButton(
   257→   257→                    onClick = onExit,
   258→   258→                    shape = RoundedCornerShape(10.dp)
   259→   259→                ) { Text("Go back") }
   260→   260→                Button(
   261→   261→                    onClick = {
   262→   262→                        sound.click()
   263→   263→                        retryCount++ // triggers LaunchedEffect reload
   264→   264→                    },
   265→   265→                    colors = ButtonDefaults.buttonColors(containerColor = theme.primary),
   266→   266→                    shape = RoundedCornerShape(10.dp)

   268→   268→                    Icon(Icons.Default.Refresh, null, modifier = Modifier.size(18.dp))

   289→   289→            Button(onClick = onExit, colors = ButtonDefaults.buttonColors(containerColor = theme.primary)) { Text("Go back") }

   293→   293→
   294→   294→    val currentQuestion = t.items.getOrNull(currentIdx)

   296→   296→    // ─── Result screen ──────────────────────────────────────────────────────

   846→   846→/// Reference-style 4-column grid of PERFECT SQUARE cells with 2px black borders.
   847→   847→/// Solved = blue fill + white text. Current = blue border + glow.
   848→   848→/// Uses continuous numbering: Reading 1-20, Listening 21-40 (from globalIdx).
   849→   849→/// showAllBlocks: true = pad grid to expected count (20) with blank cells so
   850→   850→/// the user sees which questions are added vs blank.
   851→   851→/// false = only show cells for questions that actually exist.
   852→   852→/// Cells ALWAYS maintain 1:1 aspect ratio (perfect square) regardless of count.
   853→   853→/// Grid is scrollable so all blocks are reachable.
   854→   854→@Composable
   855→   855→private fun QuestionGridRef(
   856→   856→    test: TestDetail,
   857→   857→    items: List<TestItemDetail>,
   858→   858→    answers: SnapshotStateMap<String, Any>,
   859→   859→    currentIdx: Int,
   860→   860→    sound: SoundManager,
   861→   861→    haptic: androidx.compose.ui.hapticfeedback.HapticFeedback,
   862→   862→    filterMode: Boolean?,
   863→   863→    accentBlue: Color,
   864→   864→    showAllBlocks: Boolean,
   865→   865→    onPick: (Int) -> Unit,
   866→   866→) {
   867→   867→    val globalIndices = items.mapNotNull { item -> test.items.indexOfFirst { it.question.id == item.question.id }.takeIf { it >= 0 } }
   868→   868→    val cols = 4  // 4 columns per the HTML reference
   869→   869→    val gridScrollState = rememberScrollState()
   870→   870→
   871→   871→    // Determine total cells to render
   872→   872→    val expectedTotal = if (showAllBlocks) {
   873→   873→        val isAudio = items.isNotEmpty() && items[0].question.blockType == "audio"
   874→   874→        val blockCount = if (isAudio) test.audioBlockCount else test.textBlockCount
   875→   875→        maxOf(blockCount, items.size)
   876→   876→    } else {
   877→   877→        items.size
   878→   878→    }
   879→   879→    val rowsCount = (expectedTotal + cols - 1) / cols
   880→   880→
   881→   881→    // Scrollable grid — cells are PERFECT SQUARES (aspectRatio 1f)
   882→   882→    Column(
   883→   883→        modifier = Modifier
   884→   884→            .fillMaxWidth()
   885→   885→            .verticalScroll(gridScrollState),
   886→   886→        verticalArrangement = Arrangement.spacedBy(8.dp)
   887→   887→    ) {
   888→   888→        for (rowIdx in 0 until rowsCount) {
   889→   889→            Row(
   890→   890→                modifier = Modifier.fillMaxWidth(),
   891→   891→                horizontalArrangement = Arrangement.spacedBy(8.dp)
   892→   892→            ) {
   893→   893→                for (colIdx in 0 until cols) {
   894→   894→                    val localIdx = rowIdx * cols + colIdx
   895→   895→                    if (localIdx < items.size) {
   896→   896→                        val globalIdx = globalIndices[localIdx]
   897→   897→                        // DISPLAY NUMBER: Reading shows 1-20, Listening shows 21-40
   898→   898→                        // Uses blockNumber (1-20 within each block) + 20 offset for audio
   899→   899→                        val q = items[localIdx].question
   900→   900→                        val displayNum = if (q.blockType == "audio") {
   901→   901→                            (q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)) + 20
   902→   902→                        } else {
   903→   903→                            q.blockNumber.takeIf { it > 0 } ?: (localIdx + 1)
   904→   904→                        }
   905→   905→                        val isAnswered = answers.containsKey(items[localIdx].question.id)
   906→   906→                        val isCurrent = globalIdx == currentIdx
   907→   907→                        val isFilteredOut = when (filterMode) {
   908→   908→                            true -> !isAnswered
   909→   909→                            false -> isAnswered
   910→   910→                            null -> false
   911→   911→                        }
   912→   912→                        Box(
   913→   913→                            modifier = Modifier
   914→   914→                                .weight(1f)
   915→   915→                                .aspectRatio(1f)  // PERFECT SQUARE — always 1:1
   916→   916→                                .clip(RoundedCornerShape(6.dp))
   917→   917→                                .border(
   918→   918→                                    width = if (isCurrent) 2.5.dp else 1.5.dp,
   919→   919→                                    color = when {
   920→   920→                                        isCurrent -> accentBlue
   921→   921→                                        isAnswered -> accentBlue
   922→   922→                                        else -> Color(0xFF111111)
   923→   923→                                    },
   924→   924→                                    shape = RoundedCornerShape(6.dp)
   925→   925→                                )
   926→   926→                                .background(if (isAnswered) accentBlue else Color.White)
   927→   927→                                .alpha(if (isFilteredOut) 0.15f else 1f)
   928→   928→                                .clickable(enabled = !isFilteredOut) {
   929→   929→                                    sound.click()
   930→   930→                                    haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
   931→   931→                                    onPick(globalIdx)
   932→   932→                                },
   933→   933→                            contentAlignment = Alignment.Center
   934→   934→                        ) {
   935→   935→                            Text(
   936→   936→                                "$displayNum",
   937→   937→                                color = if (isAnswered) Color.White else Color(0xFF111111),
   938→   938→                                fontSize = 15.sp,
   939→   939→                                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Medium,
   940→   940→                            )
   941→   941→                        }
   942→   942→                    } else if (showAllBlocks) {
   943→   943→                        // Empty placeholder cell — perfect square
   944→   944→                        // Reading placeholders: 1-20, Listening placeholders: 21-40
   945→   945→                        val isAudioGrid = items.isNotEmpty() && items[0].question.blockType == "audio"
   946→   946→                        val placeholderNum = if (isAudioGrid) localIdx + 21 else localIdx + 1
   947→   947→                        Box(
   948→   948→                            modifier = Modifier
   949→   949→                                .weight(1f)
   950→   950→                                .aspectRatio(1f)  // PERFECT SQUARE
   951→   951→                                .clip(RoundedCornerShape(6.dp))
   952→   952→                                .border(1.5.dp, Color(0xFFEEEEEE), RoundedCornerShape(6.dp))
   953→   953→                                .background(Color(0xFFFAFAFA)),
   954→   954→                            contentAlignment = Alignment.Center
   955→   955→                        ) {
   956→   956→                            Text(
   957→   957→                                "$placeholderNum",
   958→   958→                                color = Color(0xFFCCCCCC),
   959→   959→                                fontSize = 13.sp,
   960→   960→                                fontWeight = FontWeight.Normal,
   961→   961→                            )
   962→   962→                        }
   963→   963→                    } else {
   964→   964→                        // showAllBlocks=false, no question — invisible spacer (perfect square)
   965→   965→                        Spacer(modifier = Modifier.weight(1f).aspectRatio(1f))
   966→   966→                    }
   967→   967→                }
   968→   968→            }
   969→   969→        }

  1010→  1010→@Composable
  1011→  1011→fun AudioPlayerCard(
  1012→  1012→    theme: AppTheme,
  1013→  1013→    url: String,
  1014→  1014→    loopCount: Int,
  1015→  1015→    loopDelaySec: Int,
  1016→  1016→    sound: SoundManager,
  1017→  1017→    questionId: String? = null,
  1018→  1018→    playCounts: SnapshotStateMap<String, Int>? = null,
  1019→  1019→) {
  1020→  1020→    val context = LocalContext.current
  1021→  1021→    var mediaPlayer by remember { mutableStateOf<android.media.MediaPlayer?>(null) }
  1022→  1022→    var isPlaying by remember { mutableStateOf(false) }
  1023→  1023→
  1024→  1024→    // PERSISTENT play count — stored in parent map (survives navigation).
  1025→  1025→    // Prevents cheat where student navigates away and back to reset plays.
  1026→  1026→    val persistentCount = playCounts?.get(questionId) ?: 0

  1028→  1028→    val disabled = persistentCount >= maxPlays
  1029→  1029→    val scope = rememberCoroutineScope()
  1030→  1030→
  1031→  1031→    fun incrementPlayCount() {
  1032→  1032→        if (questionId != null && playCounts != null) {
  1033→  1033→            playCounts[questionId] = (playCounts[questionId] ?: 0) + 1
  1034→  1034→        }
  1035→  1035→    }
  1036→  1036→
  1037→  1037→    DisposableEffect(url) {
  1038→  1038→        onDispose {
  1039→  1039→            mediaPlayer?.release()
  1040→  1040→            mediaPlayer = null
  1041→  1041→        }
  1042→  1042→    }
  1043→  1043→
  1044→  1044→    Surface(color = theme.cardBg, shape = RoundedCornerShape(12.dp), modifier = Modifier.fillMaxWidth(), shadowElevation = 1.dp) {
  1045→  1045→        Row(
  1046→  1046→            modifier = Modifier.padding(14.dp),
  1047→  1047→            verticalAlignment = Alignment.CenterVertically
  1048→  1048→        ) {
  1049→  1049→            // Audio icon
  1050→  1050→            Surface(
  1051→  1051→                color = if (disabled) Color(0xFFE2E8F0) else theme.primary.copy(alpha = 0.15f),
  1052→  1052→                shape = RoundedCornerShape(10.dp),
  1053→  1053→                modifier = Modifier.size(44.dp)
  1054→  1054→            ) {
  1055→  1055→                Box(contentAlignment = Alignment.Center, modifier = Modifier.fillMaxSize()) {
  1056→  1056→                    Icon(
  1057→  1057→                        Icons.Default.Headphones,
  1058→  1058→                        null,
  1059→  1059→                        tint = if (disabled) Color(0xFF94A3B8) else theme.primary,
  1060→  1060→                        modifier = Modifier.size(22.dp)
  1061→  1061→                    )
  1062→  1062→                }
  1063→  1063→            }
  1064→  1064→            Spacer(Modifier.width(12.dp))

  1068→  1068→                    if (disabled) "Audio played ($maxPlays/$maxPlays)" else "Audio — Play $persistentCount / $maxPlays",
  1069→  1069→                    color = if (disabled) Color(0xFF94A3B8) else theme.darkText,
  1070→  1070→                    fontSize = 13.sp,
  1071→  1071→                    fontWeight = FontWeight.SemiBold
  1072→  1072→                )
  1073→  1073→                if (!disabled) {
  1074→  1074→                    Text(

  1076→  1076→                        color = theme.subText, fontSize = 11.sp
  1077→  1077→                    )
  1078→  1078→                } else {

  1080→  1080→                }
  1081→  1081→            }
  1082→  1082→            // Play/Pause button — DISABLED after all plays used

  1096→  1096→                                    start()

  1098→  1098→                                    incrementPlayCount()

  1100→  1100→                                setOnCompletionListener {
  1101→  1101→                                    val currentCount = playCounts?.get(questionId) ?: 0
  1102→  1102→                                    if (currentCount < maxPlays) {
  1103→  1103→                                        scope.launch {
  1104→  1104→                                            if (loopDelaySec > 0) delay(loopDelaySec * 1000L)
  1105→  1105→                                            val latestCount = playCounts?.get(questionId) ?: 0

  1124→  1124→                        } catch (_: Exception) {
  1125→  1125→                            isPlaying = false

  1137→  1137→                    null,

  1139→  1139→                    modifier = Modifier.size(32.dp)
  1140→  1140→                )
  1141→  1141→            }
  1142→  1142→        }
  1143→  1143→    }
