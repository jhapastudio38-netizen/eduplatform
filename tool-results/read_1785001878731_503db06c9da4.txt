   347→   347→function ExamEditor({ test, onClose }: { test: Test; onClose: () => void }) {
   348→   348→  const [activeBlock, setActiveBlock] = useState<"text" | "audio">("text");
   349→   349→  const [activeNumber, setActiveNumber] = useState(1);
   350→   350→  const [questions, setQuestions] = useState<Record<string, QuestionData>>({});
   351→   351→  const [loading, setLoading] = useState(true);
   352→   352→  const [clipboard, setClipboard] = useState<string>("");
   353→   353→  const [showPasteDialog, setShowPasteDialog] = useState(false);
   354→   354→  const [pasteCode, setPasteCode] = useState("");
   355→   355→  const [pushing, setPushing] = useState(false);
   356→   356→  const [isPublished, setIsPublished] = useState(test.isPublished);
   357→   357→
   358→   358→  const textCount = test.textBlockCount || 20;
   359→   359→  const audioCount = test.audioBlockCount || 20;
   360→   360→
   361→   361→  function key(blockType: string, blockNumber: number) {
   362→   362→    return `${blockType}-${blockNumber}`;
   363→   363→  }
   364→   364→
   365→   365→  useEffect(() => {
   366→   366→    // Load existing questions
   367→   367→    fetch(`/api/admin/tests/${test.id}/questions`)
   368→   368→      .then((r) => r.json())
   369→   369→      .then((d) => {
   370→   370→        const map: Record<string, QuestionData> = {};
   371→   371→        for (const q of d.questions || []) {
   372→   372→          map[key(q.blockType, q.blockNumber)] = q;
   373→   373→        }
   374→   374→        setQuestions(map);
   375→   375→      })
   376→   376→      .finally(() => setLoading(false));
   377→   377→  }, [test.id]);
   378→   378→
   379→   379→  const currentKey = key(activeBlock, activeNumber);
   380→   380→  const currentQuestion = questions[currentKey] || emptyQuestion(activeBlock, activeNumber);
   381→   381→
   382→   382→  function updateQuestion(q: QuestionData) {
   383→   383→    setQuestions((prev) => ({ ...prev, [currentKey]: q }));
   384→   384→  }
   385→   385→
   386→   386→  async function saveQuestion() {
   387→   387→    const q = currentQuestion;
   388→   388→    if (!q.stem.trim()) { toast.error("Question text required"); return; }
   389→   389→    try {
   390→   390→      const res = await fetch(`/api/admin/tests/${test.id}/questions`, {
   391→   391→        method: "POST",
   392→   392→        headers: { "Content-Type": "application/json" },
   393→   393→        body: JSON.stringify(q),
   394→   394→      });
   395→   395→      if (!res.ok) { const d = await res.json(); toast.error(d.error || "Save failed"); return; }
   396→   396→      toast.success(`Question ${q.blockNumber} saved`);
   397→   397→    } catch {
   398→   398→      toast.error("Save failed");
   399→   399→    }
   400→   400→  }
   401→   401→
   402→   402→  function copyQuestion() {
   403→   403→    const q = currentQuestion;
   404→   404→    if (!q.stem.trim()) { toast.error("Nothing to copy — question is empty"); return; }
   405→   405→    // Generate a copy code — admin can define their own prefix
   406→   406→    const code = `DK-${activeBlock.toUpperCase()}-${q.blockNumber}-${Date.now().toString(36).toUpperCase()}`;
   407→   407→    const data = JSON.stringify({ code, question: q });
   408→   408→    // Store in localStorage so admin can paste later
   409→   409→    const allCopies = JSON.parse(localStorage.getItem("dk_copies") || "{}");
   410→   410→    allCopies[code] = data;
   411→   411→    localStorage.setItem("dk_copies", JSON.stringify(allCopies));
   412→   412→    setClipboard(code);
   413→   413→    // Copy code to clipboard
   414→   414→    navigator.clipboard.writeText(code);
   415→   415→    toast.success(`Copied! Code: ${code}`);
   416→   416→  }
   417→   417→
   418→   418→  function pasteQuestion() {
   419→   419→    setShowPasteDialog(true);
   420→   420→  }
   421→   421→
   422→   422→  async function pushToApp() {
   423→   423→    const filledCount = Object.values(questions).filter(q => q.stem.trim()).length;
   424→   424→    if (filledCount === 0) {
   425→   425→      toast.error("Cannot push: add at least one question first");
   426→   426→      return;
   427→   427→    }
   428→   428→    if (!confirm(`Push this exam to all student apps?\n\n${filledCount} question(s) will be live.\n\nStudents will see it in their Exams page immediately.`)) {
   429→   429→      return;
   430→   430→    }
